```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.4;

import "forge-std/Test.sol";

interface IRISING {
    function owner() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function enableAntiwhale(bool value) external;
    function setMaxTxPercent(uint256 maxTxPercent) external;
    function isExcludedFromTXLimit(address account) external view returns (bool);
    function _maxTxAmount() external view returns (uint256);
    function approve(address spender, uint256 amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
}

contract ExploitTest is Test {
    IRISING public constant token = IRISING(0x33555617F481be5973E93b432d8ecEa2D20524be);
    address public owner;
    address public user = address(0xBEEF);
    uint256 public constant MIN_TX_PERCENT = 1; // 0.01%? Actually basis points, 1 = 0.01% of total supply
    uint256 public constant TRANSFER_AMOUNT = 2 * 10**15; // 0.0002% of supply, but we set balance higher

    function setUp() public {
        // Get the current owner from the contract
        owner = token.owner();
        vm.label(owner, "Owner");
        vm.label(user, "User");

        // Fund the user with some tokens from the owner
        vm.startPrank(owner);
        uint256 userBalance = 2 * 10**16; // 20,000 tokens (decimals 9 => 20*10^9? better to use raw)
        // Actually total supply is 10^11 * 10^9 = 10^20 wei (100 billion tokens)
        // We'll give user 2 * 10^16 wei = 20,000,000 tokens? That's fine.
        token.transfer(user, userBalance);
        vm.stopPrank();

        // Verify user received tokens
        assertEq(token.balanceOf(user), userBalance);
    }

    function test_OwnerCanStopTransactionsForUsers() public {
        // Before attack: user can transfer normally
        uint256 transferAmount = 10**15; // small amount
        vm.startPrank(user);
        token.transfer(address(0x1234), transferAmount);
        vm.stopPrank();

        // Now the owner sets max tx limit to the smallest possible amount (0.01% of total supply)
        vm.prank(owner);
        token.enableAntiwhale(true);
        vm.prank(owner);
        token.setMaxTxPercent(MIN_TX_PERCENT);

        // Check the new maxTxAmount
        uint256 maxTxAmount = token._maxTxAmount();
        assertEq(maxTxAmount, 10**20 * MIN_TX_PERCENT / 10000); // _tTotal=10^20 => 10^20 / 10000 = 10^16

        // Now user tries to transfer an amount larger than maxTxAmount - should revert
        vm.startPrank(user);
        vm.expectRevert("Transfer amount exceeds the maxTxAmount.");
        token.transfer(address(0xDEAD), 2 * maxTxAmount); // 2 * 10^16
        vm.stopPrank();

        // User can still transfer <= maxTxAmount
        vm.startPrank(user);
        token.transfer(address(0xCAFE), maxTxAmount / 2);
        vm.stopPrank();

        // Owner is excluded from the tx limit, so can transfer anything
        vm.startPrank(owner);
        token.transfer(address(0xABCD), 100 * maxTxAmount); // huge transfer, should work
        vm.stopPrank();

        // Assert that the transfer succeeded (balance of receiver increased)
        assertGt(token.balanceOf(address(0xABCD)), 0);

        // The user's remaining balance after attack is still significant but they cannot transfer large amounts
        // Demonstrating that the owner has effectively stopped large transactions for non-owner users
    }

    function test_OwnerCanStillTransfer() public {
        // After restriction, owner can transfer any amount
        vm.prank(owner);
        token.enableAntiwhale(true);
        vm.prank(owner);
        token.setMaxTxPercent(MIN_TX_PERCENT);

        uint256 ownerBalance = token.balanceOf(owner);
        vm.startPrank(owner);
        token.transfer(user, ownerBalance);
        vm.stopPrank();
        // No revert
        assertEq(token.balanceOf(user), ownerBalance);
    }
}
```
