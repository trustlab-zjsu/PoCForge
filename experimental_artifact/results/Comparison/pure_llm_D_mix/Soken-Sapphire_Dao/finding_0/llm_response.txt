```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface ISapToken {
    function owner() external view returns (address);
    function vault() external view returns (address);
    function setVault(address vault_) external returns (bool);
    function mint(address account_, uint256 amount_) external;
    function balanceOf(address account) external view returns (uint256);
    function totalSupply() external view returns (uint256);
}

contract ExploitTest is Test {
    // Deployed SapToken contract on BSC
    address constant SAP_TOKEN = 0x039639Ed22A7654f74C1B0017C9d0DF85C26B63B;

    ISapToken sapToken;

    function setUp() public {
        sapToken = ISapToken(SAP_TOKEN);
        // Label for better traces
        vm.label(SAP_TOKEN, "SapToken");
    }

    function testSetVaultToZeroAddress() public {
        // Get current owner
        address owner = sapToken.owner();
        vm.label(owner, "Owner");

        // Ensure vault is not already zero (if it were, exploit already happened)
        address initialVault = sapToken.vault();
        assertTrue(initialVault != address(0), "Precondition: vault is already zero");

        // Impersonate owner and set vault to zero address
        vm.prank(owner);
        bool success = sapToken.setVault(address(0));
        assertTrue(success, "setVault should succeed");

        // Verify vault is now address(0)
        address newVault = sapToken.vault();
        assertEq(newVault, address(0), "Vault should be zero address after setVault call");

        // Now show that mint becomes impossible because onlyVault modifier fails
        // The onlyVault modifier requires msg.sender == _vault. _vault is now zero address.
        // Calling mint from any address (including zero) will revert because zero address cannot be msg.sender (or not equal).
        vm.expectRevert("VaultOwned: caller is not the Vault");
        // Try to call mint from a random address (not zero, but vault is zero, so it reverts)
        vm.prank(address(0xDEAD));
        sapToken.mint(address(this), 1 ether);

        // Also verify that no mint can happen even from zero address (though zero address can't be pranked in Foundry)
        // This suffices to demonstrate the vulnerability.
    }
}
```
