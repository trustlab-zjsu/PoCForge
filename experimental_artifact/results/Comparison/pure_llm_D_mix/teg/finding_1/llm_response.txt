```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.4;

import "forge-std/Test.sol";

// Minimal interface for TEGPAY contract
interface ITEGPAY {
    function owner() external view returns (address);
    function _taxFee() external view returns (uint256);
    function _developmentFee() external view returns (uint256);
    function _liquidityFee() external view returns (uint256);
    function setTaxFeePercent(uint256 taxFee) external;
    function setDevelopmentFeePercent(uint256 developmentFee) external;
    function setLiquidityFeePercent(uint256 liquidityFee) external;
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function isExcludedFromFee(address account) external view returns (bool);
}

contract ExploitTest is Test {
    address constant TEGPAY = 0xc35B753d5F2e86cae7C42f755eb1812855aB10c0;
    address constant OWNER = 0xBd51b6D98574327734C1621A7283778c5e6Af86e;
    address constant USER = address(0x1234);

    ITEGPAY tegpay;

    function setUp() public {
        tegpay = ITEGPAY(TEGPAY);
        vm.label(TEGPAY, "TEGPAY");
        vm.label(OWNER, "Owner");
        vm.label(USER, "User");

        // Fund user with some TEGPAY tokens (via deal)
        uint256 amount = 1_000_000 * 1e18;
        deal(TEGPAY, USER, amount);
    }

    function test_OwnerCanSetFeesExceeding25Percent() public {
        // Pre-condition: fees are low (initial values: 30,20,70 -> 0.3%,0.2%,0.7% = all <250)
        assertLt(tegpay._taxFee(), 250);
        assertLt(tegpay._developmentFee(), 250);
        assertLt(tegpay._liquidityFee(), 250);

        // Impersonate owner and set all three fees to 1000 (100%)
        vm.startPrank(OWNER);
        tegpay.setTaxFeePercent(1000);
        tegpay.setDevelopmentFeePercent(1000);
        tegpay.setLiquidityFeePercent(1000);
        vm.stopPrank();

        // Verify fees are now above the 25% threshold (250)
        assertGt(tegpay._taxFee(), 250);
        assertGt(tegpay._developmentFee(), 250);
        assertGt(tegpay._liquidityFee(), 250);

        // Confirm values are exactly 1000
        assertEq(tegpay._taxFee(), 1000);
        assertEq(tegpay._developmentFee(), 1000);
        assertEq(tegpay._liquidityFee(), 1000);
    }

    function test_NonOwnerCannotSetFees() public {
        // Try calling setter as user - should revert
        vm.startPrank(USER);
        vm.expectRevert("Ownable: caller is not the owner");
        tegpay.setTaxFeePercent(250);
        vm.expectRevert("Ownable: caller is not the owner");
        tegpay.setDevelopmentFeePercent(250);
        vm.expectRevert("Ownable: caller is not the owner");
        tegpay.setLiquidityFeePercent(250);
        vm.stopPrank();
    }

    function test_HighFeesCauseExcessiveDeductionOnTransfer() public {
        // Set fees to 1000 (100%) as owner
        vm.prank(OWNER);
        tegpay.setTaxFeePercent(1000);
        vm.prank(OWNER);
        tegpay.setDevelopmentFeePercent(1000);
        vm.prank(OWNER);
        tegpay.setLiquidityFeePercent(1000);

        // User balance before
        uint256 balanceBefore = tegpay.balanceOf(USER);
        uint256 transferAmount = 1000 * 1e18;
        address recipient = address(0xdead);

        // User sends tokens
        vm.prank(USER);
        tegpay.transfer(recipient, transferAmount);

        // Expected: fee total = 1000 + 1000 + 1000 = 3000 out of 1000 (since 1000 per thousand means 300%? Actually per mille: fee = amount * fee% / 1000.
        // So with tax=1000, dev=1000, liq=1000, sum=3000, so tTransferAmount = amount * (1000 - 3000)/1000 = -2000, but underflow? In _getTValues, it calculates tFee = tAmount * _taxFee /1000 = 1000e18, tLiquidity same, tDevelopment same, total=3000e18, then tTransferAmount = tAmount.sub(tFee).sub(tLiquidity).sub(tDevelopment) => 1000e18 - 1000e18 -1000e18 -1000e18 = -2000e18 -> revert due to SafeMath sub with underflow.
        // So actual transfer reverts. That demonstrates extreme manipulation.

        vm.expectRevert();
        vm.prank(USER);
        tegpay.transfer(recipient, transferAmount);

        // Alternatively test with lower fees to show >25% deduction.
        // Reset fees to for example 300 (30%) each, total 90% deduction.
        vm.startPrank(OWNER);
        tegpay.setTaxFeePercent(300);
        tegpay.setDevelopmentFeePercent(300);
        tegpay.setLiquidityFeePercent(300);
        vm.stopPrank();

        uint256 amount2 = 1000 * 1e18;
        vm.prank(USER);
        tegpay.transfer(recipient, amount2);

        uint256 balanceAfter = tegpay.balanceOf(USER);
        // Original user had 1_000_000e18, after first transfer (which reverted?) Actually the first transfer with 100% fees reverted, so user still has 1000000e18.
        // After this transfer, balance should drop by amount2 (1000e18) plus fees? Wait, user does transfer, the amount sent to recipient is reduced by fees. But user's balance reduces by the full transfer amount specified (tAmount). The recipient gets less. So user's balanceAfter = balanceBefore - amount2 = 999,000e18.
        // That is fine. But the fee deduction is >25% of the transferred tokens. Let's check the recipient's received amount.
        uint256 recipientBalance = tegpay.balanceOf(recipient);
        // For each fee = 300 (30%), total fee = 90% = 900e18, recipient gets 100e18. So 90% > 25%.
        assertLt(recipientBalance, amount2 * 25 / 100); // recipient got less than 25% of the intended transfer
    }
}
```
