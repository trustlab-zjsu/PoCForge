```solidity
// Exploit.t.sol
pragma solidity ^0.8.19;
import "forge-std/Test.sol";

interface IStarlinkSatire {
    function owner() external view returns (address);
    function tradeEnabled() external view returns (bool);
    function openTrade() external;
    function renounceOwnership() external;
}

contract ExploitTest is Test {
    address constant TARGET = 0x21032d09cb4f8949883bdef62d3f1eb5288259bc;
    IStarlinkSatire target = IStarlinkSatire(TARGET);

    function setUp() public {
        vm.label(TARGET, "StarlinkSatire_BEP20");
    }

    function test_IrreversibleTradeEnable() public {
        // Identify the contract owner
        address owner = target.owner();
        vm.label(owner, "Owner");

        // Initially trade is disabled
        assertFalse(target.tradeEnabled());

        // Owner enables trade
        vm.prank(owner);
        target.openTrade();

        // Trade is now enabled
        assertTrue(target.tradeEnabled());

        // Owner cannot disable – calling openTrade again has no effect
        vm.prank(owner);
        target.openTrade();
        assertTrue(target.tradeEnabled());

        // Even after renouncing ownership, trade remains enabled
        vm.prank(owner);
        target.renounceOwnership();
        assertTrue(target.tradeEnabled());

        // The new owner (address(0)) cannot call openTrade anymore
        vm.prank(address(0));
        vm.expectRevert("Ownable: caller is not the owner");
        target.openTrade();

        // tradeEnabled is permanently true, confirming irreversibility
        assertTrue(target.tradeEnabled());
    }
}
```
