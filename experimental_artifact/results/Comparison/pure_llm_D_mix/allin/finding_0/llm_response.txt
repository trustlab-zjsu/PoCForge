```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.9;

import "forge-std/Test.sol";

interface IAllin {
    function getOwner() external view returns (address);
    function tradingStatus(bool _status) external;
    function transfer(address recipient, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
}

contract ExploitTest is Test {
    address constant ALLIN = 0x05cd34a1d5df5010f5edb3412077b3c291ddb7a8;
    IAllin public allin;
    address public owner;

    function setUp() public {
        allin = IAllin(ALLIN);
        owner = allin.getOwner();
        vm.label(owner, "Owner");
        vm.label(ALLIN, "Allin");

        // Ensure trading is open initially (owner must open it)
        vm.prank(owner);
        allin.tradingStatus(true);
    }

    function testOwnerCanStopTransactions() public {
        address alice = address(0x1234);
        address bob = address(0x5678);
        vm.label(alice, "Alice");
        vm.label(bob, "Bob");

        // Owner closes trading for everyone
        vm.prank(owner);
        allin.tradingStatus(false);

        // Non-owner (Alice) tries to transfer – reverts
        vm.prank(alice);
        vm.expectRevert("Trading not open yet");
        allin.transfer(bob, 1);

        // Owner can still transfer – give owner tokens via storage
        // _balances mapping is at slot 3 (after _totalSupply, _maxTxAmount, _maxWalletToken)
        bytes32 slot = keccak256(abi.encode(owner, uint256(3)));
        vm.store(ALLIN, slot, bytes32(uint256(1000)));

        vm.prank(owner);
        bool success = allin.transfer(bob, 100);
        assertTrue(success);

        // Verify Bob received the tokens
        assertEq(allin.balanceOf(bob), 100);
    }
}
```
