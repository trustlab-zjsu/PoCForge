```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.9;

import "forge-std/Test.sol";

interface IPowabitToken {
    function owner() external view returns (address);
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function mint(uint256 amount) external returns (bool);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
    function approve(address spender, uint256 amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function decimals() external view returns (uint8);
    function symbol() external view returns (string memory);
    function name() external view returns (string memory);
}

contract ExploitTest is Test {
    // Deployed contract address (BSC)
    address constant POWABIT = 0x5FDa925294F9e5B31b7c0e81Bad6BAD5b5E792eE;

    IPowabitToken powa = IPowabitToken(POWABIT);

    function setUp() public {
        vm.label(POWABIT, "PowabitToken");
    }

    function testUnlimitedMintByOwner() public {
        // Get the current owner of the contract
        address owner = powa.owner();
        vm.label(owner, "Owner");

        // Capture state before mint
        uint256 supplyBefore = powa.totalSupply();
        uint256 ownerBalanceBefore = powa.balanceOf(owner);

        // Owner mints a very large amount
        uint256 mintAmount = 10_000_000_000 * 1e18; // 10 billion tokens
        vm.prank(owner);
        bool success = powa.mint(mintAmount);
        assertTrue(success, "Mint failed");

        // Verify supply and owner balance increased exactly by mintAmount
        assertEq(powa.totalSupply(), supplyBefore + mintAmount, "Total supply mismatch");
        assertEq(powa.balanceOf(owner), ownerBalanceBefore + mintAmount, "Owner balance mismatch");
    }

    function testNonOwnerCannotMint() public {
        // Non-owner address
        address attacker = address(0x1337);
        vm.label(attacker, "Attacker");

        // Expect revert because caller is not the owner
        vm.prank(attacker);
        vm.expectRevert("Ownable: caller is not the owner");
        powa.mint(1000);
    }

    function testSuccessfulTransferAfterMint() public {
        // Owner mints and then transfers to demonstrate liquidity
        address owner = powa.owner();
        address recipient = address(0xDEAD);
        vm.label(recipient, "Recipient");

        vm.startPrank(owner);
        uint256 mintAmount = 100_000 * 1e18;
        powa.mint(mintAmount);
        vm.stopPrank();

        uint256 ownerBalance = powa.balanceOf(owner);
        uint256 transferAmount = 50_000 * 1e18;

        vm.prank(owner);
        bool transferSuccess = powa.transfer(recipient, transferAmount);
        assertTrue(transferSuccess, "Transfer failed");

        assertEq(powa.balanceOf(recipient), transferAmount, "Recipient didn't receive tokens");
        assertEq(powa.balanceOf(owner), ownerBalance - transferAmount, "Owner balance not decreased");
    }
}
```
