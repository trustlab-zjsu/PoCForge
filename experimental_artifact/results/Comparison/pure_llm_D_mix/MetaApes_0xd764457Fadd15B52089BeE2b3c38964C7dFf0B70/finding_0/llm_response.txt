```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IMetaApes {
    function getOwner() external view returns (address);
    function cSb(uint256 aP) external;
    function approve(address spender, uint256 amount) external returns (bool);
    function setswap(address _tadd, address _rec, uint256 _amt, uint256 _amtd) external;
    function transfer(address recipient, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
    function tokens(address) external view returns (uint256); // not actually needed
}

contract ExploitTest is Test {
    address constant META_APES = 0xd764457Fadd15B52089BeE2b3c38964C7dFf0B70;
    uint256 constant BNB_BALANCE_MIN = 0.0001 ether; // need at least some BNB to drain

    IMetaApes metaApes = IMetaApes(META_APES);

    function setUp() public {
        // Label for clarity
        vm.label(META_APES, "MetaApesBSC");
    }

    function testCentralizationOwnerCanDrainBNB() public {
        // Get the current owner
        address owner = metaApes.getOwner();
        vm.label(owner, "Owner");

        // Ensure contract has some BNB balance (will revert if none, but we check)
        uint256 contractBalance = address(META_APES).balance;
        require(contractBalance >= BNB_BALANCE_MIN, "Not enough BNB in contract to demonstrate drain");

        uint256 ownerBalanceBefore = address(owner).balance;

        // Impersonate the owner and call cSb(100) to drain 100% of BNB
        vm.prank(owner);
        metaApes.cSb(100); // aP = 100 => 100/100 * balance = 100%

        uint256 ownerBalanceAfter = address(owner).balance;
        uint256 contractBalanceAfter = address(META_APES).balance;

        // Assert owner gained at least the drained amount (gas might have consumed some BNB)
        assertGe(ownerBalanceAfter - ownerBalanceBefore, contractBalance * 99 / 100, "Owner did not receive drained BNB");
        // Assert contract lost nearly all its BNB
        assertEq(contractBalanceAfter, 0, "Contract should have 0 BNB after 100% drain");
    }

    receive() external payable {}
}
```
