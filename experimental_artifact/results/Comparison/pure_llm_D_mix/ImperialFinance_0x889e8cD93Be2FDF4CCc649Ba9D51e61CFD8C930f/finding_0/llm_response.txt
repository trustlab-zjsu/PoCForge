pragma solidity ^0.8.12;

import "forge-std/Test.sol";

interface IImperialToken {
    function transfer(address to, uint256 value) external returns (bool);
    function withdrawAllToOwner(address _receiver) external;
    function owner() external view returns (address);
    function balanceOf(address who) external view returns (uint256);
    function _totalSupply() external view returns (uint256);
    function getCirculatingSupply() external view returns (uint256);
    function _gonBalances(address) external view returns (uint256);
    function _gonsPerFragment() external view returns (uint256);
    function claim() external;
    // other functions as needed
}

interface IBUSD {
    function balanceOf(address who) external view returns (uint256);
}

contract ExploitTest is Test {
    address constant IMPERIAL = 0x889e8cd93be2fdf4ccc649ba9d51e61cfd8c930f;
    address constant BUSD = 0xe9e7CEA3DedcA5984780Bafc599bD69ADd087D56;
    address constant OWNER = 0xE44d394BB296739739617497aaD26c9117A43c77;
    address constant ROUTER = 0x10ED43C718714eb63d5aA57B78B54704E256024E;
    address constant PAIR = 0x7213d2cCF84fb1e880E5Ef6Bf908C0ac8E5f9bE7; // from source? Not needed

    IImperialToken imperial = IImperialToken(IMPERIAL);
    IBUSD busd = IBUSD(BUSD);

    address attacker = address(0x1337); // receiver of stolen funds

    function setUp() public {
        vm.label(IMPERIAL, "ImperialToken");
        vm.label(BUSD, "BUSD");
        vm.label(OWNER, "Owner");
        vm.label(attacker, "Attacker");
    }

    // Demonstrate centralization risk: owner can drain fees via withdrawAllToOwner
    function testOwnerCanDrainContract() public {
        // Ensure contract has some token balance (simulate accumulated fees)
        uint256 amountToSend = 1000 * 1e18; // 1000 Imperial tokens
        vm.prank(OWNER);
        imperial.transfer(address(imperial), amountToSend);

        uint256 balBefore = busd.balanceOf(attacker);

        // Owner calls withdrawAllToOwner to drain to attacker
        vm.prank(OWNER);
        imperial.withdrawAllToOwner(attacker);

        uint256 balAfter = busd.balanceOf(attacker);
        assertGt(balAfter, balBefore, "Attacker should have received BUSD from swap");
    }

    // Negative test: non-owner cannot call withdrawAllToOwner
    function testNonOwnerCannotWithdraw() public {
        vm.prank(address(0xdead));
        vm.expectRevert(); // Ownable onlyOwner modifier reverts
        imperial.withdrawAllToOwner(address(this));
    }

    // Additional demonstration: owner can blacklist any contract address
    function testOwnerCanBlacklist() public {
        // Create a dummy contract to blacklist
        address dummyContract = address(new Dummy());
        vm.prank(OWNER);
        imperial.setBotBlacklist(dummyContract, true);
        assertTrue(imperial.blacklist(dummyContract), "Contract should be blacklisted");
    }
}

contract Dummy {}
