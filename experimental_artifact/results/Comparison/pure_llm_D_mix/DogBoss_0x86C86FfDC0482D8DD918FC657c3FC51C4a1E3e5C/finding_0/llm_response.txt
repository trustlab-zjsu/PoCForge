```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface ICoinToken {
    function owner() external view returns (address);
    function mint(address account, uint256 amount) external;
    function balanceOf(address owner) external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function name() external view returns (string memory);
    function symbol() external view returns (string memory);
    function decimals() external view returns (uint256);
}

contract ExploitTest is Test {
    ICoinToken constant coin = ICoinToken(0x86c86ffdc0482d8dd918fc657c3fc51c4a1e3e5c);

    function setUp() public {
        vm.label(address(coin), "CoinToken");
    }

    function testMintByOwner() public {
        address owner = coin.owner();
        vm.label(owner, "Owner");

        uint256 initialSupply = coin.totalSupply();
        uint256 initialBalance = coin.balanceOf(address(this));

        // Owner mints a large amount to this contract
        uint256 mintAmount = 1_000_000 * 10**18;
        vm.startPrank(owner);
        coin.mint(address(this), mintAmount);
        vm.stopPrank();

        uint256 finalSupply = coin.totalSupply();
        uint256 finalBalance = coin.balanceOf(address(this));

        assertEq(finalSupply, initialSupply + mintAmount, "Total supply not increased correctly");
        assertEq(finalBalance, initialBalance + mintAmount, "Balance not increased correctly");

        // Additionally demonstrate unlimited minting to any address
        vm.startPrank(owner);
        coin.mint(address(0xdead), 1 ether);
        vm.stopPrank();
        assertEq(coin.balanceOf(address(0xdead)), 1 ether, "Owner can mint to arbitrary address");
    }
}
```
