```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.4;

import "forge-std/Test.sol";

interface ICartoondoge {
    function owner() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function UpdateTaxes(uint8 burnTaxes, uint8 buybackTaxes, uint8 devTaxes, uint8 marketingTaxes, uint8 liquidityTaxes, uint8 stakingTaxes, uint8 buyTax, uint8 sellTax, uint8 transferTax) external;
    function getTaxes() external view returns (uint256 burnTax, uint256 buybackTax, uint256 devTax, uint256 marketingTax, uint256 liquidityTax, uint256 rewardsTax, uint256 buyTax, uint256 sellTax, uint256 transferTax);
    function ExcludeAccountFromFees(address account) external;
    function IncludeAccountToFees(address account) external;
}

contract ExploitTest is Test {
    address constant TARGET = 0xeb58B47404F386286b0BF09cAE3C14d8b82F65CF;
    ICartoondoge cartoondoge = ICartoondoge(TARGET);

    address user1 = address(0x1001);
    address user2 = address(0x1002);

    function setUp() public {
        vm.label(TARGET, "Cartoondoge");
        vm.label(user1, "User1");
        vm.label(user2, "User2");
    }

    function testTransferTaxExceeds25Percent() public {
        // Get the contract owner
        address owner = cartoondoge.owner();
        vm.label(owner, "Owner");

        // Step 1: Ensure user1 has some tokens (from owner, feeless)
        uint256 initialSupply = 1_000_000 * 1e18; // use a reasonable amount
        vm.prank(owner);
        cartoondoge.transfer(user1, initialSupply);

        uint256 balanceUser1Before = cartoondoge.balanceOf(user1);
        assertEq(balanceUser1Before, initialSupply);

        // Step 2: Check current transferTax (should be 15 by default)
        (,,,,,,, uint256 sellTaxBefore, uint256 transferTaxBefore) = cartoondoge.getTaxes();
        console.log("Default transferTax:", transferTaxBefore);
        assertEq(transferTaxBefore, 15);

        // Step 3: Owner sets transferTax to 50 (exceeds 25%)
        vm.prank(owner);
        cartoondoge.UpdateTaxes(10, 0, 10, 60, 20, 0, 10, 10, 50);

        (,,,,,,, sellTaxBefore, transferTaxBefore) = cartoondoge.getTaxes();
        console.log("New transferTax:", transferTaxBefore);
        assertEq(transferTaxBefore, 50);

        // Step 4: Perform a transfer from user1 to user2 (non-excluded)
        uint256 transferAmount = 1000 * 1e18;
        vm.prank(user1);
        cartoondoge.transfer(user2, transferAmount);

        uint256 balanceUser2 = cartoondoge.balanceOf(user2);
        uint256 expectedReceived = transferAmount * 50 / 100; // 50% fee, so 50% received
        assertEq(balanceUser2, expectedReceived, "User2 should have received only 50% of the transfer amount");

        // Additional assertion: the contract and burn address got the fee
        // Total fee = transferAmount * transferTax / 100 = transferAmount * 50 / 100 = half
        // So the sum of contract balance increase (including burn) should be the other half
        uint256 feeAmount = transferAmount - expectedReceived;
        // The fee is split into burn and contract; we can check that the circulating supply decreased by burn portion
        // But for proof of concept, the main assertion is that user2 didn't receive full amount.
    }
}
```
