```solidity
// Exploit.t.sol
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IBEP20 {
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function owner() external view returns (address);
    function mint(uint256 amount) external returns (bool);
    // Other functions omitted for brevity
}

contract ExploitTest is Test {
    // Deployed BEP20 token (Ethereum NFT, ENT) on BSC
    address constant TOKEN = 0xa8cba6d57d15f2ccc636054165054ec3410e76e4;
    IBEP20 token;

    function setUp() public {
        token = IBEP20(TOKEN);
        // Label for readability
        vm.label(TOKEN, "ENT Token");
    }

    function testMintInflation() public {
        // Get the current owner
        address owner = token.owner();
        vm.label(owner, "Owner");

        // Record initial state
        uint256 initialTotalSupply = token.totalSupply();
        uint256 initialOwnerBalance = token.balanceOf(owner);
        console.log("Initial totalSupply:", initialTotalSupply);
        console.log("Initial owner balance:", initialOwnerBalance);

        // Impersonate the owner and mint a large amount
        uint256 mintAmount = 1_000_000_000 * 1e18; // 1 billion tokens
        vm.prank(owner);
        bool success = token.mint(mintAmount);
        assertTrue(success, "mint failed");

        // Verify state changes
        uint256 finalTotalSupply = token.totalSupply();
        uint256 finalOwnerBalance = token.balanceOf(owner);
        console.log("Final totalSupply:", finalTotalSupply);
        console.log("Final owner balance:", finalOwnerBalance);

        // Total supply increased by mintAmount
        assertEq(finalTotalSupply, initialTotalSupply + mintAmount, "Total supply not increased");
        // Owner balance increased by mintAmount
        assertEq(finalOwnerBalance, initialOwnerBalance + mintAmount, "Owner balance not increased");
        // Demonstrate inflation: owner can mint arbitrarily
        console.log("Owner minted", mintAmount / 1e18, "tokens, causing inflation");
    }

    // Optional: verify non-owner cannot mint
    function testNonOwnerCannotMint() public {
        address attacker = address(0xBEEF);
        vm.prank(attacker);
        vm.expectRevert("Ownable: caller is not the owner");
        token.mint(100);
    }
}
```
