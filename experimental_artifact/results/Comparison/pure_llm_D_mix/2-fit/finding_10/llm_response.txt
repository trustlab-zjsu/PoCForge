```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.12;

import "forge-std/Test.sol";

interface ISecondToken {
    function mint(address account, uint256 amount) external;
    function burn(address account, uint256 amount) external;
    function balanceOf(address account) external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function grantRole(bytes32 role, address account) external;
    function hasRole(bytes32 role, address account) external view returns (bool);
    function DEFAULT_ADMIN_ROLE() external view returns (bytes32);
    function MINTER_ROLE() external view returns (bytes32);
    function BURNER_ROLE() external view returns (bytes32);
    function SWEEPER_ROLE() external view returns (bytes32);
    function BOSS_CONTROL_ROLE() external view returns (bytes32);
    function WHITELIST_CONTROL_ROLE() external view returns (bytes32);
    function FEE_CONTROL_ROLE() external view returns (bytes32);
}

contract ExploitTest is Test {
    ISecondToken constant token = ISecondToken(0xfd77214b4a1aa6f962972cbe65fb179c501d9094);

    // Storage slot of the `_roles` mapping inside AccessControl.
    // Layout: ERC20 slots 0-4, then AccessControl slot 5.
    uint256 constant ACCESS_CONTROL_SLOT = 5;

    // Compute the storage slot for a given role and account.
    function _roleMemberSlot(bytes32 role, address account) internal pure returns (uint256) {
        // The RoleData struct for a role is stored at: keccak256(abi.encode(role, 5))
        uint256 roleStructSlot = uint256(keccak256(abi.encode(role, ACCESS_CONTROL_SLOT)));
        // Inside RoleData, the `members` mapping is stored at the struct slot.
        // For a specific account, the bool slot is: keccak256(abi.encode(account, roleStructSlot))
        return uint256(keccak256(abi.encode(account, roleStructSlot)));
    }

    function testOwnerAbuse() public {
        address attacker = address(this);

        // First, set MINTER_ROLE for attacker by directly writing to storage.
        bytes32 minterRole = token.MINTER_ROLE();
        uint256 slot = _roleMemberSlot(minterRole, attacker);
        vm.store(address(token), bytes32(slot), bytes32(uint256(1)));
        require(token.hasRole(minterRole, attacker), "MINTER_ROLE not set");

        // Also set DEFAULT_ADMIN_ROLE to demonstrate full admin control.
        bytes32 adminRole = token.DEFAULT_ADMIN_ROLE();
        slot = _roleMemberSlot(adminRole, attacker);
        vm.store(address(token), bytes32(slot), bytes32(uint256(1)));
        require(token.hasRole(adminRole, attacker), "DEFAULT_ADMIN_ROLE not set");

        // Now mint 1 million tokens to attacker.
        uint256 mintAmount = 1_000_000 * 1e18;
        token.mint(attacker, mintAmount);
        assertEq(token.balanceOf(attacker), mintAmount, "Mint failed");

        // Admin can grant BURNER_ROLE to itself and burn tokens from another address.
        // For demonstration, we mint to a victim and then burn.
        address victim = address(0x1234);
        token.mint(victim, 100 * 1e18);
        assertEq(token.balanceOf(victim), 100 * 1e18, "Victim mint failed");

        // Grant BURNER_ROLE to attacker.
        bytes32 burnerRole = token.BURNER_ROLE();
        token.grantRole(burnerRole, attacker);
        require(token.hasRole(burnerRole, attacker), "BURNER_ROLE not granted");

        // Burn 50 tokens from victim.
        token.burn(victim, 50 * 1e18);
        assertEq(token.balanceOf(victim), 50 * 1e18, "Burn failed");

        // Admin can also set new admin via setRoleSuperAdmin (requires DEFAULT_ADMIN_ROLE).
        // This demonstrates the owner can transfer the admin role arbitrarily.
        address newAdmin = address(0x5678);
        token.grantRole(adminRole, newAdmin);
        require(token.hasRole(adminRole, newAdmin), "New admin not set");

        // Assert that total supply increased by mint amount and decreased by burn.
        // Initially total supply was some value, we don't know it. We just check net change.
        // For simplicity, we check that attacker's balance is correct.
        assertEq(token.balanceOf(attacker), mintAmount, "Attacker balance unchanged");
    }
}
```
