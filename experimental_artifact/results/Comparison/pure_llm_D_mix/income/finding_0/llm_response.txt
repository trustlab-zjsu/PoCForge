```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IINCOME {
    function transfer(address recipient, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
    function openTrade() external;
    function closeTrade() external;
    function isOpen() external view returns (bool);
    function owner() external view returns (address);
}

contract ExploitTest is Test {
    IINCOME constant token = IINCOME(0x3a91A88Ef4bB813BB99EF5a745bcb56e3A3b403b);
    
    function testOwnerCanStopTransactions() public {
        address owner = token.owner();
        vm.label(owner, "Owner");
        address user = address(0x1234);
        vm.label(user, "User");
        address recipient = address(0x5678);
        vm.label(recipient, "Recipient");

        // Ensure trade is open for normal users
        vm.prank(owner);
        token.openTrade();
        assertTrue(token.isOpen());

        // Fund user with some tokens from owner
        uint256 fundAmount = 1000 * 1e18;
        vm.prank(owner);
        token.transfer(user, fundAmount);
        assertEq(token.balanceOf(user), fundAmount);

        // Close trade
        vm.prank(owner);
        token.closeTrade();
        assertFalse(token.isOpen());

        // User attempts to transfer, should revert
        vm.prank(user);
        vm.expectRevert("Not Open");
        token.transfer(recipient, 100);

        // Owner can still transfer
        vm.prank(owner);
        token.transfer(user, 100);
        assertEq(token.balanceOf(user), fundAmount + 100);
    }
}
```
