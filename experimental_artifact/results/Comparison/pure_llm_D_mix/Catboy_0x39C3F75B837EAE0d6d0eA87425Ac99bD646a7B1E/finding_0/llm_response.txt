import "forge-std/Test.sol";

interface ICATBOY {
    function getOwner() external view returns (address);
    function isBlacklisted(address) external view returns (bool);
    function manage_blacklist(address[] calldata addresses, bool status) external;

contract ExploitTest is Test {
    address constant TARGET = 0x39c3f75b837eae0d6d0ea87425ac99bd646a7b1e;
    ICATBOY constant catboy = ICATBOY(TARGET);

    function setUp() public {
        vm.label(TARGET, "CATBOY");
    }

    function testCentralization_OwnerCanBlacklist() public {
        // Get the current owner from the contract
        address owner = catboy.getOwner();
        assertTrue(owner != address(0), "Owner should exist");

        // Choose a victim address
        address victim = address(0x123456789);
        assertFalse(catboy.isBlacklisted(victim), "Victim should not be blacklisted initially");

        // Owner blacklists the victim
        address[] memory victims = new address[](1);
        victims[0] = victim;
        vm.prank(owner);
        catboy.manage_blacklist(victims, true);

        // Verify that the victim is now blacklisted
        assertTrue(catboy.isBlacklisted(victim), "Victim should be blacklisted after owner action");
    }

    function testCentralization_OwnerCanClearStuckBalance() public {
        // Owner can withdraw any BNB held by the contract
        address owner = catboy.getOwner();

        // First, give some BNB to the contract (simulate stuck balance)
        uint256 initialBNB = 1 ether;
        vm.deal(TARGET, initialBNB);

        // Record owner's BNB balance before
        uint256 ownerBalanceBefore = owner.balance;

        // Owner calls clearStuckBalance to withdraw 100% of BNB
        vm.prank(owner);
        catboy.clearStuckBalance(100);

        // Check that owner received the BNB (minus gas, but we can assert >=)
        uint256 ownerBalanceAfter = owner.balance;
        assertGe(ownerBalanceAfter, ownerBalanceBefore + initialBNB, "Owner should receive stuck BNB");
    }

        // Owner can disable trading at any time
