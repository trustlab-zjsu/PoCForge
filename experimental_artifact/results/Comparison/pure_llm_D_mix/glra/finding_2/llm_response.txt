```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.10;

import "forge-std/Test.sol";

interface ICoinToken {
    function owner() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function enableBlacklist(address account) external;
    function isBlacklisted(address account) external view returns (bool);
    function exclude(address account) external;
    function decimals() external view returns (uint8);
}

contract ExploitTest is Test {
    address constant COIN_TOKEN = 0xF9881a0Bf01b1ec048506A188B06d48808087657;
    ICoinToken public token;

    address testUser = address(0x1234);
    address receiver = address(0x5678);

    function setUp() public {
        token = ICoinToken(COIN_TOKEN);

        // Label addresses for clarity in traces
        vm.label(COIN_TOKEN, "CoinToken");
        vm.label(testUser, "TestUser");
        vm.label(receiver, "Receiver");

        // Ensure testUser has tokens: impersonate owner and transfer some tokens to testUser
        address owner = token.owner();
        vm.label(owner, "Owner");

        // Check owner balance and transfer a portion to testUser
        uint256 ownerBalance = token.balanceOf(owner);
        require(ownerBalance > 0, "Owner has no tokens");

        vm.startPrank(owner);
        // Owner might be excluded from tax; transfer without tax issues
        token.transfer(testUser, 1000 * 10**token.decimals());
        vm.stopPrank();

        // Verify testUser received tokens
        assertGt(token.balanceOf(testUser), 0);
    }

    function testBlacklistPreventsTransfer() public {
        // Step 1: Owner enables blacklist for testUser
        address owner = token.owner();
        vm.prank(owner);
        token.enableBlacklist(testUser);

        // Verify testUser is blacklisted
        assertTrue(token.isBlacklisted(testUser));

        // Step 2: testUser attempts to transfer tokens – should revert
        uint256 amount = token.balanceOf(testUser);
        vm.prank(testUser);
        vm.expectRevert("CoinToken: sender blacklisted");
        token.transfer(receiver, amount);
    }
}
```
