```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.13;

import "forge-std/Test.sol";

interface IMartianDoge {
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function setBlacklist(address account, bool value) external;
    function owner() external view returns (address);
    function isBlacklist(address) external view returns (bool);
    function decimals() external view returns (uint256);
}

contract ExploitTest is Test {
    // Deployed contract address on BSC
    IMartianDoge constant token = IMartianDoge(0x0Ea9790b1385ce5efAfc273623A013D007e50254);
    address constant victim = address(0xDEADBEEF);
    address constant dummy = address(0x123456);

    function setUp() public {
        vm.label(address(token), "MartianDoge");
        vm.label(victim, "Victim");
        vm.label(dummy, "Dummy");
    }

    function testBlacklistBlocksTransfer() public {
        address owner = token.owner();
        vm.label(owner, "Owner");
        require(owner != address(0), "Owner should exist");

        // Ensure victim has tokens – transfer from owner
        uint256 ownerBalance = token.balanceOf(owner);
        require(ownerBalance > 0, "Owner has no tokens");
        uint256 decimals = token.decimals();
        uint256 transferAmount = 1000 * 10**decimals;

        vm.startPrank(owner);
        token.transfer(victim, transferAmount);
        vm.stopPrank();

        uint256 victimBalance = token.balanceOf(victim);
        assertGt(victimBalance, 0, "Victim should have tokens");

        // Victim can transfer before blacklist
        vm.startPrank(victim);
        token.transfer(dummy, 100);
        vm.stopPrank();

        // Owner blacklists victim
        vm.prank(owner);
        token.setBlacklist(victim, true);
        assertTrue(token.isBlacklist(victim), "Victim should be blacklisted");

        // Victim tries to transfer again – must revert
        vm.startPrank(victim);
        vm.expectRevert("You are banned.");
        token.transfer(dummy, 100);
        vm.stopPrank();

        // Optional: verify that attacker (owner) retains control
        // Blacklist can be undone
        vm.prank(owner);
        token.setBlacklist(victim, false);
        assertFalse(token.isBlacklist(victim), "Victim should no longer be blacklisted");

        // Now victim can transfer again
        vm.startPrank(victim);
        token.transfer(dummy, 100);
        vm.stopPrank();
    }
}
```
