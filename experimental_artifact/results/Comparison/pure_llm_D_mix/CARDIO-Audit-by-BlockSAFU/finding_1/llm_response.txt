```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.7;

import "forge-std/Test.sol";

interface ICMOVE {
    function owner() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function addBlackList(address add) external;
    function removeBlackList(address add) external;
    function addBlackLists(address[] calldata listAddress) external;
    function removeBlackLists(address[] calldata listAddress) external;
}

contract ExploitTest is Test {
    ICMOVE constant target = ICMOVE(0xdfFF8a26e4112945D9a850d63F1a864595fCbFE6);

    address victim = address(0xdead);
    address attacker = address(0xbeee); // not used, just placeholder

    function setUp() public {
        vm.label(address(target), "CMOVE");
        vm.label(victim, "victim");

        // get tokens into victim's wallet for transfer tests
        address owner = target.owner();
        vm.label(owner, "owner");

        // Distribute some tokens to victim via transfer from owner
        vm.startPrank(owner);
        uint256 amountToTransfer = 1000 * 1e18;
        target.transfer(victim, amountToTransfer);
        vm.stopPrank();

        assertEq(target.balanceOf(victim), amountToTransfer, "victim should hold tokens");
    }

    function test_ownerCanBlacklistAddress() public {
        address owner = target.owner();

        // 1. Owner blacklists victim
        vm.prank(owner);
        target.addBlackList(victim);

        // 2. Victim cannot transfer tokens (should revert with "Address is blacklisted")
        vm.expectRevert("Address is blacklisted");
        vm.prank(victim);
        target.transfer(attacker, 1 ether);

        // 3. Owner removes blacklist
        vm.prank(owner);
        target.removeBlackList(victim);

        // 4. Now victim can transfer
        vm.prank(victim);
        bool success = target.transfer(attacker, 1 ether);
        assertTrue(success, "transfer should succeed after removal");
    }

    function test_ownerCanBlacklistMultipleAddresses() public {
        address owner = target.owner();
        address[] memory victims = new address[](3);
        victims[0] = address(0xaaa);
        victims[1] = address(0xbbb);
        victims[2] = address(0xccc);

        // Mint tokens to each victim via owner transfers
        vm.startPrank(owner);
        for (uint i = 0; i < victims.length; i++) {
            target.transfer(victims[i], 1 ether);
        }
        vm.stopPrank();

        // Owner blacklists all three at once
        vm.prank(owner);
        target.addBlackLists(victims);

        // Each victim should fail to transfer
        for (uint i = 0; i < victims.length; i++) {
            vm.expectRevert("Address is blacklisted");
            vm.prank(victims[i]);
            target.transfer(attacker, 1);
        }

        // Remove blacklist for all
        vm.prank(owner);
        target.removeBlackLists(victims);

        // Now they can transfer
        for (uint i = 0; i < victims.length; i++) {
            vm.prank(victims[i]);
            bool success = target.transfer(attacker, 1);
            assertTrue(success, "transfer should succeed after mass removal");
        }
    }
}
```
