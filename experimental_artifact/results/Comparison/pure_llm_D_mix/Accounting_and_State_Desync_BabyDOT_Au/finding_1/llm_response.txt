```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.13;

import "forge-std/Test.sol";

interface IDoctorShiba {
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function owner() external view returns (address);
    function mint(address _to, uint256 _amount) external returns (bool);
    function transfer(address recipient, uint256 amount) external returns (bool);
}

contract ExploitTest is Test {
    IDoctorShiba constant doctorShiba = IDoctorShiba(0x783e086d67670c0d85c4ef34ebbee5a8253b38ee);

    address victim;
    address attacker;

    function setUp() public {
        victim = address(0x1234);
        attacker = address(0x5678);
        vm.label(address(doctorShiba), "DoctorShiba");
        vm.label(doctorShiba.owner(), "Owner");
    }

    function testOwnerCanMintAndInflateSupply() public {
        // Get initial state
        address owner = doctorShiba.owner();
        uint256 initialTotalSupply = doctorShiba.totalSupply();
        uint256 initialOwnerBalance = doctorShiba.balanceOf(owner);

        // Mint a large amount of tokens to the owner (or any address)
        uint256 mintAmount = 100_000_000 * 10**9; // 100M tokens with 9 decimals
        vm.prank(owner);
        bool success = doctorShiba.mint(owner, mintAmount);
        assertTrue(success);

        // Verify total supply increased
        uint256 newTotalSupply = doctorShiba.totalSupply();
        assertEq(newTotalSupply, initialTotalSupply + mintAmount, "Total supply not increased by mint amount");

        // Verify owner balance increased by mint amount
        uint256 newOwnerBalance = doctorShiba.balanceOf(owner);
        assertEq(newOwnerBalance, initialOwnerBalance + mintAmount, "Owner balance not increased by mint amount");

        // Verify that a random victim (who holds some tokens originally?) 
        // Since all initial supply is minted to owner, victim balance is 0.
        // Show that owner's share increased absolutely, diluting any future holders.
        // For completeness, we can also show that a non-owner cannot mint.
    }

    function testNonOwnerCannotMint() public {
        address nonOwner = victim;
        vm.prank(nonOwner);
        vm.expectRevert("Ownable: caller is not the owner");
        doctorShiba.mint(nonOwner, 1 ether);
    }
}
```
