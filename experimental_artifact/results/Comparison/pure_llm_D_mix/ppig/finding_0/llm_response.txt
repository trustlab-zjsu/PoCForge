```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;

import "forge-std/Test.sol";

interface IPPIG {
    function owner() external view returns (address);
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function approve(address, uint256) external returns (bool);
    function totalSupply() external view returns (uint256);
    function decimals() external view returns (uint8);
    function isMarketPair(address) external view returns (bool);
    function setBuyTaxes(uint256, uint256) external;
    function setsellTaxes(uint256, uint256) external;
    function GoTrade() external;
    function buyMarketingFee() external view returns (uint256);
    function buyBurnFee() external view returns (uint256);
    function totalTaxIfBuying() external view returns (uint256);
    function sellMarketingFee() external view returns (uint256);
    function sellBurnFee() external view returns (uint256);
    function totalTaxIfSelling() external view returns (uint256);
    function uniswapPair() external view returns (address);
    function isExcludedFromFee(address) external view returns (bool);
}

contract ExploitTest is Test {
    address constant PPIG_ADDR = 0x3fd7ee4a8ae0389763b080c871e1827b167c3cbd;
    IPPIG ppig = IPPIG(PPIG_ADDR);

    function setUp() public {
        // Label the contract for readability
        vm.label(PPIG_ADDR, "PPIG");
    }

    function testExceedsFeeLimit() public {
        // Get the current owner from the contract
        address owner = ppig.owner();
        vm.label(owner, "Owner");

        // 1. Turn trading on (required for transfers)
        vm.prank(owner);
        ppig.GoTrade();

        // 2. Owner sets buy taxes to 50% total (25% marketing + 25% burn)
        vm.prank(owner);
        ppig.setBuyTaxes(25, 25);
        // After this call we expect totalTaxIfBuying == 50
        uint256 totalBuyFee = ppig.totalTaxIfBuying();
        assertEq(totalBuyFee, 50, "Buy tax should be 50%");

        // 3. Try to set a total above 50% - should revert
        vm.prank(owner);
        vm.expectRevert("Max buy fee under 50%");
        ppig.setBuyTaxes(30, 30);

        // 4. Similarly for sell taxes
        vm.prank(owner);
        ppig.setsellTaxes(25, 25);
        uint256 totalSellFee = ppig.totalTaxIfSelling();
        assertEq(totalSellFee, 50, "Sell tax should be 50%");

        vm.prank(owner);
        vm.expectRevert("Max sell fee under 50%");
        ppig.setsellTaxes(25, 26);

        // === Demonstrate the economic impact ===
        // Transfer some tokens to a user and let them sell to the Uniswap pair.
        // The user will lose 50% of their tokens as tax.
        // We need the Uniswap pair address.
        address pair = ppig.uniswapPair();
        vm.label(pair, "UniswapPair");

        // Mint tokens to a test user (simulate transfer from owner)
        address user = address(0xBEEF);
        vm.label(user, "User");
        uint256 initialBalance = 1000 * 10**9; // 1000 tokens (9 decimals)
        vm.prank(owner);
        ppig.transfer(user, initialBalance);
        assertEq(ppig.balanceOf(user), initialBalance);

        // User approves the contract (or we can transfer directly)
        // The fee will be applied because user is sending to the market pair (sell).
        uint256 transferAmount = 100 * 10**9; // 100 tokens
        uint256 expectedFee = (transferAmount * 50) / 100; // 50%
        uint256 expectedReceivedByPair = transferAmount - expectedFee;

        uint256 contractBalanceBefore = ppig.balanceOf(PPIG_ADDR);
        uint256 pairBalanceBefore = ppig.balanceOf(pair);

        vm.prank(user);
        ppig.transfer(pair, transferAmount);

        // Check that the contract (fee wallet) received the fee
        uint256 contractBalanceAfter = ppig.balanceOf(PPIG_ADDR);
        assertEq(
            contractBalanceAfter - contractBalanceBefore,
            expectedFee,
            "Contract should have received the fee amount"
        );

        // Check that the pair received the net amount (only if it wasn't excluded from fee)
        // The pair is a market pair, so it is not excluded.
        uint256 pairBalanceAfter = ppig.balanceOf(pair);
        assertEq(
            pairBalanceAfter - pairBalanceBefore,
            expectedReceivedByPair,
            "Pair should receive the amount minus fee"
        );
    }
}
```
