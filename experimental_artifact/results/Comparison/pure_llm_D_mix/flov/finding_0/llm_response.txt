```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IFLOV {
    function owner() external view returns (address);
    function _maxTxAmount() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function setMaxTxPercent(uint256 maxTxAmount) external;
    function transfer(address recipient, uint256 amount) external returns (bool);
}

contract ExploitTest is Test {
    address constant FLOV = 0xca1e6f584e0d7eec74f553e9a7b64a4ded8a4b61;
    address user1 = address(0x1001);
    address user2 = address(0x2002);

    function setUp() public {
        vm.label(FLOV, "FLOV");
        vm.label(user1, "User1");
        vm.label(user2, "User2");
        // Ensure enough gas for transactions
        vm.deal(user1, 1 ether);
        vm.deal(user2, 1 ether);
    }

    function test_OwnerCanStopTransactionsForUsers() public {
        IFLOV flov = IFLOV(FLOV);
        address owner = flov.owner();
        vm.label(owner, "Owner");
        vm.deal(owner, 1 ether);

        // Owner sends some tokens to user1 (owner is excluded from fees)
        vm.prank(owner);
        flov.transfer(user1, 1000 * 1e18);

        // User1 can transfer before maxTxAmount is restricted
        uint256 user1BalanceBefore = flov.balanceOf(user1);
        vm.prank(user1);
        flov.transfer(user2, 100 * 1e18);
        uint256 user1BalanceAfter = flov.balanceOf(user1);
        assertLt(user1BalanceAfter, user1BalanceBefore, "User1 should have sent tokens");

        // Owner sets maxTxAmount to 0, effectively blocking any non-owner transfers
        vm.prank(owner);
        flov.setMaxTxPercent(0);
        assertEq(flov._maxTxAmount(), 0, "maxTxAmount should be 0");

        // User1 attempts to transfer again – should revert
        vm.prank(user1);
        vm.expectRevert("Transfer amount exceeds the maxTxAmount.");
        flov.transfer(user2, 1 * 1e18);

        // Owner can still transfer (check is bypassed for owner)
        vm.prank(owner);
        flov.transfer(user2, 1 * 1e18); // should succeed
        assertGt(flov.balanceOf(user2), 100 * 1e18, "Owner transfer succeeded");
    }
}
```
