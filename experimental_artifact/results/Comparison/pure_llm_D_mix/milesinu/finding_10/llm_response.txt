```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IMilesinu {
    function owner() external view returns (address);
    function setTaxFeePercent(uint256 taxFee) external;
    function setLiquidityFeePercent(uint256 liquidityFee) external;
    function setDevFeePercent(uint256 DevFee) external;
    function setMaxTxPercent(uint256 maxTxPercent) external;
    function setExcludeFromMaxTx(address _address, bool value) external;
    function setSwapAndLiquifyEnabled(bool _enabled) external;
    function excludeFromFee(address account) external;
    function includeInFee(address account) external;
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function _maxTxAmount() external view returns (uint256);
    function isExcludedFromFee(address account) external view returns (bool);
    function isExcludedFromMaxTx(address account) external view returns (bool);
    function setDevAddress(address payable _address) external;
    function transferOwnership(address newOwner) external;
}

contract ExploitTest is Test {
    address constant TOKEN = 0x9efeabd1ae9fe1e777ebe4d1e271510622b65711;
    IMilesinu constant token = IMilesinu(TOKEN);

    address alice = address(0x1234);
    address bob = address(0x5678);

    function setUp() public {
        vm.label(TOKEN, "Milesinu");
        // Owners can be found by querying the contract; assume deployer is initial owner
        // We'll use the actual owner from the chain
        address owner = token.owner();
        vm.label(owner, "Owner");
        vm.label(alice, "Alice");
        vm.label(bob, "Bob");

        // Fund Alice and Bob with some tokens from the owner
        uint256 supply = token.balanceOf(token.owner());
        vm.prank(token.owner());
        token.transfer(alice, 1000e9); // 1,000 tokens (9 decimals)
        vm.prank(token.owner());
        token.transfer(bob, 1000e9);
    }

    function testCentralizationOwnerCanModifyFees() public {
        address owner = token.owner();

        // Initially fees are non-zero
        // Owner sets all fees to zero
        vm.startPrank(owner);
        token.setTaxFeePercent(0);
        token.setLiquidityFeePercent(0);
        token.setDevFeePercent(0);
        vm.stopPrank();

        // Now a transfer from a fee-excluded user (owner) should incur no fee
        uint256 balBefore = token.balanceOf(bob);
        vm.prank(owner);
        token.transfer(bob, 500e9);
        uint256 balAfter = token.balanceOf(bob);
        assertEq(balAfter - balBefore, 500e9, "Owner transfer without fee should pass full amount");
    }

    function testCentralizationOwnerCanSetMaxTxToZero() public {
        address owner = token.owner();

        // Owner sets maxTx to a tiny amount
        vm.prank(owner);
        token.setMaxTxPercent(1); // 1 * 10^6 * 10^9 = 1e15 wei? Actually setMaxTxPercent does maxTxPercent * 10**6 * 10**9

        // Now a non-excluded transfer exceeding that should revert
        // Alice is not excluded from maxTx by default, so her transfer of 1000e9 should fail
        vm.expectRevert("Transfer amount exceeds the maxTxAmount.");
        vm.prank(alice);
        token.transfer(bob, 1000e9);
    }

    function testCentralizationOwnerCanExcludeFromMaxTx() public {
        address owner = token.owner();

        // Owner excludes Alice from maxTx
        vm.prank(owner);
        token.setExcludeFromMaxTx(alice, true);

        // Then set maxTx to a small amount
        vm.prank(owner);
        token.setMaxTxPercent(1); // small

        // Alice's transfer should now succeed (she is excluded)
        vm.prank(alice);
        token.transfer(bob, 1000e9);

        assertGt(token.balanceOf(bob), 0, "Bob should have received tokens");
    }

    function testCentralizationOwnerCannotBeOverruledByNonOwner() public {
        // Non-owner should not be able to change critical parameters
        vm.prank(alice);
        vm.expectRevert("Ownable: caller is not the owner");
        token.setTaxFeePercent(0);

        vm.prank(alice);
        vm.expectRevert("Ownable: caller is not the owner");
        token.setMaxTxPercent(0);

        vm.prank(alice);
        vm.expectRevert("Ownable: caller is not the owner");
        token.setSwapAndLiquifyEnabled(true);
    }

    function testCentralizationOwnerCanDrainETHviaDevAddress() public {
        address owner = token.owner();
        address dev = address(0xdead); // attacker controlled address
        vm.deal(dev, 0); // ensure fresh

        // Owner sets dev address to a controlled wallet
        vm.prank(owner);
        token.setDevAddress(payable(dev));

        // To trigger swapAndLiquify we need:
        // - swapAndLiquifyEnabled = true
        // - contract token balance >= numTokensSellToAddToLiquidity (default 50e15)
        // - from != uniswapV2Pair
        // We'll enable swap
        vm.prank(owner);
        token.setSwapAndLiquifyEnabled(true);

        // Get the contract's token balance; it may already hold some from fees.
        // If not, we can send tokens to the contract via owner transfer (from owner to token itself)
        uint256 contractBal = token.balanceOf(TOKEN);
        if (contractBal < 50e15) {
            vm.prank(owner);
            token.transfer(TOKEN, 100e15); // send 100 tokens (9 decimals) to contract
        }

        // Now trigger a transfer that will cause swapAndLiquify
        // The condition uses numTokensSellToAddToLiquidity = 50e9 (50*10^6*10^9) -> 50e9 units.
        // We'll do a small transfer from a non-excluded user to the pool? Actually from != uniswapV2Pair condition.
        // We can transfer from owner (since owner excluded from maxTx but not from this condition) to the contract? That would trigger because contract balance check sees >= numTokensSellToAddToLiquidity and from != pair.
        // But we already sent tokens to contract, so next transfer will trigger swap.
        // Any transfer from owner to any address will trigger? The condition is checked in _transfer, but it only triggers if from != uniswapV2Pair.
        // So we do a transfer from owner to Bob.
        vm.prank(owner);
        token.transfer(bob, 1e9); // small transfer

        // After swap, some ETH should be sent to dev address
        uint256 devBalance = dev.balance;
        assertGt(devBalance, 0, "Dev address should receive ETH from swap");
    }
}
```
