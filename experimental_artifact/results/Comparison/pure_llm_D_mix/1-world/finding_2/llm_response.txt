```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface WBNB {
    function deposit() external payable;
    function withdraw(uint256 wad) external;
    function transfer(address dst, uint256 wad) external returns (bool);
    function transferFrom(address src, address dst, uint256 wad) external returns (bool);
    function approve(address guy, uint256 wad) external returns (bool);
    function balanceOf(address) external view returns (uint256);
    function allowance(address, address) external view returns (uint256);
}

contract ExploitTest is Test {
    address constant WBNB_ADDR = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c;

    function testExploit() public {
        WBNB wbnb = WBNB(WBNB_ADDR);

        // Obtain some WBNB by depositing ETH
        vm.deal(address(this), 10 ether);
        wbnb.deposit{value: 1 ether}();
        uint userBalance = wbnb.balanceOf(address(this));
        require(userBalance > 0, "no balance");

        // Approve a separate attacker account to spend all
        address attacker = address(0xbad);
        vm.prank(address(this));
        wbnb.approve(attacker, type(uint256).max);

        // Now attacker can transfer all balance without any multisig confirmation
        vm.prank(attacker);
        wbnb.transferFrom(address(this), attacker, userBalance);

        assertEq(wbnb.balanceOf(attacker), userBalance);
        // Multisig bypass demonstrated: single approval and transfer executed by one address.
    }
}
```
