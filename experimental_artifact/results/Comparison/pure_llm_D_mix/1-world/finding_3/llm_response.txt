```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.13;

import "forge-std/Test.sol";

// Minimal interface for WBNB
interface IWBNB {
    function deposit() external payable;
    function withdraw(uint256 wad) external;
    function approve(address guy, uint256 wad) external returns (bool);
    function transfer(address dst, uint256 wad) external returns (bool);
    function transferFrom(address src, address dst, uint256 wad) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
    function allowance(address owner, address spender) external view returns (uint256);
    function totalSupply() external view returns (uint256);
}

// Exploit test: Demonstrates the misleading function implementation in WBNB.
// The `transfer` function is actually an alias for `transferFrom(msg.sender, dst, wad)`,
// which can be used to transfer tokens without paying any fee (there are no fees).
// More importantly, the `transferFrom` function treats an allowance of `uint(-1)` as infinite,
// effectively providing a fee‑free transfer mechanism once such an allowance is granted.
contract ExploitTest is Test {
    address constant WBNB = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c;
    IWBNB wbnb = IWBNB(WBNB);

    address alice = address(0x1111);
    address bob = address(0x2222);
    address attacker = address(0x3333);

    function setUp() public {
        // Label addresses for readability
        vm.label(WBNB, "WBNB");
        vm.label(alice, "Alice");
        vm.label(bob, "Bob");
        vm.label(attacker, "Attacker");

        // Give some native token to Alice, Bob, and attacker
        vm.deal(alice, 100 ether);
        vm.deal(bob, 100 ether);
        vm.deal(attacker, 100 ether);
    }

    function testFeeFreeTransferThroughInfiniteAllowance() public {
        // Alice deposits 10 WBNB
        vm.startPrank(alice);
        wbnb.deposit{value: 10 ether}();
        vm.stopPrank();

        assertEq(wbnb.balanceOf(alice), 10 ether);

        // Alice mistakenly approves attacker with max allowance (uint(-1))
        vm.prank(alice);
        wbnb.approve(attacker, type(uint256).max);

        assertEq(wbnb.allowance(alice, attacker), type(uint256).max);

        // Attacker now can transfer Alice's tokens to Bob without any cost (gas aside)
        vm.startPrank(attacker);
        bool success = wbnb.transferFrom(alice, bob, 5 ether);
        vm.stopPrank();

        assertTrue(success);

        // Check balances after transfer: Alice lost 5, Bob gained 5
        assertEq(wbnb.balanceOf(alice), 5 ether);
        assertEq(wbnb.balanceOf(bob), 5 ether);
        assertEq(wbnb.allowance(alice, attacker), type(uint256).max); // allowance remains intact

        // The transfer succeeded without any fee deduction, demonstrating the
        // "fee‑free transfer mechanism" described in the vulnerability report.
        // The misleading implementation is that `transfer` is actually `transferFrom`,
        // and the contract does not enforce any fee logic.
    }

    function testTransferCallsTransferFrom() public {
        // Alice deposits 10 WBNB
        vm.startPrank(alice);
        wbnb.deposit{value: 10 ether}();
        vm.stopPrank();

        // Alice calls `transfer` (which internally calls transferFrom(msg.sender, dst, wad))
        vm.prank(alice);
        bool success = wbnb.transfer(bob, 3 ether);

        assertTrue(success);
        assertEq(wbnb.balanceOf(alice), 7 ether);
        assertEq(wbnb.balanceOf(bob), 3 ether);

        // No allowance was required because `src == msg.sender` in transferFrom
        // This can be confused with a standard transfer function but actually
        // reuses the same code path as `transferFrom`.
    }
}
```
