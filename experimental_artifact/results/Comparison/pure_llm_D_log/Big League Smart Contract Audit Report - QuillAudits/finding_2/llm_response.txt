```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.5;

import "forge-std/Test.sol";

// Minimal interface for the target StandardToken contract
interface IStandardToken {
    function owner() external view returns (address);
    function forbidMint() external;
    function setFees(uint256 _liquidity, uint256 _holders, uint256 _burn, uint256 _denominator) external;
    function MINT_FORBIDDEN() external view returns (bool);
    function fees() external view returns (uint256 liquidity, uint256 holders, uint256 burn, uint256 total, uint256 denominator);
    function mint(address _who, uint256 _value) external;
    function balanceOf(address) external view returns (uint256);
    function pause() external;
    function unpause() external;
    function paused() external view returns (bool);
    function setBlacklisted(address, bool) external;
    function setPair(address, bool) external;
    function approve(address, uint256) external returns (bool);
    function transferOwnership(address) external;
}

contract ExploitTest is Test {
    // Deployed contract address on BSC
    address constant TOKEN = 0xea01d8d9eacca9996db6bb3377c1fe64308e7328;
    IStandardToken token;

    function setUp() public {
        token = IStandardToken(TOKEN);
        // Label for clarity
        vm.label(TOKEN, "BGLG_Token");
    }

    function test_forbidMint_noEvent() public {
        address owner = token.owner();
        vm.startPrank(owner);

        // Before: minting should work (if not forbidden)
        // Record logs to check for events afterward
        vm.recordLogs();
        token.forbidMint();
        // Get logs recorded during forbidMint call
        bytes32[] memory logs = vm.getRecordedLogs();
        // Assert no events were emitted (inadequate event emission)
        assertEq(logs.length, 0, "forbidMint should not emit any events");
        assertTrue(token.MINT_FORBIDDEN(), "MINT_FORBIDDEN flag must be set to true");

        // After forbidMint, mint must revert
        vm.expectRevert("mint is forbidden forever. sorry");
        token.mint(owner, 1e18);
        vm.stopPrank();
    }

    function test_setFees_noEvent() public {
        address owner = token.owner();
        vm.startPrank(owner);

        // Get initial fee values
        (uint256 liq, uint256 hold, uint256 burn, uint256 total, uint256 denom) = token.fees();

        // Set new fees
        uint256 newLiq = 1;
        uint256 newHold = 2;
        uint256 newBurn = 1;
        uint256 newDenom = 100;
        uint256 newTotal = newLiq + newHold; // 3
        require(newTotal + newBurn <= newDenom, "valid fee setting");

        vm.recordLogs();
        token.setFees(newLiq, newHold, newBurn, newDenom);
        bytes32[] memory logs = vm.getRecordedLogs();
        // Assert no events were emitted (inadequate event emission)
        assertEq(logs.length, 0, "setFees should not emit any events");

        // Verify fees changed
        (uint256 rliq, uint256 rhold, uint256 rburn, uint256 rtotal, uint256 rdenom) = token.fees();
        assertEq(rliq, newLiq, "liquidity fee mismatch");
        assertEq(rhold, newHold, "holders fee mismatch");
        assertEq(rburn, newBurn, "burn fee mismatch");
        assertEq(rtotal, newTotal, "total fee mismatch");
        assertEq(rdenom, newDenom, "denominator mismatch");

        vm.stopPrank();
    }

    // Additional test to show that both forbidMint and setFees are callable only by owner
    function test_onlyOwner() public {
        // Non-owner should not be able to call
        address nonOwner = makeAddr("attacker");
        vm.prank(nonOwner);
        vm.expectRevert();
        token.forbidMint();

        vm.prank(nonOwner);
        vm.expectRevert();
        token.setFees(1,1,1,100);
    }
}
```
