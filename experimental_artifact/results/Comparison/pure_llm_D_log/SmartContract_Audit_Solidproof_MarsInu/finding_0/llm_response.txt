pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IMARSINU {
    function owner() external view returns (address);
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function setTaxFeePercent(uint256) external;
    function setLiquidityFeePercent(uint256) external;
    function setMaxTxAmount(uint256) external;
}

contract ExploitTest is Test {
    IMARSINU public constant token = IMARSINU(0xfbbcb4bcaf3033a51e4a2ad1d904927ca0edbde7);
    address owner;
    address user1 = address(0x1);
    address user2 = address(0x2);

    function setUp() public {
        vm.label(address(token), "MARSINU");
        owner = token.owner();
        vm.label(owner, "Owner");
        vm.label(user1, "User1");
        vm.label(user2, "User2");

        // Transfer initial tokens from owner to user1
        vm.startPrank(owner);
        uint256 amount = 1000 * 1e9; // 1000 tokens (9 decimals)
        bool success = token.transfer(user1, amount);
        require(success, "Initial transfer failed");
        vm.stopPrank();
    }

    function testOwnerUnrestrictedPrivileges() public {
        // ---- 1. Normal user can transfer before parameter change ----
        uint256 user1BalBefore = token.balanceOf(user1);
        vm.prank(user1);
        bool success = token.transfer(user2, user1BalBefore);
        assertTrue(success);
        assertEq(token.balanceOf(user1), 0, "User1 should be drained");
        assertGt(token.balanceOf(user2), 0, "User2 should receive tokens");

        // ---- 2. Owner abuses privileges to disable transfers ----
        vm.startPrank(owner);
        token.setMaxTxAmount(0);          // blocks non-owner transfers
        token.setTaxFeePercent(99);       // extreme tax
        token.setLiquidityFeePercent(99); // extreme liquidity fee
        vm.stopPrank();

        // User2 tries to send even 1 token back to user1 – must revert
        vm.prank(user2);
        vm.expectRevert("Transfer amount exceeds the maxTxAmount.");
        token.transfer(user1, 1);

        // Owner can still transfer (from == owner bypasses maxTxAmount)
        vm.prank(owner);
        success = token.transfer(user2, 1);
        assertTrue(success, "Owner transfer should succeed");

        // ---- 3. If maxTxAmount were not an issue, extreme tax would be visible ----
        // For completeness, restore maxTxAmount and demonstrate high tax
        vm.startPrank(owner);
        token.setMaxTxAmount(type(uint256).max);
        vm.stopPrank();

        // Now User2 sends a small amount and loses almost everything to fees
        uint256 amountToSend = 100 * 1e9; // 100 tokens
        uint256 user2BalBefore = token.balanceOf(user2);
        vm.prank(user2);
        success = token.transfer(user1, amountToSend);
        assertTrue(success);
        // The recipient (user1) gets almost nothing because 98% fee (2% tax + 96% liquidity? actually tax=99, liq=99 => tTransferAmount = 0)
        // But since tax+liq > 100, the calculation underflows? Actually tTransferAmount = tAmount - tFee - tLiquidity, if tFee+tLiquidity > tAmount, tFee+tLiquidity > tAmount? SafeMath sub will revert. So it will revert.
        //Let's compute: tFee=99%*amount, tLiquidity=99%*amount => tFee+tLiquidity=198%*amount > amount => SafeMath sub underflow revert.
        // So we expect revert.
        // This further proves owner can break the token completely.
    }
}
