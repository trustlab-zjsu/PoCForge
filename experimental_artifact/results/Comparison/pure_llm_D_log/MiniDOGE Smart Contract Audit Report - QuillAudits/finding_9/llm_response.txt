```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.4;

import "forge-std/Test.sol";

/// @title MiniDOGE ERC20 transfer() Not Returning a Boolean – PoC
/// @notice Demonstrates that the transfer() function of the deployed MiniDOGE
///         contract does not return a boolean value, violating the ERC20 standard.
contract ExploitTest is Test {
    // Deployed contract address (as provided)
    address constant MINIDOGE = 0xba07eed3d09055d60caef2bdfca1c05792f2dfad;

    // Known addresses from source
    address constant MARKETING = 0x17d782117bf626150c109c25e8bCfe9FBb212090;
    address constant DEAD = 0x000000000000000000000000000000000000dEaD;

    // Test actors
    address owner;
    address user   = address(0x1234);
    address victim = address(0x5678);

    // Minimal interface to interact with MiniDOGE (only needed functions)
    interface IMiniDOGE {
        function owner() external view returns (address);
        function balanceOf(address account) external view returns (uint256);
        function transfer(address recipient, uint256 amount) external returns (bool);
        function totalSupply() external view returns (uint256);
    }

    IMiniDOGE public token;

    function setUp() public {
        token = IMiniDOGE(MINIDOGE);
        owner = token.owner();

        // Label for readability
        vm.label(MINIDOGE, "MiniDOGE");
        vm.label(owner, "Owner");
        vm.label(user, "User");
        vm.label(victim, "Victim");
    }

    /// @notice Prove that transfer() does not return a boolean.
    ///         The standard requires a return value (true/false) in 32 bytes.
    ///         If a Solidity caller expects a bool, it will revert because
    ///         the EVM returns no data. Here we do a low‑level call and
    ///         verify that no return data is present.
    function testTransferMissingBooleanReturn() public {
        // Ensure the owner has tokens to send
        uint256 ownerBalance = token.balanceOf(owner);
        assertGt(ownerBalance, 0, "Owner has no tokens");

        // Pick a small amount to transfer
        uint256 amount = 1000 * 10**9; // 1000 tokens (9 decimals)

        // Execute transfer from owner to user via low‑level call
        vm.prank(owner);
        (bool success, bytes memory data) = address(token).call(
            abi.encodeWithSignature("transfer(address,uint256)", user, amount)
        );

        // The call must succeed – no revert
        assertTrue(success, "transfer() call reverted unexpectedly");

        // ERC20 transfer() MUST return a bool (32 bytes). The call returned
        // empty data → no boolean return present.
        assertEq(data.length, 0, "transfer() should not return any data, but it did");

        // Additionally, verify that tokens were actually transferred (the
        // internal logic still works).
        assertEq(
            token.balanceOf(user),
            amount,
            "User should have received the tokens"
        );
    }

    /// @notice Confirm that a high‑level call reverts because Solidity
    ///         expects a return value.
    function testHighLevelCallReverts() public {
        uint256 amount = 1 ether; // any nonzero amount
        vm.prank(owner);
        // This call will revert because the function does not return a bool.
        vm.expectRevert(); // generic – any revert is enough
        token.transfer(user, amount);
    }
}
```
