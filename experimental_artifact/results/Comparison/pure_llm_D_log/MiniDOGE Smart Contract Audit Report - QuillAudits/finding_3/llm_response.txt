pragma solidity ^0.8.4;

import "forge-std/Test.sol";

interface IMiniDOGE {
    function owner() external view returns (address);
    function SetBuyBackDivisor(uint256 newDivisor) external;
    function Sweep() external;
    // add other functions only if needed
}

contract ExploitTest is Test {
    // Deployed MiniDOGE address on BSC (provided)
    address constant MINIDOGE = 0xba07eed3d09055d60caef2bdfca1c05792f2dfad;
    IMiniDOGE mini = IMiniDOGE(MINIDOGE);

    function setUp() public {
        vm.label(MINIDOGE, "MiniDOGE");
    }

    /// @notice Demonstrates that SetBuyBackDivisor (a significant owner transaction) emits no event.
    function testMissingEvent_SetBuyBackDivisor() public {
        // Impersonate the contract owner
        address owner = mini.owner();
        vm.startPrank(owner);

        // Record all logs emitted during the call
        vm.record();
        mini.SetBuyBackDivisor(5); // arbitrary divisor
        bytes32[] memory logs = vm.getRecordedLogs();

        // Assert that no event was emitted
        assertEq(logs.length, 0, "SetBuyBackDivisor should not emit any event");

        vm.stopPrank();
    }

    /// @notice Demonstrates that Sweep() (transfer of all ETH to owner) emits no event.
    function testMissingEvent_Sweep() public {
        // Impersonate the contract owner
        address owner = mini.owner();
        vm.startPrank(owner);

        // Give the contract some ETH so the transfer is non-trivial
        deal(MINIDOGE, 1 ether);

        // Record logs
        vm.record();
        mini.Sweep();
        bytes32[] memory logs = vm.getRecordedLogs();

        // Assert no events emitted
        assertEq(logs.length, 0, "Sweep should not emit any event");

        vm.stopPrank();
    }
}
