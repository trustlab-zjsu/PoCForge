```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.13;

import "forge-std/Test.sol";

interface IMiniDOGE {
    function owner() external view returns (address);
    function Sweep() external;
}

contract ExploitTest is Test {
    IMiniDOGE constant miniDoge = IMiniDOGE(0xba07eed3d09055d60caef2bdfca1c05792f2dfad);
    address owner;
    address attacker = address(0xdead);

    function setUp() public {
        // Label addresses
        vm.label(address(miniDoge), "MiniDOGE");
        owner = miniDoge.owner();
        vm.label(owner, "Owner");
        vm.label(attacker, "Attacker");
    }

    function test_CentralizationOwnerCanSweepFunds() public {
        // Ensure contract has some BNB
        uint256 initialContractBalance = 1 ether;
        deal(address(miniDoge), initialContractBalance);
        assertEq(address(miniDoge).balance, initialContractBalance, "Contract should have BNB");

        // Record owner's BNB balance before
        uint256 ownerBalanceBefore = owner.balance;

        // Owner calls Sweep() - drains all BNB from contract
        vm.startPrank(owner);
        miniDoge.Sweep();
        vm.stopPrank();

        // Assertions
        assertEq(address(miniDoge).balance, 0, "Contract BNB should be drained");
        assertEq(owner.balance, ownerBalanceBefore + initialContractBalance, "Owner should receive all BNB");
    }
}
```
