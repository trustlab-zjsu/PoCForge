```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;

import "forge-std/Test.sol";

interface IBossBaby {
    function transfer(address recipient, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
    function owner() external view returns (address);
    function tradingEnabled() external view returns (bool);
    function enableTrading() external;
    function isExcludedFromFee(address account) external view returns (bool);
}

contract ExploitTest is Test {
    address constant BOSS = 0xd8dbe5ac60cd1c0b40c9182cb3eb2392d3df371f;
    IBossBaby boss = IBossBaby(BOSS);

    address user1 = address(0x123456);
    address user2 = address(0x789012);

    function setUp() public {
        // Label for debug
        vm.label(BOSS, "BossBaby");
        vm.label(user1, "User1");
        vm.label(user2, "User2");
    }

    function test_StopsTransactions_BeforeTradingEnabled() public {
        address owner = boss.owner();
        vm.label(owner, "Owner");

        // Check initial trading state
        assertFalse(boss.tradingEnabled(), "Trading should be disabled initially");

        // Owner transfers some tokens to user1 (owner is excluded from fees)
        vm.prank(owner);
        uint256 amount = 1_000_000 * 1e18; // 1M tokens
        bool success = boss.transfer(user1, amount);
        assertTrue(success, "Owner transfer should succeed");

        // Verify user1 balance
        uint256 user1Initial = boss.balanceOf(user1);
        assertEq(user1Initial, amount, "User1 should have received tokens");

        // User1 tries to transfer to user2 - must revert
        uint256 transferAmount = 1e18;
        vm.prank(user1);
        vm.expectRevert("Trading is not enabled yet");
        boss.transfer(user2, transferAmount);

        // Owner enables trading
        vm.prank(owner);
        boss.enableTrading();
        assertTrue(boss.tradingEnabled(), "Trading should now be enabled");

        // Now user1 can transfer
        vm.prank(user1);
        success = boss.transfer(user2, transferAmount);
        assertTrue(success, "Transfer should succeed after enabling trading");

        // Verify balances
        assertEq(boss.balanceOf(user2), transferAmount, "User2 should have received tokens");
        assertEq(boss.balanceOf(user1), user1Initial - transferAmount, "User1 balance should decrease");
    }
}
```
