```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;

import "forge-std/Test.sol";

// Minimal interface for the BossBaby contract
interface IBossBaby {
    function owner() external view returns (address);
    function claimStuckTokens(address token) external;
    function balanceOf(address account) external view returns (uint256);
}

// Mock ERC20 that always returns false on transfer
contract BadERC20 {
    string public name = "BadToken";
    string public symbol = "BAD";
    uint8 public decimals = 18;
    uint256 public totalSupply = 1_000_000 * 1e18;
    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    bool public transferReturn = false; // can be toggled

    constructor() {
        balanceOf[msg.sender] = totalSupply;
    }

    function transfer(address to, uint256 amount) external returns (bool) {
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return transferReturn;
    }

    function approve(address spender, uint256 amount) external returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }

    function transferFrom(address from, address to, uint256 amount) external returns (bool) {
        require(allowance[from][msg.sender] >= amount, "allowance");
        allowance[from][msg.sender] -= amount;
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        return transferReturn;
    }
}

contract ExploitTest is Test {
    address constant BOSS_BABY = 0xd8dbe5ac60cd1c0b40c9182cb3eb2392d3df371f;
    IBossBaby boss = IBossBaby(BOSS_BABY);

    function setUp() public {
        vm.label(BOSS_BABY, "BossBaby");
    }

    function testUncheckedTransferInClaimStuckTokens() public {
        // Get the contract owner
        address owner = boss.owner();
        vm.label(owner, "Owner");

        // Deploy a token that returns false on transfer
        BadERC20 bad = new BadERC20();
        vm.label(address(bad), "BadToken");

        // Send some bad tokens to the BossBaby contract
        uint256 amountToSend = 1234 * 1e18;
        bad.transfer(BOSS_BABY, amountToSend);
        uint256 balanceBefore = bad.balanceOf(BOSS_BABY);
        assertEq(balanceBefore, amountToSend, "Tokens should be in BossBaby");

        // Verify owner has zero balance of bad tokens initially
        assertEq(bad.balanceOf(owner), 0, "Owner should have 0 bad tokens");

        // Impersonate owner and call claimStuckTokens
        vm.startPrank(owner);
        boss.claimStuckTokens(address(bad));
        vm.stopPrank();

        // Check that tokens are still in the contract (transfer silently failed)
        uint256 balanceAfter = bad.balanceOf(BOSS_BABY);
        assertEq(balanceAfter, amountToSend, "Tokens should remain in BossBaby (transfer returned false)");

        // Owner should still have zero tokens
        assertEq(bad.balanceOf(owner), 0, "Owner should not have received any tokens");

        // Also verify that the function did not revert (we already reached this point)
        // This demonstrates the unchecked return value vulnerability
    }
}
```
