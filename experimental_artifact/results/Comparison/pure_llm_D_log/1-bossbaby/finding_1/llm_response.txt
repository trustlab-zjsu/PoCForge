pragma solidity 0.8.17;

import "forge-std/Test.sol";

interface IUniswapV2Pair {
    function totalSupply() external view returns (uint);
    function getReserves() external view returns (uint112 reserve0, uint112 reserve1, uint32 blockTimestampLast);

    function swapExactTokensForETH(
        uint amountIn,
        uint amountOutMin,
        address[] calldata path,
        address to,
        uint deadline
    ) external returns (uint[] memory amounts);
    function factory() external pure returns (address);
    function WETH() external pure returns (address);
}

interface IBossBaby {
    function _liquidityFee() external view returns (uint256);
    function _marketingFee() external view returns (uint256);
    function _taxFee() external view returns (uint256);
    function uniswapV2Router() external view returns (address);
    function uniswapV2Pair() external view returns (address);
    function owner() external view returns (address);
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function totalSupply() external view returns (uint256);
    function swapTokensAtAmount() external view returns (uint256);
    function swapEnabled() external view returns (bool);
    function tradingEnabled() external view returns (bool);

contract ExploitTest is Test {
    IBossBaby public token;
    IUniswapV2Router02 public router;
    IUniswapV2Pair public pair;
    address public owner;
    address public attacker;

    function setUp() public {
        // Deployed contract address on BSC
        token = IBossBaby(0xd8dbe5ac60cd1c0b40c9182cb3eb2392d3df371f);
        
        // Retrieve router and pair from token's getters
        router = IUniswapV2Router02(token.uniswapV2Router());
        pair = IUniswapV2Pair(token.uniswapV2Pair());
        
        owner = token.owner();
        attacker = address(this); // test contract
        
        // Label addresses for clarity
        vm.label(address(token), "BossBaby");
        vm.label(address(router), "Router");
        vm.label(address(pair), "WBNB-BossBaby-Pair");
        vm.label(owner, "Owner");
        vm.label(attacker, "Attacker");

        // Enable trading (owner's privilege) and whitelist attacker from fees for simplicity
        vm.startPrank(owner);
        token.enableTrading();
        token.excludeFromFees(attacker, true);
        vm.stopPrank();
    }

    function testRedundantLiquidityNoLiquidityAdded() public {
        // Verify that liquidity fees are permanently zero
        assertEq(token._liquidityFee(), 0, "liquidity fee should be 0");
        
        // Verify that the pair's total supply is zero (no liquidity ever added)
        assertEq(pair.totalSupply(), 0, "pair total supply must be 0 because no liquidity was added");
        
        // Additionally, check that the pair reserves are zero
        (uint112 reserve0, uint112 reserve1, ) = pair.getReserves();
        assertEq(reserve0, 0, "pair token reserve should be 0");
        assertEq(reserve1, 0, "pair BNB reserve should be 0");
    }

    function testCannotSellDueToNoLiquidity() public {
        // Transfer some tokens from owner to attacker to attempt a swap
        uint256 amount = 1000 * 10**9; // 1000 tokens with 9 decimals
        vm.startPrank(owner);
        token.transfer(attacker, amount);
        vm.stopPrank();

        // Approve router to spend attacker's tokens
        token.approve(address(router), amount);

        // Prepare swap path: BossBaby -> WBNB
        address[] memory path = new address[](2);
        path[0] = address(token);
        path[1] = router.WETH();

        // Expect revert when swapping because the pair has no liquidity
        vm.expectRevert();
        router.swapExactTokensForETH(
            amount,
            0, // accept any output
            path,
            attacker,
            block.timestamp
        );
    }
}
