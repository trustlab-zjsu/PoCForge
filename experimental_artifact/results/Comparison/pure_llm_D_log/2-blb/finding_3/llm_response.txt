```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;

import "forge-std/Test.sol";

// Minimal interface for the BLBToken contract
interface IBLBToken {
    function mint(address account, uint256 amount) external;
    function cap() external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function getRoleMember(bytes32 role, uint256 index) external view returns (address);
    function getRoleMemberCount(bytes32 role) external view returns (uint256);
    function DEFAULT_ADMIN_ROLE() external view returns (bytes32);
    function MINTER_ROLE() external view returns (bytes32);
    function hasRole(bytes32 role, address account) external view returns (bool);
}

contract ExploitTest is Test {
    // Mainnet BSC address of BLBToken (from report)
    address constant TOKEN = 0x13D67Fd10BDBe8301E978e4AdCBD2c0AD26F7549;

    IBLBToken public token;
    address public admin;

    function setUp() public {
        token = IBLBToken(TOKEN);
        vm.label(TOKEN, "BLBToken");

        // Retrieve the actual admin (first member with MINTER_ROLE)
        bytes32 minterRole = token.MINTER_ROLE();
        uint256 count = token.getRoleMemberCount(minterRole);
        require(count > 0, "No MINTER_ROLE member found");
        admin = token.getRoleMember(minterRole, 0);
        vm.label(admin, "Admin");
    }

    function testOwnerCanMintArbitraryAmount() public {
        // Verify that admin has MINTER_ROLE
        assertTrue(token.hasRole(token.MINTER_ROLE(), admin));

        uint256 initialTotalSupply = token.totalSupply();
        uint256 initialBalance = token.balanceOf(address(this));

        // Admin mints 1 million tokens (1e6 * 1e18) to this contract
        uint256 mintAmount = 1_000_000 * 1e18;
        vm.prank(admin);
        token.mint(address(this), mintAmount);

        // Check effects
        assertEq(token.balanceOf(address(this)), initialBalance + mintAmount);
        assertEq(token.totalSupply(), initialTotalSupply + mintAmount);
    }

    function testNonMinterCannotMint() public {
        // Try to mint as a random address (no roles)
        address nonAdmin = address(0xBEEF);
        vm.prank(nonAdmin);
        vm.expectRevert(
            abi.encodeWithSelector(
                bytes4(keccak256("Error(string)")),
                ""
            )
        );
        token.mint(nonAdmin, 1 ether);

        // More specific revert check using AccessControl error pattern
        // Since we expect the revert message to contain "AccessControl: account"
        vm.prank(nonAdmin);
        vm.expectRevert("AccessControl: account 0x000000000000000000000000000000000000BEEF is missing role 0x9f2df0fed2c77648de5860a4cc508cd0818c85b8b8a1ab4ceeef8d981c8956a6");
        token.mint(nonAdmin, 1 ether);
    }

    function testMintBypassesFreeRoleCheck() public {
        // Ensure that mint does not charge any fee - only transfers do.
        // This test verifies minting is free of transaction fees.
        uint256 balanceBefore = token.balanceOf(address(this));

        vm.prank(admin);
        token.mint(address(this), 1000 ether);

        assertEq(token.balanceOf(address(this)), balanceBefore + 1000 ether);
    }
}
```
