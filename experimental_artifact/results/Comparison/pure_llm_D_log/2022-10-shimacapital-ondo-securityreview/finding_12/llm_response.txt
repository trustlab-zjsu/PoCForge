```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface ITetherToken {
    function owner() external view returns (address);
    function issue(uint256 amount) external;
    function transfer(address to, uint256 value) external;
    function balanceOf(address account) external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function addBlackList(address user) external;
    function isBlackListed(address user) external view returns (bool);
    function destroyBlackFunds(address user) external;
    // Not needed for the test but included for completeness
    function decimals() external view returns (uint8);
}

contract ExploitTest is Test {
    ITetherToken constant usdt = ITetherToken(0xdac17f958d2ee523a2206206994597c13d831ec7);
    address owner;

    function setUp() public {
        // Real owner of USDT on mainnet
        owner = usdt.owner();
        vm.label(owner, "Owner");
        vm.label(address(usdt), "USDT");
    }

    function testNonTokenFunctionImpact() public {
        // Ensure the contract is not paused or deprecated (default state on mainnet)
        // We will execute non-token functions to demonstrate their effect.

        // 1. Owner mints new tokens via issue()
        uint256 mintAmount = 1000 * 1e6; // USDT uses 6 decimals
        uint256 supplyBefore = usdt.totalSupply();
        vm.prank(owner);
        usdt.issue(mintAmount);
        assertEq(usdt.totalSupply(), supplyBefore + mintAmount, "Total supply should increase");
        assertEq(usdt.balanceOf(owner), mintAmount, "Owner should have minted tokens");

        // 2. Owner transfers some tokens to a victim address
        address victim = address(0xBEEF);
        uint256 victimAmount = 500 * 1e6;
        vm.prank(owner);
        usdt.transfer(victim, victimAmount);
        assertEq(usdt.balanceOf(victim), victimAmount, "Victim should receive tokens");

        // 3. Owner adds victim to blacklist
        vm.prank(owner);
        usdt.addBlackList(victim);
        assertTrue(usdt.isBlackListed(victim), "Victim should be blacklisted");

        // 4. Owner destroys blacklisted funds (non-standard behavior)
        uint256 supplyBeforeDestroy = usdt.totalSupply();
        vm.prank(owner);
        usdt.destroyBlackFunds(victim);
        assertEq(usdt.balanceOf(victim), 0, "Victim's balance should be zero after destroy");
        assertEq(usdt.totalSupply(), supplyBeforeDestroy - victimAmount, "Total supply should decrease by destroyed amount");

        // Impact demonstrated: The owner can arbitrarily destroy tokens belonging to any blacklisted address.
        // This is a non-standard function that goes beyond typical ERC20 behavior.
    }
}
```
