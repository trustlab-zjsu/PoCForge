```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.13;

import "forge-std/Test.sol";

interface IUSDT {
    function owner() external view returns (address);
    function issue(uint256 amount) external;
    function balanceOf(address who) external view returns (uint256);
    function totalSupply() external view returns (uint256);
}

contract ExploitTest is Test {
    IUSDT public usdt = IUSDT(0xdac17f958d2ee523a2206206994597c13d831ec7);

    function setUp() public {
        vm.label(address(usdt), "USDT");
        // Retrieve and label the owner for clarity
        address owner = usdt.owner();
        vm.label(owner, "Owner");
    }

    function testOwnerCanMintArbitraryTokens() public {
        // Get current state
        address owner = usdt.owner();
        uint256 supplyBefore = usdt.totalSupply();
        uint256 ownerBalanceBefore = usdt.balanceOf(owner);
        // Mint a large amount of USDT (18 decimals? USDT uses 6 decimals, but any amount works)
        uint256 mintAmount = 1_000_000 * 1e6; // 1 million USDT
        vm.prank(owner);
        usdt.issue(mintAmount);
        // Verify minting occurred correctly
        assertEq(usdt.totalSupply(), supplyBefore + mintAmount, "Total supply not increased");
        assertEq(usdt.balanceOf(owner), ownerBalanceBefore + mintAmount, "Owner balance not increased");
    }
}
```
