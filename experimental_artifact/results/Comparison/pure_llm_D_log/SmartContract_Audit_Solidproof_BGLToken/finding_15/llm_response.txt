```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.10;

import "forge-std/Test.sol";

interface IBGLToken {
    function decimals() external view returns (uint8);
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function isTradingEnabled() external view returns (bool);
    function launch() external;
    function activateTrading() external;
}

contract ExploitTest is Test {
    IBGLToken constant BGL = IBGLToken(0x225d0236Ce8C9d753dFAe9CD0265dA2d2699Ba4F);

    function testUnusedVariables() public {
        // --- 1. Confirm the existence of unused storage slots ---
        // According to the contract source, BGLToken declares:
        //   string private _name = "BGL Token";   // slot 8
        //   string private _symbol = "BGL";        // slot 9
        //   uint8 private _decimals = 18;          // slot 10
        // These are never read by the contract logic; decimals() always returns 18.
        // The slots are currently zero because the ERC20 constructor uses its own storage.

        // Read slots 8,9,10 (each is one 32-byte word for length, then data hashes)
        // Simplification: we directly check the first word of the string (length) is 0.
        // For uint8 slot 10, we read the whole slot and mask the last byte.
        bytes32 slot8 = vm.load(address(BGL), bytes32(uint256(8)));
        bytes32 slot9 = vm.load(address(BGL), bytes32(uint256(9)));
        bytes32 slot10 = vm.load(address(BGL), bytes32(uint256(10)));
        // Assert they are zero initially
        assertEq(uint256(slot8), 0, "slot8 (unused _name length) should be 0");
        assertEq(uint256(slot9), 0, "slot9 (unused _symbol length) should be 0");
        // slot10 last byte is _decimals (uint8)
        assertEq(uint8(uint256(slot10)), 0, "slot10 (_decimals) should be 0, not 18");

        // --- 2. Demonstrate that these slots are unused by overwriting them ---
        // Write arbitrary values to slots 8,9,10
        vm.store(address(BGL), bytes32(uint256(8)), bytes32(uint256(0x1234)));
        vm.store(address(BGL), bytes32(uint256(9)), bytes32(uint256(0x5678)));
        vm.store(address(BGL), bytes32(uint256(10)), bytes32(uint256(0x99))); // set _decimals to 0x99

        // --- 3. Verify that core contract logic is unaffected ---
        // decimals() should still return 18, not 0x99
        assertEq(BGL.decimals(), 18, "decimals() must remain 18 despite slot write");
        // totalSupply and balanceOf should still work
        assertTrue(BGL.totalSupply() > 0);
        // trading functions should not be broken
        assertEq(BGL.isTradingEnabled(), false, "trading should be disabled initially");
        // (Optionally call activateTrading if needed, but not necessary)

        // --- 4. Conclusion ---
        // The slots for _name, _symbol, _decimals are completely unused.
        // This confirms the presence of unused variables as flagged.
        emit log("Unused variables confirmed: slots 8,9,10 have no functional effect.");
    }
}
```
