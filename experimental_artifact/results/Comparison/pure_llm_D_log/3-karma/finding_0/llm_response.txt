// SPDX-License-Identifier: MIT
pragma solidity 0.8.17;

import "forge-std/Test.sol";

interface IERC20Extended {
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function approve(address spender, uint256 amount) external returns (bool);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
}

interface IKARMA is IERC20Extended {
    function tradingEnabled() external view returns (bool);
    function enableTrading() external;
    function owner() external view returns (address);
    function uniswapV2Pair() external view returns (address);
    function uniswapV2Router() external view returns (address);
    function setSwapEnabled(bool _enabled) external;
    function setEnableMaxTransactionLimit(bool enable) external;
    function isExcludedFromFee(address account) external view returns (bool);
    function excludeFromFees(address account, bool excluded) external;
    function setExcludeFromMaxTransactionLimit(address account, bool exclude) external;
}

contract ExploitTest is Test {
    address constant KARMA = 0x8f9ba08cbf014ab233a33067b21bedf3876c6605;
    address constant ROUTER = 0x10ED43C718714eb63d5aA57B78B54704E256024E;

    IKARMA public token;
    address public owner;
    address public user;
    address public pair;

    function setUp() public {
        token = IKARMA(KARMA);
        owner = token.owner();
        user = address(0x1234);
        pair = token.uniswapV2Pair();

        vm.label(KARMA, "KARMA");
        vm.label(ROUTER, "Router");
        vm.label(owner, "Owner");
        vm.label(user, "User");
        vm.label(pair, "Pair");

        // Disable swap logic and max transaction limit to avoid side effects
        vm.prank(owner);
        token.setSwapEnabled(false);
        vm.prank(owner);
        token.setEnableMaxTransactionLimit(false);
    }

    function testEnableTrading_TransactionsStoppedThenEnabled() public {
        // 1. Initially trading is disabled
        assertFalse(token.tradingEnabled());

        // 2. Owner transfers some tokens to user (owner is excluded from fees, so transfer works)
        uint256 transferAmount = 1000 * 1e9; // 1000 tokens with 9 decimals
        vm.prank(owner);
        token.transfer(user, transferAmount);
        uint256 balanceUserBefore = token.balanceOf(user);
        assertEq(balanceUserBefore, transferAmount);

        // 3. User tries to transfer to the Uniswap pair → should revert with "Trading is not enabled yet"
        uint256 sellAmount = 100 * 1e9;
        vm.startPrank(user);
        vm.expectRevert("Trading is not enabled yet");
        token.transfer(pair, sellAmount);
        vm.stopPrank();

        // 4. Owner enables trading
        vm.prank(owner);
        token.enableTrading();
        assertTrue(token.tradingEnabled());

        // 5. Now user can transfer to the pair
        vm.prank(user);
        token.transfer(pair, sellAmount);
        uint256 balanceUserAfter = token.balanceOf(user);
        assertLt(balanceUserAfter, balanceUserBefore, "User balance should decrease after transfer");
        assertGt(token.balanceOf(pair), 0, "Pair should have received tokens");
    }
}
