```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.15;

import "forge-std/Test.sol";

interface IEthBurn {
    function owner() external view returns (address);
    function uniswapV2Pair() external view returns (address);
    function automatedMarketMakerPairs(address) external view returns (bool);
    function setAutomatedMarketMakerPair(address pair, bool value) external;
    function DEAD() external view returns (address);
    function balanceOf(address) external view returns (uint256);
    function totalSupply() external view returns (uint256);
}

contract ExploitTest is Test {
    // Target contract address (deployed on BSC)
    address constant TARGET = 0xD1c0c25c5d7BA21df222Bd2623eCBC0CB264b5f4;
    IEthBurn ethBurn;

    function setUp() public {
        ethBurn = IEthBurn(TARGET);
        vm.label(TARGET, "EthBurn");
    }

    function testUnlockedAMMPair() public {
        // Get the current owner
        address owner = ethBurn.owner();
        vm.label(owner, "Owner");

        // Get the current Uniswap pair
        address currentPair = ethBurn.uniswapV2Pair();
        vm.label(currentPair, "UniswapPair");

        // Verify that the pair is initially set as AMM
        assertTrue(ethBurn.automatedMarketMakerPairs(currentPair), "Pair should be AMM initially");

        // Impersonate owner and modify the mapping
        vm.startPrank(owner);

        // Remove the real pair
        ethBurn.setAutomatedMarketMakerPair(currentPair, false);
        assertFalse(ethBurn.automatedMarketMakerPairs(currentPair), "Pair should no longer be AMM");

        // Add a fake pair (any address)
        address fakePair = address(0x1234);
        vm.label(fakePair, "FakePair");
        ethBurn.setAutomatedMarketMakerPair(fakePair, true);
        assertTrue(ethBurn.automatedMarketMakerPairs(fakePair), "Fake pair should be AMM now");

        // Revert to original pair for consistency (optional)
        ethBurn.setAutomatedMarketMakerPair(fakePair, false);
        ethBurn.setAutomatedMarketMakerPair(currentPair, true);

        vm.stopPrank();

        // Final assertion: mapping is back to original state
        assertTrue(ethBurn.automatedMarketMakerPairs(currentPair), "Pair should be AMM again");
        assertFalse(ethBurn.automatedMarketMakerPairs(fakePair), "Fake pair should not be AMM");
    }
}
```
