```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.15;

import "forge-std/Test.sol";

interface IEthBurn {
    function airdropToWallets(address[] calldata airdropWallets, uint256[] calldata amount) external;
    function dividendTracker() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function dividendTokenBalanceOf(address account) external view returns (uint256);
    function claim() external;
    function getAccountDividendsInfo(address account) external view returns (address, int256, int256, uint256, uint256, uint256);
    function owner() external view returns (address);
    function isExcludedFromFees(address account) external view returns (bool);
    function setPresaleWallet(address wallet) external;
    function setExcludeFees(address account, bool excluded) external;
    function setExcludeDividends(address account) external;
}

interface IEthBurnDividendTracker {
    function balanceOf(address account) external view returns (uint256);
    function withdrawableDividendOf(address account) external view returns (uint256);
    function setBalance(address account, uint256 newBalance) external;
    function processAccount(address payable account, bool automatic) external returns (bool);
}

contract ExploitTest is Test {
    address constant ETHBURN = 0xD1c0c25c5d7BA21df222Bd2623eCBC0CB264b5f4;
    IEthBurn ethBurn = IEthBurn(ETHBURN);

    // Known owner from constructor
    address constant OWNER = 0x6042F172bc1c0409c63B17db5a12a662Ab6199F5; // marketingWallet address; owner is msg.sender in constructor which is deployer

    // But deployer might be different; we'll look up using the contract's owner() 
    address owner;

    address attacker = address(0xdeadbeef);
    address victim = address(0xbeef);

    function setUp() public {
        // Determine owner
        owner = ethBurn.owner();
        vm.label(owner, "Owner");
        vm.label(attacker, "Attacker");
        vm.label(victim, "Victim");
        // Give victim some tokens via airdrop to become dividend recipient
        vm.startPrank(owner);
        address[] memory wallets = new address[](1);
        wallets[0] = victim;
        uint256[] memory amounts = new uint256[](1);
        amounts[0] = 1; // 1 token (18 decimals)
        ethBurn.airdropToWallets(wallets, amounts);
        vm.stopPrank();
    }

    function test_MediumSeverityFunctionAccess() public {
        // The report mentions medium severity functions.
        // Here we demonstrate that airdropToWallets can be used by owner
        // to arbitrarily set dividend tracker balances, which could be abused.
        // After airdrop, victim's dividend balance should reflect the token balance.
        uint256 dividendBalance = ethBurn.dividendTokenBalanceOf(victim);
        uint256 tokenBalance = ethBurn.balanceOf(victim);
        assertEq(tokenBalance, 1 * 1e18, "Victim should have received 1 token");
        // dividend tracker balance should be equal or more due to staking bonuses (none active)
        assertGt(dividendBalance, 0, "Dividend balance > 0");

        // Show that attacker cannot call airdropToWallets
        vm.prank(attacker);
        vm.expectRevert("Ownable: caller is not the owner");
        ethBurn.airdropToWallets(new address[](0), new uint256[](0));

        // Also show that processDividendTracker (external) can be called by anyone
        // but it only processes dividends, no direct impact.
        // The true vulnerability is the owner's ability to manipulate dividend balances.
        // For demonstration, we assert that the owner can call setExcludeFees etc.
        // All marked medium severity functions are owner-only.
        vm.prank(owner);
        ethBurn.setExcludeFees(victim, true);
        assertTrue(ethBurn.isExcludedFromFees(victim), "Victim excluded from fees");
    }
}
```
