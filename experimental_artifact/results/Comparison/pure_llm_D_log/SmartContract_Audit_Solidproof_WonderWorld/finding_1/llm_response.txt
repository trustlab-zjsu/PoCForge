```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

// ---------- Interfaces ----------

interface IWWOD {
    function owner() external view returns (address);
    function treasury() external view returns (address);
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function approve(address, uint256) external;
    function setTreasury(address) external;
    function setSwapAndLiquifyEnabled(bool) external;
    function setFees(uint256) external;
    function pancakeswapV2Router() external view returns (address);
    function pancakeswapV2Pair() external view returns (address);
    function _earnFee() external view returns (uint256);
    function _maxTxAmount() external view returns (uint256);
    function swapAndLiquifyEnabled() external view returns (bool);
    function swapCoolDownTime() external view returns (uint256);
    function excludeFromFee(address) external;
    function includeInFee(address) external;
}

interface IPancakeRouter {
    function swapExactTokensForETHSupportingFeeOnTransferTokens(
        uint256 amountIn,
        uint256 amountOutMin,
        address[] calldata path,
        address to,
        uint256 deadline
    ) external;
    function WETH() external view returns (address);
}

interface IWBNB {
    function deposit() external payable;
    function transfer(address, uint256) external returns (bool);
    function balanceOf(address) external view returns (uint256);
}

// ---------- Exploit Test ----------

contract ExploitTest is Test {
    address constant WWOD_ADDRESS = 0x0b5cad3cdda7d242ce16ddf81a555d5f8e3a6054;
    address constant ROUTER = 0x10ED43C718714eb63d5aA57B78B54704E256024E;
    address constant WBNB = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c;

    IWWOD public wwod;
    IPancakeRouter public router;
    IWBNB public wbnb;

    address public owner;
    address public treasury;
    address public exploitAccount;

    uint256 public constant SELL_AMOUNT = 500_000 * 1e18;  // 500,000 tokens to sell
    uint256 public constant NUM_TOKENS_TO_SWAP = 50_000 * 1e18; // contract threshold

    function setUp() public {
        wwod = IWWOD(WWOD_ADDRESS);
        router = IPancakeRouter(ROUTER);
        wbnb = IWBNB(WBNB);
        owner = wwod.owner();
        treasure = address(this); // monitor ETH balance
        exploitAccount = address(0xBAD); // dummy exploiter address

        vm.label(WWOD_ADDRESS, "WWOD");
        vm.label(ROUTER, "PancakeRouter");
        vm.label(WBNB, "WBNB");
        vm.label(owner, "Owner");
        vm.label(treasury, "Treasury");
        vm.label(exploitAccount, "Exploiter");
    }

    function test_BrokenSwapMechanism() public {
        // Step 1: Owner prepares the contract
        vm.startPrank(owner);

        // Set treasury to our test contract to monitor ETH balance
        wwod.setTreasury(treasury);

        // Enable swap-and-liquify mechanism
        wwod.setSwapAndLiquifyEnabled(true);

        // Ensure earn fee is set to 1 (default)
        // If not 1, set it
        if (wwod._earnFee() != 1) {
            wwod.setFees(1);
        }

        // Give exploitAccount a large amount of tokens (from owner's balance)
        uint256 ownerBalance = wwod.balanceOf(owner);
        require(ownerBalance >= SELL_AMOUNT * 2, "Owner not enough tokens");
        wwod.transfer(exploitAccount, SELL_AMOUNT * 2);

        // Exclude exploitAccount from fees? No, we want fees to be taken
        // But ensure exploitAccount is NOT excluded
        vm.stopPrank();

        // Step 2: ExploitAccount prepares for a large swap
        vm.startPrank(exploitAccount);
        wwod.approve(ROUTER, type(uint256).max);
        vm.stopPrank();

        // Step 3: Ensure we have enough WBNB to acquire some initial liquidity? Not needed.
        // We will simply sell tokens through the router.

        // Before any actions, record initial states
        uint256 contractTokenBefore = wwod.balanceOf(WWOD_ADDRESS);
        uint256 treasuryEthBefore = address(treasury).balance;
        uint256 exploitTokenBefore = wwod.balanceOf(exploitAccount);

        // Step 4: Perform a large sell that should trigger swapToStorehouse.
        // Conditions for conversion:
        // - contract token balance >= numTokensToSwap (50k tokens)
        // - from != pair (exploitAccount != pair)
        // - swapAndLiquifyEnabled = true
        // - block.timestamp >= lastSwapTime + swapCoolDownTime (10 seconds)
        // Initially lastSwapTime is 0, so condition holds after warp.
        vm.warp(block.timestamp + 20); // surpass cooldown

        // To accumulate contract balance, we can do a first sell (smaller) or directly
        // send tokens to contract. But we need the conversion to happen during a transfer
        // that also adds fees. The best is to perform a sell large enough so that
        // the fee (1% of amount) pushes contract balance over the threshold.
        // However, we can also pre-fund the contract with tokens to trigger conversion
        // on the very first sell.

        // Pre-fund the contract with exactly NUM_TOKENS_TO_SWAP tokens by a direct transfer.
        // This simulates previous fee accumulation.
        vm.startPrank(owner);
        wwod.transfer(WWOD_ADDRESS, NUM_TOKENS_TO_SWAP);
        vm.stopPrank();

        // Now contract balance is >= NUM_TOKENS_TO_SWAP.
        // The next transfer that meets the from!=pair condition will trigger swapToStorehouse.

        // Perform the large sell from exploitAccount to pair.
        address[] memory path = new address[](2);
        path[0] = WWOD_ADDRESS;
        path[1] = WBNB;

        vm.startPrank(exploitAccount);
        router.swapExactTokensForETHSupportingFeeOnTransferTokens(
            SELL_AMOUNT,
            0,
            path,
            exploitAccount,
            block.timestamp
        );
        vm.stopPrank();

        // Step 5: Assertions

        // Contract's token balance: it should have increased by the fee from the sell,
        // but the swapToStorehouse should have attempted to convert tokens and failed
        // due to the buggy division. Therefore, the contract balance should NOT have decreased
        // by the expected swap amount (it should still hold most of the pre-funded tokens + fee).
        uint256 contractTokenAfter = wwod.balanceOf(WWOD_ADDRESS);
        // The fee from the sell: 1% of SELL_AMOUNT = SELL_AMOUNT * 1 / 100
        uint256 expectedFee = SELL_AMOUNT / 100;
        uint256 expectedContractToken = NUM_TOKENS_TO_SWAP + expectedFee;
        // Allow a small rounding difference (1 wei)
        assertApproxEqAbs(contractTokenAfter, expectedContractToken, 1, "Contract token balance not correct");

        // Treasury ETH balance should be unchanged (or zero) because swapToStorehouse
        // would have attempted to send ETH, but no swap actually happened (tokenBalance became 0).
        uint256 treasuryEthAfter = address(treasury).balance;
        assertEq(treasuryEthAfter, treasuryEthBefore, "Treasury received unexpected ETH");

        // Additionally, we can check that the exploit account sold tokens and received WBNB.
        uint256 exploitWbnbAfter = wbnb.balanceOf(exploitAccount);
        assertGt(exploitWbnbAfter, 0, "Exploiter should have received WBNB");

        // The bug is demonstrated: the contract holds lots of collected fees but never
        // sends ETH to treasury, breaking the intended economic design.
    }
}
```
