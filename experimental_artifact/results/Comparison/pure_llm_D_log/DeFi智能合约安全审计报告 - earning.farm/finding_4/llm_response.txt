pragma solidity ^0.8.13;

import "forge-std/Test.sol";

// Minimal interface for the CRVExchange contract (only needed functions)
interface ICRVExchange {
    function owner() external view returns (address);
    function transferOwnership(address newOwner) external;
    function addPath(address dex, address[] calldata path) external;
    function removePath(address dex, address[] calldata path) external;
    function handleExtraToken(address from, address target_token, uint256 amount, uint256 min_amount) external;
    function crv_token() external view returns (address);
}

contract ExploitTest is Test {
    address constant TARGET = 0x65ff67d6a61ae6bab8821c584f5f394d35e9b50e;
    ICRVExchange exchange;

    function setUp() public {
        vm.label(TARGET, "CRVExchange");
        exchange = ICRVExchange(TARGET);
    }

    function testOwnerCanTransferOwnershipArbitrarily() public {
        // Step 1: Retrieve the current owner from storage slot 0
        address currentOwner = address(uint160(uint256(vm.load(TARGET, bytes32(uint256(0))))));
        vm.label(currentOwner, "CurrentOwner");

        // Step 2: The owner can transfer ownership to any address without restriction
        address maliciousOwner = address(0xdead);
        vm.startPrank(currentOwner);
        exchange.transferOwnership(maliciousOwner);
        vm.stopPrank();

        // Step 3: Verify that ownership has been transferred
        assertEq(exchange.owner(), maliciousOwner, "Ownership not transferred");

        // Step 4: The new owner can now perform dangerous privileged actions
        // For example: add a fake path that could be exploited
        // (This demonstrates the follow-on impact)
        vm.startPrank(maliciousOwner);
        address fakeDex = address(0xaabb);
        address[] memory fakePath = new address[](2);
        fakePath[0] = exchange.crv_token(); // from CRV
        fakePath[1] = address(0x1234);      // to some token
        // The addPath call will revert if the path already exists or if getAmountsOut fails.
        // To avoid revert we'd need to set up a real DEX. Skipping for this PoC.
        // vm.expectRevert(); // Uncomment if path is invalid
        // exchange.addPath(fakeDex, fakePath);
        vm.stopPrank();

        // The core risk: ownership can be hijacked without any timelock or community consent.
        // This PoC proves the finding that owner permissions are not properly decentralized.
    }
}
