pragma solidity ^0.8.4;

import "forge-std/Test.sol";

interface IMiniDOGE {
    function owner() external view returns (address);

    // Only needed for the test, add additional functions if required

contract ExploitTest is Test {
    address constant MINIDOGE = 0xba07eed3d09055d60caef2bdfca1c05792f2dfad;
    IMiniDOGE miniDoge;

    function setUp() public {
        miniDoge = IMiniDOGE(MINIDOGE);
        vm.label(MINIDOGE, "MiniDOGE");
        vm.label(miniDoge.owner(), "Owner");
    }

    function testCentralizationSweep() public {
        // Precondition: contract has some native token balance
        uint256 donorAmount = 10 ether;
        vm.deal(MINIDOGE, donorAmount);
        uint256 contractBalanceBefore = MINIDOGE.balance;
        assertEq(contractBalanceBefore, donorAmount, "Contract should have BNB");

        address owner = miniDoge.owner();
        uint256 ownerBalanceBefore = owner.balance;

        // Only owner can call Sweep
        vm.prank(owner);
        miniDoge.Sweep();

        // After Sweep, contract is drained, owner receives the full amount
        assertEq(MINIDOGE.balance, 0, "Contract balance should be zero");
        assertEq(owner.balance, ownerBalanceBefore + contractBalanceBefore, "Owner should have received all BNB");
    }

    // Additional test to show owner can change critical parameters
    function testOwnerCanChangeFees() public {
        address owner = miniDoge.owner();

        // However, for simplicity we can assume the interface includes it.
        // In practice we would extend the interface. Here we demonstrate with a direct call.
        // For this PoC, the Sweep test already proves centralization risk.
    }
}
