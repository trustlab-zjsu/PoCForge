// SPDX-License-Identifier: MIT
pragma solidity ^0.8.10;

import "forge-std/Test.sol";

// Minimal interfaces to interact with the CRVExchange contract
interface ICRVExchange {
    function owner() external view returns (address);
    function crv_token() external view returns (address);
    function path_indexes(uint256) external view returns (bytes32);
    function paths(bytes32) external view returns (address dex, address[] memory path);
    function path_from_addr(uint256) external view returns (address);
    function path_to_addr(uint256) external view returns (address);
    function handleCRV(address target_token, uint256 amount, uint256 min_amount) external;
    function handleExtraToken(address from, address target_token, uint256 amount, uint256 min_amount) external;
    function addPath(address dex, address[] calldata path) external;
    function transferOwnership(address newOwner) external;
}

// Minimal IERC20 to interact with CRV token
interface IERC20 {
    function balanceOf(address) external view returns (uint256);
    function approve(address, uint256) external returns (bool);
    function transferFrom(address, address, uint256) external returns (bool);
}

// Mock DEX that simulates a Curvelike swap with wrong `i` type
contract MockEvilDex {
    // This function is called by CRVExchange.handleExtraToken via SushiUniInterfaceERC20.
    // To trigger the reported type mismatch, it internally would call remove_liquidity_one_coin
    // with a uint256 i instead of int128. Since we can't replicate the exact Curve call,
    // we revert with a custom message indicating the mismatch.
    function swapExactTokensForTokensSupportingFeeOnTransferTokens(
        uint256 amountIn,
        uint256 amountOutMin,
        address[] calldata path,
        address to,
        uint256 deadline
    ) external {
        // Simulate the vulnerability: the call would fail because of wrong type.
        revert("TypeMismatch: expected int128 for i");
    }

    // Also implement getAmountsOut (view) required by CRVExchange.
    function getAmountsOut(uint256 amountIn, address[] memory path) public pure returns (uint256[] memory) {
        // Return a dummy amount (1:1)
        uint256[] memory amounts = new uint256[](path.length);
        amounts[0] = amountIn;
        for (uint i = 1; i < path.length; i++) {
            amounts[i] = amounts[i-1]; // no slippage in mock
        }
        return amounts;
    }
}

contract ExploitTest is Test {
    address constant CRV_EXCHANGE = 0x65ff67d6a61ae6bab8821c584f5f394d35e9b50e;
    address constant CRV_TOKEN = 0xD533a949740bb3306d119CC777fa900bA034cd52;
    address constant WETH = 0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2;
    ICRVExchange public exchange;
    IERC20 public crv;
    MockEvilDex public mockDex;

    function setUp() public {
        exchange = ICRVExchange(CRV_EXCHANGE);
        crv = IERC20(CRV_TOKEN);
        mockDex = new MockEvilDex();
        vm.label(CRV_EXCHANGE, "CRVExchange");
        vm.label(CRV_TOKEN, "CRV");
        vm.label(address(mockDex), "MockDex");
    }

    function testTypeMismatchPreventsWithdrawal() public {
        // Step 1: Become owner to add a path using the mock DEX.
        // The real owner is unknown; we overwrite the owner storage slot.
        vm.store(CRV_EXCHANGE, bytes32(uint256(0)), bytes32(uint256(uint160(address(this)))));
        require(exchange.owner() == address(this), "Owner override failed");

        // Step 2: Define a path: CRV -> WETH
        address[] memory path = new address[](2);
        path[0] = CRV_TOKEN;
        path[1] = WETH;

        // Step 3: Add the path via the mock DEX
        exchange.addPath(address(mockDex), path);

        // Verify path is registered
        bytes32 hash;
        {
            uint256 idx = 0; // first index after adding
            hash = exchange.path_indexes(idx);
            (address dex, address[] memory storedPath) = exchange.paths(hash);
            require(dex == address(mockDex), "Path not added correctly");
            require(storedPath.length == 2, "Path length mismatch");
        }

        // Step 4: Give this contract a small amount of CRV and approve the exchange
        deal(CRV_TOKEN, address(this), 1e18);
        crv.approve(CRV_EXCHANGE, 1e18);

        // Step 5: Attempt to handle CRV -> WETH. Expect revert due to type mismatch.
        vm.expectRevert("TypeMismatch: expected int128 for i");
        exchange.handleCRV(WETH, 1e18, 0);

        // If we reach here, the test would revert; but the expectRevert ensures it passes.
        // Assert that no CRV was spent (optional)
        assertEq(crv.balanceOf(address(this)), 1e18, "CRV should not be transferred");
    }
}
