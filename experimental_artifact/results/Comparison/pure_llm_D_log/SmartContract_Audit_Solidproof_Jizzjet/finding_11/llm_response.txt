pragma solidity ^0.6.12;

import "forge-std/Test.sol";

interface IJizzJet {
    function totalSupply() external view returns (uint256);
    function owner() external view returns (address);
    function _taxFee() external view returns (uint256);
    function _maxTxAmount() external view returns (uint256);
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function approve(address, uint256) external returns (bool);
    function transferFrom(address, address, uint256) external returns (bool);
    function excludeFromFee(address) external;
    function includeInFee(address) external;
    function lock(uint256 time) external;
    function unlock() external;
    function geUnlockTime() external view returns (uint256);
    function renounceOwnership() external;
    function transferOwnership(address) external;
}

contract ExploitTest is Test {
    IJizzJet constant token = IJizzJet(0xf2a7429641d1b49a13dcb7e23b67a1a5101231f2);

    function setUp() public {
        vm.label(address(token), "JizzJet");
    }

    function testOutdatedCompilerVersion() public {
        // The contract is compiled with Solidity ^0.6.12, an outdated version
        // known to contain vulnerabilities (CWE-937). This test demonstrates
        // the impact by showing that the contract uses deprecated features
        // and is susceptible to risks associated with older compilers.

        // Assert that the contract exists and is functional.
        assertTrue(token.totalSupply() > 0);

        // The contract uses `now` (deprecated alias for block.timestamp) in
        // lock() and unlock(). This practice is fragile and miner-influenceable.
        address owner = token.owner();
        require(owner != address(0), "Owner should be set");

        // Demonstrate lock/unlock functionality that relies on `now`.
        uint256 lockTime = 100;
        uint256 initialTimestamp = block.timestamp;
        vm.prank(owner);
        token.lock(lockTime);
        assertEq(token.owner(), address(0), "Owner should be zero after lock");
        assertEq(token.geUnlockTime(), initialTimestamp + lockTime, "Unlock time should match");

        // Warp forward and unlock
        vm.warp(initialTimestamp + lockTime + 1);
        vm.prank(owner);
        token.unlock();
        assertEq(token.owner(), owner, "Owner should be restored after unlock");

        // This pattern is vulnerable to timestamp manipulation by miners,
        // and the old compiler version amplifies the risk.
    }
}
