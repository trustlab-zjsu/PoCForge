```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.4;

import "forge-std/Test.sol";

interface ITokenTool {
    function _owner() external view returns (address);
    function _isBlacklisted(address) external view returns (bool);
    function balanceOf(address) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function multipleBotlistAddress(address[] calldata accounts, bool excluded) external;
    function isTxLimitExempt(address) external view returns (bool);
    function isExcludedFromFee(address) external view returns (bool);
    function isWalletLimitExempt(address) external view returns (bool);
    function isMarketPair(address) external view returns (bool);
    // Other functions if needed
}

contract ExploitTest is Test {
    ITokenTool public token;
    address public constant TOKEN = 0x3c6C6d32609d253d54951dB187DAEc272b1d848a;

    address public user1 = address(0xABCD);
    address public user2 = address(0xDCBA);

    function setUp() public {
        token = ITokenTool(TOKEN);
        vm.label(address(token), "TokenTool");
        vm.label(user1, "User1");
        vm.label(user2, "User2");

        // Get owner address from contract
        address owner = token._owner();
        vm.label(owner, "Owner");

        // Fund user1 with tokens from owner (owner holds all supply initially)
        uint256 ownerBalance = token.balanceOf(owner);
        require(ownerBalance > 0, "Owner has no tokens");

        vm.prank(owner);
        bool success = token.transfer(user1, 1e18);
        require(success, "Initial transfer to user1 failed");

        // Verify user1 received tokens
        assertEq(token.balanceOf(user1), 1e18, "User1 should have 1e18 tokens");
    }

    function testOwnerCanBlockTransactionsViaBlacklist() public {
        // Precondition: user1 is not blacklisted
        assertFalse(token._isBlacklisted(user1), "User1 should not be blacklisted initially");

        // Owner blacklists user1
        address owner = token._owner();
        address[] memory accounts = new address[](1);
        accounts[0] = user1;

        vm.prank(owner);
        token.multipleBotlistAddress(accounts, true);

        // Check that user1 is now blacklisted
        assertTrue(token._isBlacklisted(user1), "User1 should be blacklisted");

        // Attempt transfer from user1 -> user2 should revert
        vm.expectRevert(bytes("Blacklisted address"));
        vm.prank(user1);
        token.transfer(user2, 1e15);
    }

    function testOwnerIsNotBlacklisted() public {
        address owner = token._owner();
        assertFalse(token._isBlacklisted(owner), "Owner should not be blacklisted");
    }

    function testNonBlacklistedUserCanTransfer() public {
        // user1 was not blacklisted in setUp, can transfer to user2
        vm.prank(user1);
        bool success = token.transfer(user2, 1e15);
        assertTrue(success, "Transfer from unblocked user should succeed");

        // Verify balances
        assertEq(token.balanceOf(user2), 1e15, "User2 should receive tokens");
    }
}
```
