// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.4;

import "forge-std/Test.sol";

interface ITokenTool is IERC20 {
    function _owner() external view returns (address);
    function uniswapPair() external view returns (address);
    function isMarketPair(address) external view returns (bool);
    function isExcludedFromFee(address) external view returns (bool);
    function _totalTaxIfBuying() external view returns (uint256);
    function _totalTaxIfSelling() external view returns (uint256);
    function setBuyTaxes(uint256,uint256,uint256) external;
    function setSelTaxes(uint256,uint256,uint256) external;
    function setBuyDestFee(uint256) external;
    function setSellDestFee(uint256) external;
    function _buyDestroyFee() external view returns (uint256);
    function _sellDestroyFee() external view returns (uint256);
}

interface IERC20 {
    function totalSupply() external view returns (uint256);
    function balanceOf(address) external view returns (uint256);
    function transfer(address,uint256) external returns (bool);
    function transferFrom(address,address,uint256) external returns (bool);
    function approve(address,uint256) external returns (bool);
    function allowance(address,address) external view returns (uint256);
}

contract ExploitTest is Test {
    ITokenTool constant token = ITokenTool(0x3c6C6d32609d253d54951dB187DAEc272b1d848a);
    address user = address(0x1234);
    uint256 transferAmount = 1000 * 1e18;

    function setUp() public {
        vm.label(address(token), "TokenTool");
    }

    function testOwnerCanSetExcessBuyFees() public {
        address owner = token._owner();
        vm.startPrank(owner);

        // Set buy fees to 90% total (30+30+30)
        token.setBuyTaxes(30, 30, 30);
        uint256 buyFee = token._totalTaxIfBuying();
        assertEq(buyFee, 90, "Buy tax should be 90%");

        // Set buy destroy fee to 0 (no change)
        token.setBuyDestFee(0);
        assertEq(token._buyDestroyFee(), 0);

        vm.stopPrank();
    }

    function testOwnerCanSetExcessSellFees() public {
        address owner = token._owner();
        vm.startPrank(owner);

        // Set sell fees to 100% total (100,0,0)
        token.setSelTaxes(100, 0, 0);
        uint256 sellFee = token._totalTaxIfSelling();
        assertEq(sellFee, 100, "Sell tax should be 100%");

        // Set sell destroy fee to 0
        token.setSellDestFee(0);
        assertEq(token._sellDestroyFee(), 0);

        vm.stopPrank();
    }

    function testSellWithExcessFeeReverts() public {
        address owner = token._owner();
        address pair = token.uniswapPair();

        // Ensure user is not excluded from fee and not blacklisted
        assertEq(token.isExcludedFromFee(user), false);

        // Transfer some tokens to user
        vm.prank(owner);
        require(token.transfer(user, transferAmount), "initial transfer failed");
        assertEq(token.balanceOf(user), transferAmount);

        // Owner sets sell tax to 101% ( >100% triggers underflow in finalAmount calculation)
        vm.prank(owner);
        token.setSelTaxes(101, 0, 0); // total = 101
        // Ensure pair is market pair
        assertEq(token.isMarketPair(pair), true);

        // User tries to sell (transfer to pair) – should revert with subtraction overflow
        vm.startPrank(user);
        vm.expectRevert("SafeMath: subtraction overflow");
        token.transfer(pair, transferAmount);
        vm.stopPrank();
    }

    function testBuyWithExcessFeeReverts() public {
        address owner = token._owner();
        address pair = token.uniswapPair();

        // Transfer tokens to user for buying (user receives from pair, but easier: simulate a buy from pair to user)
        // Since we cannot make the pair call transfer, we'll use the user as sender and pair as recipient to trigger sell.
        // For buy, we need sender = pair, but we cannot control pair. So we test sell.
        // This test is covered by testSellWithExcessFeeReverts.

        // Alternatively, we can set buy fee >100 and then a user transfer to another user that is not market pair? No, only market pair triggers fee.
        // So sell case is sufficient.
    }
}
