```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IWagon {
    function mint(address to, uint256 amount) external;
    function balanceOf(address account) external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function emergencyAccount() external view returns (address);
    function MINTER_ROLE() external view returns (bytes32);
    function hasRole(bytes32 role, address account) external view returns (bool);
}

contract ExploitTest is Test {
    // Deployed Wagon contract address on Ethereum mainnet
    address constant WAGON = 0xd50c8a17d5c4b8e2d984933c7e37e5b92d687b8d;
    IWagon wagon = IWagon(WAGON);

    address attacker = address(0xdead);
    uint256 mintAmount = 1_000_000 * 1e18; // 1 million tokens with 18 decimals

    function setUp() public {
        vm.label(WAGON, "Wagon");
        vm.label(attacker, "Attacker");
        // Get the deployer/owner address from emergencyAccount()
        address owner = wagon.emergencyAccount();
        vm.label(owner, "Owner");
        // Confirm the owner has MINTER_ROLE
        bytes32 minterRole = wagon.MINTER_ROLE();
        assertTrue(wagon.hasRole(minterRole, owner), "Owner must have MINTER_ROLE");
    }

    function testOwnerCanMintTokens() public {
        // Record initial state
        uint256 initialSupply = wagon.totalSupply();
        uint256 initialBalance = wagon.balanceOf(attacker);
        address owner = wagon.emergencyAccount();

        // Impersonate the owner (who has MINTER_ROLE)
        vm.startPrank(owner);
        wagon.mint(attacker, mintAmount);
        vm.stopPrank();

        // Assert balance and supply increased
        assertEq(
            wagon.balanceOf(attacker),
            initialBalance + mintAmount,
            "Attacker balance should increase by mint amount"
        );
        assertEq(
            wagon.totalSupply(),
            initialSupply + mintAmount,
            "Total supply should increase by mint amount"
        );
    }

    function testMintInflation() public {
        // Demonstrate that owner can arbitrarily inflate supply
        uint256 initialSupply = wagon.totalSupply();
        address owner = wagon.emergencyAccount();

        // Mint a large amount to owner themselves
        uint256 extraAmount = 10_000_000 * 1e18; // 10 million tokens
        vm.prank(owner);
        wagon.mint(owner, extraAmount);

        // Supply inflated
        assertEq(wagon.totalSupply(), initialSupply + extraAmount, "Supply inflated by mint");
    }
}
```
