```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IxenoX {
    function owner() external view returns (address);
    function pancakePair() external view returns (address);
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function setSellTaxFeePercent(uint256) external;
    function setSellLiquidityFeePercent(uint256) external;
    function setSelldevFeePercent(uint256) external;
}

contract ExploitTest is Test {
    IxenoX constant token = IxenoX(0xcb9fD3784e9F369C4AD3BD4CB0d3a21aCbF23a51);
    address constant USER = address(0xBEEF);
    address owner;
    address pair;

    function setUp() public {
        owner = token.owner();
        pair = token.pancakePair();
        vm.label(address(token), "xenoX");
        vm.label(owner, "Owner");
        vm.label(pair, "PancakePair");
        vm.label(USER, "User");
    }

    function testOwnerSetsExorbitantFees() public {
        // Owner transfers some tokens to user (owner is excluded from fees)
        uint256 amount = 100_000 * 1e9; // 9 decimals
        vm.prank(owner);
        token.transfer(USER, amount);

        // Confirm user received tokens
        assertGt(token.balanceOf(USER), 0);

        // Owner sets seller fees to 99% each (total 297%)
        vm.prank(owner);
        token.setSellTaxFeePercent(99);
        vm.prank(owner);
        token.setSellLiquidityFeePercent(99);
        vm.prank(owner);
        token.setSelldevFeePercent(99);

        // User attempts to sell tokens back to PancakePair – must revert
        // because the calculated fees exceed the transfer amount (underflow)
        vm.prank(USER);
        vm.expectRevert();
        token.transfer(pair, amount);

        // The transaction is blocked, demonstrating the owner’s ability to
        // arbitrarily increase fees and disrupt trading.
    }
}
```
