```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.15;

import "forge-std/Test.sol";

interface IBabyDogeCoin20 {
    function owner() external view returns (address);
    function setMarketingWallet(address payable wallet) external;
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function getAllTaxes() external;
    function totalSupply() external view returns (uint256);
    function decimals() external view returns (uint8);
}

contract ExploitTest is Test {
    IBabyDogeCoin20 public constant token = IBabyDogeCoin20(0x5681152e9ad9628c1518266a9ab29b6d8b35693b);
    address public owner;

    function setUp() public {
        // Obtain the contract owner
        owner = token.owner();
        vm.label(owner, "Owner");
        vm.label(address(token), "BabyDogeCoin20");
        vm.label(address(0), "ZeroAddress");
    }

    function testSetMarketingWalletToZero() public {
        // Impersonate the owner
        vm.startPrank(owner);

        // Ensure marketing wallet is not already zero
        address initialMarketingWallet = address(uint160(uint256(vm.load(address(token), bytes32(uint256(6))))));
        // Slot 6 holds marketingWallet? We'll just call setter regardless.

        // Set marketing wallet to zero address
        token.setMarketingWallet(payable(address(0)));

        // Verify the variable was changed (optional, but we can check via storage)
        // We can also rely on the next assertion.

        // Transfer some tokens to the contract to simulate accumulated fees
        uint256 amountToSend = 1000 * 10**token.decimals();
        require(token.balanceOf(owner) >= amountToSend, "Owner does not have enough tokens");
        token.transfer(address(token), amountToSend);

        uint256 contractBalanceBefore = token.balanceOf(address(token));
        assertEq(contractBalanceBefore, amountToSend, "Contract balance before getTaxes");

        uint256 zeroBalanceBefore = token.balanceOf(address(0));

        // Call getAllTaxes() – should send all contract tokens to marketing wallet (now zero address)
        token.getAllTaxes();

        uint256 contractBalanceAfter = token.balanceOf(address(token));
        uint256 zeroBalanceAfter = token.balanceOf(address(0));

        // Assert tokens were transferred to zero address
        assertEq(contractBalanceAfter, 0, "Contract balance should be zero after getAllTaxes");
        assertEq(zeroBalanceAfter - zeroBalanceBefore, amountToSend, "Zero address should have received the tokens");

        vm.stopPrank();
    }
}
```
