```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.15;

import "forge-std/Test.sol";

// Minimal interface for BabyDogeCoin20
interface IBabyDogeCoin20 {
    function owner() external view returns (address);
    function totalSupply() external view returns (uint256);
    function decimals() external view returns (uint8);
    function maxTxAmount() external view returns (uint256);
    function maxWalletAmount() external view returns (uint256);
    function setMaxTx(uint256 _wholeNumber, uint256 _firstNumberAfterDecimal, uint256 _secondNumberAfterDecimal) external;
    function setMaxWallet(uint256 _wholeNumber, uint256 _firstNumberAfterDecimal, uint256 _secondNumberAfterDecimal) external;
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
}

contract ExploitTest is Test {
    IBabyDogeCoin20 public constant target = IBabyDogeCoin20(0x5681152e9ad9628c1518266a9ab29b6d8b35693b);

    function setUp() public {
        // Label the contract for debugging
        vm.label(address(target), "BabyDogeCoin20");
    }

    function testPrecisionLossInSetMaxTx() public {
        // Get the current owner to impersonate
        address owner = target.owner();
        vm.label(owner, "owner");

        // Get total supply
        uint256 totalSupply = target.totalSupply();

        // Ensure totalSupply is not neatly divisible by 10000 to observe rounding
        // If it is, we can still show that the computed value is truncated floor.
        // The vulnerability is that division before multiplication floors twice.

        // Choose a percentage: 0.01% i.e., 1 centi-percent = 1 centi-percent?
        // For simplicity, use 1 'whole' number = 1%
        // The setMaxTx function calculates: centi = _whole*100 + _first*10 + _second
        // So (1,0,0) gives centi = 100 (meaning 1%? Actually 100/10000 = 0.01 = 1%? No, 100/10000 = 0.01 = 1% is correct)
        uint256 whole = 1;
        uint256 first = 0;
        uint256 second = 0;
        uint256 centi = whole * 100 + first * 10 + second; // 100 (1%)

        // Expected if multiplication first: (centi * totalSupply) / 10000
        uint256 expectedFull = (centi * totalSupply) / 10000;

        // Impersonate owner and call setMaxTx
        vm.prank(owner);
        target.setMaxTx(whole, first, second);

        // Actual value stored
        uint256 actualMaxTx = target.maxTxAmount();

        // The contract computes: centi * (totalSupply / 10000)
        uint256 expectedContract = centi * (totalSupply / 10000);

        // Assert that actual equals the contract's (flawed) computation
        assertEq(actualMaxTx, expectedContract, "Actual maxTxAmount does not match contract's calculation");

        // Assert that the contract's result is less than or equal to the full precision value
        assertLe(actualMaxTx, expectedFull, "Contract value should not exceed full precision");

        // If totalSupply is not a multiple of 10000, we expect strict less
        if (totalSupply % 10000 != 0) {
            assertLt(actualMaxTx, expectedFull, "Precision loss should result in smaller value than exact calculation");
        }

        // Log for visualization
        emit log_named_uint("totalSupply", totalSupply);
        emit log_named_uint("expectedContract (floor floor)", expectedContract);
        emit log_named_uint("expectedFull (multiply first)", expectedFull);
        emit log_named_uint("actualMaxTx", actualMaxTx);
    }

    function testPrecisionLossInSetMaxWallet() public {
        address owner = target.owner();
        uint256 totalSupply = target.totalSupply();

        // Use same percentage for wallet limit
        uint256 whole = 1;
        uint256 first = 0;
        uint256 second = 0;
        uint256 centi = whole * 100 + first * 10 + second; // 100

        vm.prank(owner);
        target.setMaxWallet(whole, first, second);

        uint256 actualMaxWallet = target.maxWalletAmount();
        uint256 expectedContract = centi * (totalSupply / 10000);
        uint256 expectedFull = (centi * totalSupply) / 10000;

        assertEq(actualMaxWallet, expectedContract, "Actual maxWalletAmount does not match contract's calculation");
        assertLe(actualMaxWallet, expectedFull, "Contract value should not exceed full precision");

        if (totalSupply % 10000 != 0) {
            assertLt(actualMaxWallet, expectedFull, "Precision loss should result in smaller value than exact calculation");
        }

        emit log_named_uint("totalSupply", totalSupply);
        emit log_named_uint("expectedContract (floor floor)", expectedContract);
        emit log_named_uint("expectedFull (multiply first)", expectedFull);
        emit log_named_uint("actualMaxWallet", actualMaxWallet);
    }
}
```
