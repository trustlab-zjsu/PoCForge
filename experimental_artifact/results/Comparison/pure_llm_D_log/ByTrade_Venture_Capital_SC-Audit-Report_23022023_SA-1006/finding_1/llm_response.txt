```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface IByTradeToken {
    function name() external view returns (string memory);
    function symbol() external view returns (string memory);
    function decimals() external view returns (uint8);
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function approve(address spender, uint256 amount) external returns (bool);
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
    function increaseAllowance(address spender, uint256 addedValue) external returns (bool);
    function decreaseAllowance(address spender, uint256 subtractedValue) external returns (bool);
    function owner() external view returns (address);
    function pause() external;
    function unpause() external;
    function paused() external view returns (bool);
    function blacklist(address account) external view returns (bool);
    function renounceOwnership() external;
    function transferOwnership(address newOwner) external;
}

contract ExploitTest is Test {
    IByTradeToken constant token = IByTradeToken(0xe8bd1ca97f8a0582335407b40b0576df641fa94c);

    address constant ALICE = address(0x1111);
    address constant BOB = address(0x2222);

    function setUp() public {
        vm.label(address(token), "ByTradeToken");
        vm.label(ALICE, "ALICE");
        vm.label(BOB, "BOB");
    }

    function test_BEP20PauseViolation() public {
        address owner = token.owner();

        // Ensure we have a non‑owner user with tokens
        vm.startPrank(owner);
        uint256 transferAmount = 100 * 10**18;
        // Owner sends tokens to Alice (before pause – always allowed)
        token.transfer(ALICE, transferAmount);
        vm.stopPrank();

        // Record Alice's balance and mine a block to avoid mempool quirks
        uint256 aliceBalanceBefore = token.balanceOf(ALICE);
        assertEq(aliceBalanceBefore, transferAmount);

        // Now owner pauses the contract
        vm.prank(owner);
        token.pause();

        // Verify that the contract is paused
        assertTrue(token.paused());

        // Alice tries to transfer tokens – should revert because she is not the owner
        vm.startPrank(ALICE);
        vm.expectRevert(); // the whenNotPaused modifier causes revert
        token.transfer(BOB, 50 * 10**18);
        vm.stopPrank();

        // Verify Alice's balance unchanged
        uint256 aliceBalanceAfter = token.balanceOf(ALICE);
        assertEq(aliceBalanceAfter, aliceBalanceBefore);

        // Also verify the BOB received nothing
        assertEq(token.balanceOf(BOB), 0);
    }
}
```
