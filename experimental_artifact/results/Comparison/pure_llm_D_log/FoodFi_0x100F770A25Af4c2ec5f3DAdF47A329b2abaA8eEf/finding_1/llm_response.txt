```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.15;

import "forge-std/Test.sol";

interface IFoodFiToken {
    // No withdraw function exists - the vulnerability is that receive() is unrestricted
    function claimDividend() external;
    function balanceOf(address) external view returns (uint256);
}

contract ExploitTest is Test {
    address constant TOKEN = 0x100F770A25Af4c2ec5f3DAdF47A329b2abaA8eEf;
    IFoodFiToken token;

    function setUp() public {
        token = IFoodFiToken(TOKEN);
        vm.label(TOKEN, "Food_Fi_Token");
        // Fund the test contract with BNB to send
        deal(address(this), 10 ether);
    }

    function testReceiveFunctionUnprotected() public {
        uint256 balanceBefore = address(TOKEN).balance;

        // Send 1 ETH (BNB) to the contract via its unprotected receive()
        (bool success, ) = payable(TOKEN).call{value: 1 ether}("");
        require(success, "ETH transfer failed");

        uint256 balanceAfter = address(TOKEN).balance;
        assertEq(balanceAfter - balanceBefore, 1 ether, "Contract should have received 1 ETH");

        // Demonstrate that there is no public withdraw function to recover the ETH.
        // Any attempt to call a non‑existent withdraw function will revert.
        vm.expectRevert();
        (bool ok, ) = address(TOKEN).call(abi.encodeWithSignature("withdraw()"));
        // The call to withdraw() will revert – confirming the ETH is locked.
        // (The variable 'ok' is not used; we rely on expectRevert to pass.)
    }
}
```
